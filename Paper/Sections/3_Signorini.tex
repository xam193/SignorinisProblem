\section{Signorini's Problem}


\subsection{Physical and Geometric Setup of the Problem}

We consider an elastic body occupying a reference configuration $\Omega\subset\R^d,\,d\in\{2,3\}$.
Under the action of external forces and possible contact with a rigid foundation, the body deforms.
In the {\em small--strain} framework, the deformation is described by the {\em displacement field}
$u:\Omega\to\R^d$, and a material point $x\in\Omega$ is mapped to its deformed position
\begin{equation*}
	\varphi(x) := x' := x + u(x),\quad x\in\overline{\Omega}.
\end{equation*}
The aim of Signorini's problem is to determine an {\em equilibrium} displacement $u$ under
standard boundary conditions and {\em unilateral} contact constraints on the part of the boundary
that may touch the rigid foundation. Throughout the whole report, $\Omega\subset \R^d,\, d\in\{2,3\}$ denotes an open, bounded and connected {\em Lipschitz} domain. The later means, that the boundary $\Gamma:=\partial\Omega$ can locally be described as the graph of a Lipschitz continuous function. This ensures for example, that the outward normal vector $\mathrm{n}(x)$ is well--defined almost everywhere. The boundary is decomposed into three pairwise disjoint parts
\begin{equation*}
	\Gamma = \overline{\Gamma_D}\,\mathbin{\dot\cup}\,\overline{\Gamma_N}\,\mathbin{\dot\cup}\,\overline{\Gamma_C}
\end{equation*}
which are assumed to be open in $\Gamma$ and where each part is subject to different boundary conditions to model different physical interactions (See Figure \ref{fig:setup}). On $\Gamma_D$, we apply {\em homogeneous Dirichlet} boundary conditions. This forces the body to be {\em clamped} and this part, i.e.\ the displacement should not act on $\Gamma_D$,
\begin{equation*}
	\forall x\in\Gamma_D:\;x + u(x) = x \quad\Longrightarrow \quad\trace(u)|_{\Gamma_D} = 0.
\end{equation*}
The part $\Gamma_N$ may be subject to a {\em Neumann} boundary condition, which means we impose a known {\em surface force density}. $\Gamma_C$ denotes the {\em potential contact} boundary. We assume that $\Gamma_C$ and $\Gamma_D$ have positive $(d-1)$--Lebesgue measure. The rigid foundation is  positioned so that any actual contact, both initially and in the deformed equilibrium configuration, may only occur on $\Gamma_C$. While $\Gamma_C$ is prescribed a priori, the {\em active contact zone} $\Gamma_{\mathrm{act}}(u)\subset\Gamma_C$ is unknown and depends on the solution $u$. Later on, the boundary conditions imposed on $\Gamma_C$ enforce {\em unilateral} contact through {\em nonpenetration} and {\em complementarity}.

\subsection{Weak Formulation}

\subsubsection{Admissible Displacements}
We start by introducing the energy space and the admissible set of displacements.
Homogeneous Dirichlet boundary conditions on $\Gamma_D$ are incorporated by
\begin{equation}\label{eq:energyspace}
	V:=\left\{ v\in H^1(\Omega;\R^d)\;\middle|\; \trace(v)=0 \text{ on }\Gamma_D\right\}\subset H^1(\Omega;\R^d).
\end{equation}
Let $\gamma=\trace:H^1(\Omega;\R^d)\to H^{1/2}(\Gamma;\R^d)$ denote the trace operator.
For $v\in V$ we define the normal component of the boundary displacement on $\Gamma_C$ by
\begin{equation*}
	v_{\mathrm n} := (\gamma v)\cdot \mathrm n \quad \text{on }\Gamma_C,
\end{equation*}
where $\mathrm n$ is the outward unit normal on $\Gamma$. The unilateral nature of contact is expressed by the {\em nonpenetration} constraint. The rigid foundation is located on the exterior side of $\Gamma_C$ in direction of $\mathrm n$. Hence the body is not allowed to move across $\Gamma_C$ into the obstacle. In the small--strain setting this yields
\begin{equation*}
	u_{\mathrm n}\le 0 \quad \text{a.e.\ on }\Gamma_C.
\end{equation*}
This motivates the nonempty, closed and convex cone (\cite[Prop. 3.1]{book}) of admissible displacements
\begin{equation}\label{eq:admissible}
	K := \left\{ v\in V\;\middle|\; v_{\mathrm n}\le 0 \text{ a.e.\ on }\Gamma_C\right\}\subset V.
\end{equation}

\subsubsection{Small Strain Elasticity}

We start by defining the {\em small strain tensor} for a displacement function $u\in H^1(\Omega;\R^d)$, which is similar to the symmetric part of square matrices but for the gradient $\nabla u$ of $u$ and through which the deformation of the body $\overline{\Omega}$ by $u$ is quantified.
\begin{definition}[Small Strain Tensor]
	The {\tt small strain tensor} $\varepsilon$ is defined by
	\begin{equation*}
		\varepsilon(u):= \frac{1}{2}\left(\nabla u + \nabla u^\top\right)\in L^2(\Omega;\mathrm{Sym}^2(\R^d)),\quad u\in H^1(\Omega;\R^d).
	\end{equation*}
\end{definition}
We denote by $\mathrm{Sym}^2(\R^d)$ the space of symmetric second order tensors,
\begin{equation*}
	\mathrm{Sym}^2(\R^d)
	:=\left\{ s\in \R^d\otimes \R^d \;\middle|\; s^\top = s \right\}\cong \left\{ S\in \R^{d\times d}\;\middle|\; S^\top=S \right\}.
\end{equation*}
For $s,t\in \mathrm{Sym}^2(\R^d)$ we write denote by
\begin{equation*}
	s:t:=\sum_{i = 1}^d \sum_{j = 1}^d s_{ij}\,t_{ij}
\end{equation*}
If $A(x)$ is a fourth order tensor and $s\in \mathrm{Sym}^2(\R^d)$, we use the {\em double contraction}
\begin{equation*}
	(A(x):s)_{ij}:=\sum_{k = 1}^d \sum_{l = 1}^d A_{ijkl}(x)s_{kl},
	\quad
	s:A(x):s:=\sum_{i = 1}^d \sum_{j = 1}^d \sum_{k = 1}^d \sum_{l = 1}^dA_{ijkl}(x)s_{ij}s_{kl}.
\end{equation*}
In linear elasticity, stresses are assumed to depend linearly on strains. The material response is encoded in the fourth order {\em elasticity tensor} $A$, which maps a symmetric strain tensor to the corresponding {\em Cauchy stress tensor}. The tensor field $A$ is prescribed by the material and may vary in space, describing a {\em heterogeneous} body.
\begin{definition}[Elasticity tensor field]\label{def:elastfield}
	An {\tt elasticity tensor field} is a mapping
	\begin{equation*}
		A:\Omega\to \cL(\mathrm{Sym}^2(\R^d);\mathrm{Sym}^2(\R^d))
	\end{equation*}
	such that the coefficients satisfy the symmetries
	\begin{equation*}
		A_{ijkl}(x)=A_{jikl}(x)=A_{ijlk}(x)=A_{klij}(x),
		\quad \forall 1\le i,j,k,l\le d,
	\end{equation*}
	and the uniform ellipticity condition, i.e.\ there exists $\alpha_A>0$ such that for a.e.\ $x\in\Omega$
	\begin{equation*}
		s:A(x):s \ge \alpha_A\, (s:s)\quad \forall s\in \mathrm{Sym}^2(\R^d).
	\end{equation*}
	Moreover, there exists $C_A>0$ such that for a.e.\ $x\in\Omega$
	\begin{equation*}
		|A_{ijkl}(x)|\le C_A \quad \forall 1\le i,j,k,l\le d.
	\end{equation*}
\end{definition}

The {\em internal} and {\em surface forces} in the body $\overline{\Omega}$ are described by the {\em Cauchy stress tensor}.

\begin{definition}[Cauchy stress tensor]
	Let $A$ be an elasticity tensor field. For $u\in H^1(\Omega;\R^d)$ we define the
	\emph{(linear) Cauchy stress tensor} as the field $\sigma(u)\in L^2(\Omega;\mathrm{Sym}^2(\R^d))$
	given a.e.\ by
	\begin{equation*}
		\sigma(u)(x):=A(x):\varepsilon(u)(x).
	\end{equation*}
\end{definition}
On $V$, we define the symmetric bilinear form
\begin{equation}\label{eq:symbil}
	a(u,v):= \int_\Omega \sigma(u):\varepsilon(v)\;\dx,\quad u,v\in V
\end{equation}
representing the {\em virtual work of internal forces} and the linear bounded functional
\begin{equation}\label{eq:laodfunc}
	\ell(v):= \int_\Omega f\cdot v\;\dx  + \int_{\Gamma_N} F\cdot \gamma v\;\ds,\quad  v\in V,
\end{equation}
with {\em volume force density} $f\in L^2(\Omega;\R^d)$ on the body and {\em surface loads} $F\in L^2(\Gamma_N;\R^d)$ on the Neumann boundary, representing the {\em virtual work of external forces}. In the following, we want to prove standard properties of $a$ and $\ell$. For this, we will need the following inequalities. The first inequality is a result of {\em Korn's second inequality} (\cite[Theorem 3.3]{book} and \cite[Section 3.5.2]{book}) and the {\em Peetre--Tartar} Lemma~(\cite[Lemma~3.5]{book}). Therefore only the proof of the second one is given.

\begin{lemma}\label{lem:korn}
	Assuming $|\Gamma_D|>0$, there exists a constant $C_K>0$ such that
	\begin{equation*}
		C_K^{-1}\,\|v\|_{1,\Omega} \le \|\varepsilon(v)\|_{0,\Omega}\le \|v\|_{1,\Omega} \quad \forall v\in V.
	\end{equation*}
\end{lemma}
\begin{proof}
	We prove the second inequality. For all $v\in V$ we have
	\begin{equation*}
		|\varepsilon(v)(x)| = \left|\frac{1}{2}(\nabla v(x) +\nabla v^\top)\right|\le \frac{1}{2}(|\nabla v(x)|+|\nabla v(x)^\top|) = |\nabla v(x)|
	\end{equation*}
	and thus
	\begin{equation*}
		\|\varepsilon(v)\|_{0,\Omega}^2 =\int_\Omega |\varepsilon(v)(x)|^2\;\dx \le\int_\Omega |\nabla v(x)|^2\;\dx = \|\nabla v\|_{0,\Omega}^2\le \|v\|_{1,\Omega}^2,
	\end{equation*}
	where the last inequality holds due to the definition
	\begin{equation*}
		\|v\|_{1,\Omega}^2 = \|v\|_{0,\Omega}^2 + \|\nabla v\|_{0,\Omega}^2.
	\end{equation*}
\end{proof}

\begin{proposition}\label{pr:aproperties}
	The symmetric bilinear form $a:V\times V\to \R$ defined in \eqref{eq:symbil} is
	\begin{itemize}
		\item[i)] {\em bounded}, i.e.\ for all $v,w\in V$ there exists $C_1<\infty$ such that
		      \begin{equation*}
			      |a(u,v)|\le C_1\,\|u\|_{1,\Omega}\,\|v\|_{1,\Omega}\quad \text{with}\quad C_1 = C_A,
		      \end{equation*}
		\item[ii)] $V${\em -elliptic}, i.e.\ for all $v\in V\setminus\{0\}$ there exist $C_2>0$ such that
		      \begin{equation*}
			      a(v,v)\ge C_2\,\|v\|_{1,\Omega}^2\quad\text{with}\quad C_2 = \frac{\alpha_A}{C_K^2}.
		      \end{equation*}
	\end{itemize}
	Moreover, the linear functional $\ell$ defined in \eqref{eq:laodfunc} is {\em bounded}, i.e.\ for all $v\in V$ there exists $C_3>0$ such that
	\begin{equation}\label{eq:ellbound}
		|\ell (v)|\le C_3\,\|v\|_{1,\Omega}\quad\text{with}\quad C_3 = \max\{1,C_\mathrm{tr}\}\left(\|f\|_{0,\Omega}+\|F\|_{0,\Gamma_N}\right)\|v\|_{1,\Omega}.
	\end{equation}
\end{proposition}
\begin{proof}
	Let $v,w\in V$ be arbitrary. Then
	\begin{equation*}
		|a(v,w)|\le \|A:\varepsilon(v)\|_{0,\Omega}\,\|\varepsilon(w)\|_{0,\Omega}
	\end{equation*}
	by the Cauchy--Schwarz inequality. With Definition \ref{def:elastfield} there exists $C_A>0$ such that
	\begin{equation*}
		\|A:\varepsilon(v)\|_{0,\Omega}\le C_A\,\|\varepsilon(v)\|_{0,\Omega}.
	\end{equation*}
	Combining this with the second inequality form Lemma \ref{lem:korn} we have
	\begin{equation*}
		|a(v,w)|\le C_A\,\|\varepsilon(v)\|_{0,\Omega}\,\|\varepsilon(w)\|_{0,\Omega}\le C_A\,\|v\|_{1,\Omega}\,\|w\|_{1,\Omega}.
	\end{equation*}
	The ellipticity of $a$ follows by
	\begin{equation*}
		a(v,v) = \int_\Omega \varepsilon(v):A:\varepsilon(v)\;\dx \ge \alpha_A\int_\Omega\varepsilon(v):\varepsilon(v) = \alpha_A\,\|\varepsilon(v)\|^2_{0,\Omega},
	\end{equation*}
	where $\alpha_A>0$ exists due to Definition \ref{def:elastfield}. The first inequality of Lemma \ref{lem:korn} finally yields
	\begin{equation*}
		a(v,v)\ge \alpha_A\,\|\varepsilon(v)\|_{0,\Omega}^2\ge \alpha_A\,\left(C_K^{-1}\,\|v\|_{1,\Omega}\right)^2 = \frac{\alpha_A}{C_K^2}\,\|v\|_{1,\Omega}^2.
	\end{equation*}
	Again by Cauchy--Schwarz inequality,
	\begin{equation*}
		|\ell(v)|\le \|f\|_{0,\Omega}\,\|v\|_{0,\Omega}+\|F\|_{0,\Gamma_N}\,\|\gamma v\|_{0,\Gamma_N}\quad \forall v\in V.
	\end{equation*}
	By the Trace Theorem \ref{th:tracetheorem} there exists $C_\mathrm{tr}>0$ such that
	\begin{equation*}
		\|\gamma v\|_{0,\Gamma_N}\le \|\gamma v\|_{0,\Gamma}\le \|\gamma v\|_{\tfrac12,\Gamma}\le C_\mathrm{tr}\,\|v\|_{1,\Omega} \quad \forall v\in V.
	\end{equation*}
	Finally,
	\begin{equation*}
		|\ell(v)|\le \left(\|f\|_{0,\Omega} + C_{\mathrm{tr}}\,\|F\|_{0,\Gamma_N}\right)\|v\|_{1,\Omega}\le \max\{1,C_\mathrm{tr}\}\,\left(\|f\|_{0,\Omega}+\|F\|_{0,\Gamma_N}\right)\|v\|_{1,\Omega}\quad \forall v\in V.
	\end{equation*}
\end{proof}

\subsubsection{Energy Minimization and Well-posedness}

The total potential energy of the elastic body is given by the quadratic functional
\begin{equation}\label{eq:energyfunc}
	\cJ(v):=\frac{1}{2}\,a(v,v)-\ell(v),\quad v\in V.
\end{equation}
In the presence of unilateral contact, the admissible displacements are restricted to the closed convex set $K\subset V$. We therefore consider the following constrained minimization problem.
\begin{problem}[Signorini's Problem as Energy Minimization]\label{pr:signorini1}
Find $u\in K$ such that
\begin{equation*}
	\cJ(u)=\min_{v\in K}\cJ(v) = \min_{v\in K}\,\frac{1}{2}\,a(v,v)-\ell(v).
\end{equation*}
\end{problem}

We now what to prove well-posedness of the above Problem \ref{pr:signorini1}. This consist of existence and uniqueness of a solution and continuous dependency of the solution on the load functional $\ell$. We start by the uniqueness, which follows directly by the strict convexity of $\cJ$ and the convexity of $K$.

\begin{lemma}[Strict Convexity of $\cJ$]\label{lem:Jconvex}
	The Functional $\cJ$ defined in \eqref{eq:energyfunc} is strictly convex.
\end{lemma}
\begin{proof}
	For $u,v\in V$ and $t\in[0,1]$, we set $w_t:=(1-t)u+tv$. Then
	\begin{equation*}
		a(w_t,w_t) = (1-t)a(u,u)+ta(v,v)-t(1-t)a(u-v,u-v)
	\end{equation*}
	and
	\begin{equation*}
		\ell(w_t) = (1-t)\ell(u)+t\ell(v).
	\end{equation*}
	Together, we have
	\begin{equation*}
		\cJ(w_t) = (1-t)\cJ(u)+t\cJ(v)-\frac{1}{2}\,t(1-t)a(u-v,u-v).
	\end{equation*}
	If $u\neq v$ then $u-v\neq 0$ and thus by $V$-ellipticity of $a$ we have $a(u-v,u-v)>0$ and $t(1-t)>0$. This implies
	\begin{equation*}
		\cJ(w_t)<(1-t)\cJ(u)+t\cJ(v).
	\end{equation*}
\end{proof}

\begin{proposition}[Uniqueness of Minimizer of $\cJ$]\label{pr:Junique}
	If there exists a minimizer of the functional $\cJ$ defined in \eqref{eq:energyfunc}, then it is unique.
\end{proposition}
\begin{proof}
	Assume that there exist two minimizers $u_1,u_2\in K$ of $\cJ$ on $K$, i.e.
	\begin{equation}\label{eq:u1u2min}
		\cJ(u_1)=\min_{v\in K}\cJ(v)=\cJ(u_2).
	\end{equation}
	If $u_1=u_2$ there is nothing to show. Hence assume $u_1\neq u_2$.
	Since $K$ is convex, for every $t\in(0,1)$ the convex combination
	\begin{equation*}
		w_t := (1-t)u_1+tu_2
	\end{equation*}
	belongs to $K$. By Lemma \ref{lem:Jconvex}, $\cJ$ is strictly convex and therefore we have
	\begin{equation*}
		\cJ(w_t) < (1-t)\cJ(u_1)+t\cJ(u_2)\quad \forall t\in(0,1).
	\end{equation*}
	Using \eqref{eq:u1u2min}, this yields
	\begin{equation*}
		\cJ(w_t) < (1-t)\min_{v\in K}\cJ(v)+t\min_{v\in K}\cJ(v)=\min_{v\in K}\cJ(v).
	\end{equation*}
	This contradicts the definition of the minimum, since $w_t\in K$.
	Therefore the assumption $u_1\neq u_2$ is false and we conclude $u_1=u_2$.
\end{proof}
It remains to show the existence of at least one minimizer. For this, we verify {\em coercivity} and {\em weakly lower semicontinuity} of the functional $\cJ$. The proof requires the following lemma.

\begin{lemma}[Weak lower semicontinuity of the norm]\label{lem:normwlsc}
	Let $(X,\|\cdot\|_X)$ be a normed space. Then $\|\cdot\|_X:X\to \R_{\ge 0}$ is a weakly lower semicontinuous functional, i.e.\ if $x_j\rightharpoonup x$ in $X$, then
	\begin{equation*}
		\|x\|_X \le \liminf_{j\to\infty}\|x_j\|_X.
	\end{equation*}
\end{lemma}
\begin{proof}
	Let $J:X\to X^{**}$ be the canonical embedding defined by
	\begin{equation*}
		(Jx)(\varphi):=\varphi(x)\quad \forall \varphi\in X^*.
	\end{equation*}
	By the Hahn--Banach theorem, $J$ is an isometry, i.e.\ $\|Jx\|_{X^{**}}=\|x\|_X$ for all $x\in X$.
	Therefore,
	\begin{equation*}
		\|x\|_X
		=\|Jx\|_{X^{**}}
		=\sup_{\substack{\varphi\in X^*\\ \|\varphi\|_{X^*}\le 1}} |(Jx)(\varphi)|
		=\sup_{\substack{\varphi\in X^*\\ \|\varphi\|_{X^*}\le 1}} |\varphi(x)|.
	\end{equation*}
	If $x_j\rightharpoonup x$ in $X$, then $\varphi(x_j)\to\varphi(x)$ for all $\varphi\in X^*$ and hence
	\begin{equation*}
		\|x\|_X
		= \sup_{\substack{\varphi\in X^*\\ \|\varphi\|_{X^*}\le 1}} |\varphi(x)|
		\le \liminf_{j\to\infty}\sup_{\substack{\varphi\in X^*\\ \|\varphi\|_{X^*}\le 1}}|\varphi(x_j)|
		= \liminf_{j\to\infty}\|x_j\|_X.
	\end{equation*}
\end{proof}

\begin{lemma}[$\cJ$ is coercive and w.l.s.c.]\label{lem:Jcoercivewlsc}
	The functional $\cJ$ defined in \eqref{eq:energyfunc} is
	\begin{itemize}
		\item[i)]{\em coercive} on $V$, i.e.
		      \begin{equation*}
			      \cJ(v)\to\infty\quad\text{as}\quad\|v\|_{1,\Omega}\to\infty,
		      \end{equation*}
		\item[ii)] {\em weakly lower semicontinuous} on $V$, i.e.\ if $v_j\rightharpoonup v$ in $V$, then
		      \begin{equation*}
			      \cJ(v)\le \liminf_{j\to\infty}\cJ(v_j).
		      \end{equation*}
	\end{itemize}
\end{lemma}
\begin{proof}
	By boundedness of $\ell$ we have $|\ell(v)|\le \|\ell\|_{-1,\Omega}\|v\|_{1,\Omega}$. Hence
	\begin{equation*}
		\cJ(v)=\frac{1}{2}\,a(v,v)-\ell(v)
		\ge \frac{\alpha}{2}\,\|v\|_{1,\Omega}^2-\|\ell\|_{-1,\Omega}\,\|v\|_{1,\Omega}
		\xrightarrow{ \ \|v\|_{1,\Omega}\to\infty \ }\infty.
	\end{equation*}
	For $s\in L^2(\Omega;\mathrm{Sym}^2(\R^d))$ set
	\begin{equation*}
		\|s\|_A^2 := \int_\Omega s:A:s\;\dx.
	\end{equation*}
	By uniform ellipticity of $A$, the map $\|\cdot\|_A$ defines a norm on $L^2(\Omega;\mathrm{Sym}^2(\R^d))$
	equivalent to $\|\cdot\|_{0,\Omega}$. Moreover, $a(v,v)=\|\varepsilon(v)\|_A^2$.
	The mapping $v\mapsto \varepsilon(v)$ is linear and bounded from $V$ to
	$L^2(\Omega;\mathrm{Sym}^2(\R^d))$. Hence, if $v_j\rightharpoonup v$ in $V$, then
	\begin{equation*}
		\varepsilon(v_j)\rightharpoonup \varepsilon(v)
		\quad\text{in } L^2(\Omega;\mathrm{Sym}^2(\R^d)).
	\end{equation*}
	Since norms are weakly lower semicontinuous by Lemma \ref{lem:normwlsc}, we obtain
	\begin{equation*}
		a(v,v)=\|\varepsilon(v)\|_A^2
		\le \liminf_{j\to\infty}\|\varepsilon(v_j)\|_A^2
		= \liminf_{j\to\infty} a(v_j,v_j).
	\end{equation*}
	Furthermore, $\ell\in V'$ is weakly continuous, hence $\ell(v_j)\to \ell(v)$.
	Therefore,
	\begin{equation*}
		\cJ(v)=\frac{1}{2}\,a(v,v)-\ell(v)
		\le \frac{1}{2}\,\liminf_{j\to\infty} a(v_j,v_j) - \lim_{j\to\infty}\ell(v_j)
		= \liminf_{j\to\infty}\cJ(v_j).
	\end{equation*}
\end{proof}

\begin{proposition}[Existence of Minimizer of $\cJ$]\label{pr:Jex}
	The functional $\cJ$ defined in \ref{eq:energyfunc} has at least one minimizer $u\in K$.
\end{proposition}
\begin{proof}
	Set
	\begin{equation*}
		M := \inf_{v\in K}\cJ(v)\in[-\infty,\infty).
	\end{equation*}
	Since $0\in K$ we have $M\le \cJ(0)=0$, hence $M$ is finite from above.
	Let $\{v_j\}_{j\in\N}\subset K$ be a minimizing sequence, i.e.\
	$\cJ(v_j)\to M$ as $j\to\infty$. By Lemma \ref{lem:Jcoercivewlsc} {\em i)}, $\cJ$ is coercive and thus, the sequence $\{v_j\}_{j\in\N}$ is bounded in $V$. Since $V$ is a Hilbert space, there exist a subsequence $\{v_{j_k}\}_{k\in\N}$ and some $u\in V$ such that
	\begin{equation*}
		v_{j_k} \rightharpoonup u \quad\text{in } V.
	\end{equation*}
	As $K\subset V$ is closed and convex, it is weakly closed in $V$. Hence $u\in K$. By Lemma \ref{lem:Jcoercivewlsc} {\em ii)}, $\cJ$ is weakly lower semicontinuity and thus we obtain
	\begin{equation*}
		\cJ(u)\le \liminf_{k\to\infty}\cJ(v_{j_k})=M=\inf_{v\in K}\cJ(v),
	\end{equation*}
	so $u$ is a minimizer of $\cJ$ on $K$.
\end{proof}

\begin{corollary}[Well-posedness of Problem \ref{pr:signorini1}]\label{cor:wellposed}
	The functional $\cJ$ defined in \ref{eq:energyfunc} admits a unique minimizer $u\in K$. Moreover, the minimizer satisfies the a-priori bound
	\begin{equation}\label{eq:ustab}
		\|u\|_{1,\Omega} \le \frac{2\,C_K^2\,\max\{1,C_\mathrm{tr}\}}{\alpha_A}\,\left(\|f\|_{0,\Omega} +\|F\|_{0,\Gamma_N}\right).
	\end{equation}
	Consequently, Problem \ref{pr:signorini1} has a unique stable solution.
\end{corollary}
\begin{proof}
	By Proposition~\ref{pr:Jex} we have existence of a solution $u\in K$ to Problem~\ref{pr:signorini1} and by Proposition~\ref{pr:Junique} it is unique. It remains to prove the a-priori bound. Since $u$ is the unique minimizer, it holds
	\begin{equation*}
		\cJ(u)\le \cJ(0)=0\quad\Longrightarrow\quad \frac{1}{2}\,a(u,u)\le \ell(u).
	\end{equation*}
	By Proposition~\ref{pr:aproperties} {\em ii)} we have
	\begin{equation*}
		a(u,u)\ge \frac{\alpha_A}{C_K^2}\,\|u\|_{1,\Omega}^2
	\end{equation*}
	and thus
	\begin{equation*}
		\frac{\alpha_A}{2\,C_K^2}\,\|u\|_{1,\Omega}^2\le \frac{1}{2}\,a(u,u)\le |\ell(u)|\le  \max\{1,C_{\mathrm{tr}}\}\left(\|f\|_{0,\Omega} +\|F\|_{0,\Gamma_N}\right)\|u\|_{1,\Omega},
	\end{equation*}
	where the last inequality is \eqref{eq:ellbound}. Dividing both sides by $\|u\|_{1,\Omega}$ yields the desired estimate \eqref{eq:ustab}. For $u=0$ the estimate is trivial.
\end{proof}

\subsubsection{Variational inequality as an optimality condition}
Finally, we turn to the {\em weak formulation} of Signorini's problem. Unlike the weak formulation of conventional simple boundary value problems, here we do not end up with a variational equation, but with a variational inequality. This is derived by a first-order optimality condition on the functional $\cJ$.

\begin{lemma}[Fréchet differentiability of $\cJ$]\label{lem:Jdiff}
	The functional $\cJ$ defined in \eqref{eq:energyfunc} is Fréchet differentiable on $V$. Its derivative at $u\in V$ is given by
	\begin{equation}\label{eq:Jprime}
		\cJ'(u)[h] = a(u,h)-\ell(h)\quad \forall h\in V.
	\end{equation}
\end{lemma}
\begin{proof}
	Let $u,h\in V$ and $t\in\R$. Using bilinearity of $a$ and linearity of $\ell$ we compute
	\begin{equation*}
		\cJ(u+th)-\cJ(u)
		=\frac12\,\left(a(u+th,u+th)-a(u,u)\right)-t\ell(h)
		= t\left(a(u,h)-\ell(h)\right)+\frac{t^2}{2}\,a(h,h).
	\end{equation*}
	Hence
	\begin{equation*}
		\frac{\cJ(u+th)-\cJ(u)-t(a(u,h)-\ell(h))}{\|th\|_{1,\Omega}}
		=\frac{|t|}{2}\,\frac{|a(h,h)|}{\|h\|_{1,\Omega}}\le \frac{|t|}{2}\,C_A\,\|h\|_{1,\Omega}\xrightarrow{\ t\to 0 \ }0,
	\end{equation*}
	by boundedness of $a$ as proven in Proposition \ref{pr:aproperties}.
\end{proof}

\begin{proposition}[First-order optimality condition on convex sets]\label{pr:FOC}
	A function $u\in K$ minimizes $\cJ$ over $K$ if and only if it satisfies the variational inequality
	\begin{equation}\label{eq:firstorderopt}
		\cJ^\prime(u)[v-u]\ge 0\quad\forall v\in K.
	\end{equation}
\end{proposition}
\begin{proof}
	Assume first that $u\in K$ minimizes $\cJ$ over $K$ and let $v\in K$.
	By convexity of $K$ we have $u+t(v-u)\in K$ for all $t\in[0,1]$.
	Define $\phi(t):=\cJ(u+t(v-u))$. Then $\phi$ attains its minimum at $t=0$, hence $\phi'(0^+)\ge 0$.
	Using the chain rule and Fréchet differentiability of $\cJ$ we obtain
	\begin{equation*}
		\phi'(0^+)=\cJ'(u)[v-u]\ge 0.
	\end{equation*}
	Conversely, assume \eqref{eq:firstorderopt}. Let $v\in K$ be arbitrary and set $h:=v-u$. By Taylor expansion of the quadratic functional $\cJ$ we have
	\begin{equation*}
		\cJ(u+h)-\cJ(u)=\cJ'(u)[h]+\frac12\,a(h,h).
	\end{equation*}
	By \eqref{eq:firstorderopt}, $\cJ'(u)[h]\ge 0$, and since $a(h,h)\ge 0$ we conclude $\cJ(v)\ge \cJ(u)$ for all $v\in K$. Hence $u$ is a minimizer over $K$.
\end{proof}

Combining Lemma \ref{lem:Jdiff} and Proposition \ref{pr:FOC}, $u\in K$ solves Problem \ref{pr:signorini1} if and only if
\begin{equation*}
	\cJ^\prime(u)[v-u] = a(u,v-u)-\ell(v-u)\ge 0 \quad \forall v\in K.
\end{equation*}
This is the first weak formulation of Signorini's problem as a variational inequality.

\begin{problem}[Weak Formulation of Signorini's Problem]\label{pr:signorini2}
Find $u\in K$ such that
\begin{equation*}
	a(u,v-u)\ge \ell(v-u)\quad \forall v\in K.
\end{equation*}
\end{problem}

Well-posedness of Problem \ref{pr:signorini2} is already ensured, since it is equivalent to the minimization Problem \ref{pr:signorini1} in the sense that each solution to the one of the problems is a solution to the other problem (Proposition \ref{pr:FOC}). The solution is unique due to Corollary \ref{cor:wellposed} with the stability estimate \eqref{eq:ustab}. Another useful equivalent formulation is the following.

\begin{problem}[Weak Formulation of Signorini's Problem]\label{pr:signorini3}
Find $u\in K$ such that
\begin{equation*}
	\begin{cases}
		a(u,u) = \ell(u), &                 \\[2pt]
		a(u,v)\ge \ell(v) & \forall v\in K.
	\end{cases}
\end{equation*}
\end{problem}

\begin{proposition}[Equivalence of Problem \ref{pr:signorini2} and Problem \ref{pr:signorini3}] Any solution $u\in K$ to Problem \ref{pr:signorini2} is also a solution to Problem \ref{pr:signorini3} and vise versa.
\end{proposition}
\begin{proof}
	Let $u\in K$ solve Problem \ref{pr:signorini3}. Then
	\begin{equation*}
		a(u,v-u) = a(u,v)-a(u,u)  = a(u,v)-\ell(u)\ge \ell(v)-\ell(u) = \ell(v-u).
	\end{equation*}
	Assuming that $u\in K$ solves Problem \ref{pr:signorini2} and testing $v=0$
	\begin{equation*}
		a(u,-u)\ge \ell(-u)\quad\Longleftrightarrow\quad -a(u,u)\ge -\ell(u)\quad\Longleftrightarrow\quad a(u,u)\le \ell(u).
	\end{equation*}
	However, testing with $v=2u$ yields $a(u,u)\ge \ell(u)$ and thus $a(u,u) = \ell(u)$. Furthermore, Problem \ref{pr:signorini2} reads
	\begin{equation*}
		a(u,v-u) = a(u,v)-a(u,u)\ge \ell(v-u) = \ell(v)-\ell(u)\quad\Longrightarrow\quad a(u,v)\ge a(u,u)+\ell(v)-\ell(u).
	\end{equation*}
	Plugging in $a(u,u) = \ell(u)$ yields $a(u,v)\ge \ell(v)$.
\end{proof}


We close this section by providing another stability result on the dependency of solutions to varying source terms as input data.

\begin{proposition}[Lipschitz dependency]\label{pr:lipschitz}
	Let $f_1,f_2\in L^2(\Omega;\R^d)$ and $F_1,F_2\in L^2(\Gamma_N;\R^d)$ and let $\ell_i\in V'$
	be the corresponding load functionals
	\begin{equation*}
		\ell_i(v)=\int_\Omega f_i\cdot v\;\dx+\int_{\Gamma_N}F_i\cdot \gamma v\;\ds,
		\quad v\in V,\ i=1,2.
	\end{equation*}
	Let $u_i\in K$ be the corresponding (unique) minimizers of
	\begin{equation*}
		\cJ_i(v):=\frac{1}{2}\,a(v,v)-\ell_i(v)
	\end{equation*}
	over $K$. Then the solutions depend Lipschitz continuously on the data, i.e.
	\begin{equation*}
		\|u_1-u_2\|_{1,\Omega}
		\le \frac{C_K^2}{\alpha_A}\,\|\ell_1-\ell_2\|_{-1,\Omega}
		\le \frac{C_K^2\,\max\{1,C_{\mathrm{tr}}\}}{\alpha_A}\,\left(\|f_1-f_2\|_{0,\Omega}+\|F_1-F_2\|_{0,\Gamma_N}\right).
	\end{equation*}
\end{proposition}
\begin{proof}
	Since $u_i\in K$ minimizes $\cJ_i$ over the closed convex set $K$ by Proposition \ref{pr:FOC}, it satisfies the first-order optimality condition
	\begin{equation}\label{eq:VI_i}
		a(u_i,v-u_i)\ge \ell_i(v-u_i)\quad \forall v\in K,\ i=1,2.
	\end{equation}
	Choose $v=u_2$ in \eqref{eq:VI_i} for $i=1$ and $v=u_1$ in \eqref{eq:VI_i} for $i=2$. This yields
	\begin{align*}
		a(u_1,u_2-u_1) & \ge \ell_1(u_2-u_1), \\
		a(u_2,u_1-u_2) & \ge \ell_2(u_1-u_2).
	\end{align*}
	Adding both inequalities and using bilinearity of $a$ gives
	\begin{equation*}
		a(u_1-u_2,u_1-u_2)\le (\ell_1-\ell_2)(u_1-u_2).
	\end{equation*}
	By the $V$-ellipticity estimate from Proposition~\ref{pr:aproperties}~ii) we have
	\begin{equation*}
		a(w,w)\ge \frac{\alpha_A}{C_K^2}\,\|w\|_{1,\Omega}^2\quad \forall w\in V.
	\end{equation*}
	Applying this with $w=u_1-u_2$ and estimating the right-hand side by the dual norm
	$\|\cdot\|_{-1,\Omega}$ yields
	\begin{equation*}
		\frac{\alpha_A}{C_K^2}\,\|u_1-u_2\|_{1,\Omega}^2
		\le a(u_1-u_2,u_1-u_2)
		\le \|\ell_1-\ell_2\|_{-1,\Omega}\,\|u_1-u_2\|_{1,\Omega}.
	\end{equation*}
	If $u_1=u_2$ there is nothing to show. Otherwise we divide by $\|u_1-u_2\|_{1,\Omega}$ and obtain
	\begin{equation}\label{eq:lipschitz_dual}
		\|u_1-u_2\|_{1,\Omega}\le \frac{C_K^2}{\alpha_A}\,\|\ell_1-\ell_2\|_{-1,\Omega}.
	\end{equation}
	It remains to bound $\|\ell_1-\ell_2\|_{-1,\Omega}$ in terms of $f_1-f_2$ and $F_1-F_2$.
	Let $v\in V$ be arbitrary. By Cauchy--Schwarz,
	\begin{align*}
		|(\ell_1-\ell_2)(v)|
		 & \le \|f_1-f_2\|_{0,\Omega}\,\|v\|_{0,\Omega}
		+ \|F_1-F_2\|_{0,\Gamma_N}\,\|\gamma v\|_{0,\Gamma_N}.
	\end{align*}
	Using $\|v\|_{0,\Omega}\le \|v\|_{1,\Omega}$ and the trace estimate
	$\|\gamma v\|_{0,\Gamma_N}\le C_{\mathrm{tr}}\,\|v\|_{1,\Omega}$ we obtain
	\begin{equation*}
		|(\ell_1-\ell_2)(v)|
		\le \max\{1,C_{\mathrm{tr}}\}\left(\|f_1-f_2\|_{0,\Omega}+\|F_1-F_2\|_{0,\Gamma_N}\right)\|v\|_{1,\Omega}.
	\end{equation*}
	Dividing by $\|v\|_{1,\Omega}$ and taking the supremum over $0\neq v\in V$ yields
	\begin{equation*}
		\|\ell_1-\ell_2\|_{-1,\Omega}
		\le \max\{1,C_{\mathrm{tr}}\}\left(\|f_1-f_2\|_{0,\Omega}+\|F_1-F_2\|_{0,\Gamma_N}\right)
	\end{equation*}
	Combining this with \eqref{eq:lipschitz_dual} gives the claim.
\end{proof}

\subsection{Strong Formulation}
From the previous weak formulations of Signorini's problem, we know there exists a uniquely determined $u\in K$ that solves the problem. To derive a {\em strong} PDE formulation, we have to discuss what was $\sigma(u)\mathrm{n}$ means and in what sense the contact conditions on $\Gamma_C$ can be described as equalities and inequalities.


\subsubsection{Additional regularity and meaning of boundary tractions}

In the weak setting, the solution only satisfies $u\in K\subset H^1(\Omega;\R^d)$., i.e.\ $\varepsilon(u)\in L^2(\Omega;\mathrm{Sym}^2(\R^d))$ and, by $\sigma(u)=A:\varepsilon(u)$, we obtain $\sigma(u)\in L^2(\Omega;\mathrm{Sym}^2(\R^d))$.
This regularity is sufficient to interpret the internal virtual work
\begin{equation*}
	a(u,v)=\int_\Omega \sigma(u):\varepsilon(v)\;\dx,
\end{equation*}
but it is not enough to write the equilibrium equation $-\diver\sigma(u)=f$ as an identity in
$L^2(\Omega;\R^d)$, since the divergence of an $L^2$--field is only defined in the sense of
distributions. To formulate a strong equilibrium equation in $L^2(\Omega;\R^d)$ and to give a meaning to {\em boundary
		tractions}, we assume additional regularity of the stress,
\begin{equation}\label{eq:Hdiv}
	\sigma(u)\in H(\diver;\Omega)
	:=\left\{\tau\in L^2(\Omega;\R^{d\times d})\;\middle|\;\diver\tau\in L^2(\Omega;\R^d)\right\}.
\end{equation}
Here $\diver\tau$ is taken row-wise, i.e.
\begin{equation*}
	(\diver\tau)_i=\sum_{j=1}^d \partial_j \tau_{ij}
\end{equation*}
in the distributional sense. If $\sigma(u)\in H(\diver;\Omega)$, then $\diver\sigma(u)\in L^2(\Omega;\R^d)$,
so the equilibrium equation is well--defined. The same regularity is also exactly what is needed to define the boundary traction (or surface force density) associated with $\tau$. Formally, for smooth $\tau$ one would write
\begin{equation*}
	\tau\,\mathrm{n}\quad\text{on }\Gamma,
\end{equation*}
where $\mathrm{n}:\Gamma\to\R^d$ is the outward unit normal and $\tau\,\mathrm{n}$ denotes the pointwise
matrix--vector product. However, for $\tau\in L^2(\Omega;\R^{d\times d})$ a pointwise boundary trace is not available, so $\tau\,\mathrm{n}$ cannot be defined as an $L^2(\Gamma)$--function in general. One therefore starts from the classical integration by parts formula. If $\tau$ and $v$ are smooth, then
\begin{equation}\label{eq:green_smooth}
	\int_\Omega \tau:\nabla v\;\dx + \int_\Omega (\diver\tau)\cdot v\;\dx
	= \int_\Gamma (\tau\,\mathrm{n})\cdot v\;\ds.
\end{equation}
The key point now is that the left-hand side still makes sense for $\tau\in H(\diver;\Omega)$ and
$v\in H^1(\Omega;\R^d)$. Hence, for fixed $\tau\in H(\diver;\Omega)$ the map
\begin{equation*}
	F_\tau(v) :=\int_\Omega \tau:\nabla v\;\dx + \int_\Omega (\diver\tau)\cdot v\;\dx,
	\quad v\in H^1(\Omega;\R^d),
\end{equation*}
defines a continuous linear functional on $H^1(\Omega;\R^d)$. The crucial observation is that $F_\tau$ depends only on the boundary trace $\gamma v$. Indeed, if $w\in H^1_0(\Omega;\R^d)$, then $\gamma w=0$, and we claim that $F_\tau(w)=0$. To see this, we take a sequence $(w_m)_{m\in\N}\subset C_c^\infty(\Omega;\R^d)$ with
$w_m\to w$ in $H^1(\Omega;\R^d)$. Since $\diver\tau\in L^2(\Omega;\R^d)$, we have for all $w_m\in C_c^\infty(\Omega;\R^d)$, that
\begin{equation*}
	\int_\Omega (\diver\tau)\cdot w_m\;\dx = -\int_\Omega \tau:\nabla w_m\;\dx,
\end{equation*}
and therefore $F_\tau(w_m)=0$ for all $m\in\N$. Passing to the limit $m\to\infty$ using the continuity of
$F_\tau$ on $H^1(\Omega;\R^d)$ yields $F_\tau(w)=0$. Consequently, if $v_1,v_2\in H^1(\Omega;\R^d)$ satisfy $\gamma v_1=\gamma v_2$, then $v_1-v_2\in H^1_0(\Omega;\R^d)$ and thus $F_\tau(v_1)=F_\tau(v_2)$. This is exactly the statement that $F_\tau$ depends only on $\gamma v$. Since $\gamma:H^1(\Omega;\R^d)\to H^{1/2}(\Gamma;\R^d)$ is continuous and $\ker(\gamma)=H^1_0(\Omega;\R^d)$, the previous argument implies that $F_\tau$ induces a well-defined continuous linear functional on the trace space $H^{1/2}(\Gamma;\R^d)$. Hence there exists a unique element
$\gamma_{\mathrm{n}}(\tau)\in H^{-1/2}(\Gamma;\R^d)$ such that
\begin{equation}\label{eq:greenHdiv}
	\langle \gamma_{\mathrm{n}}(\tau),\,\gamma v\rangle_{\Gamma}
	= \int_\Omega \tau:\nabla v\;\dx + \int_\Omega (\diver\tau)\cdot v\;\dx
	\quad
	\forall v\in H^1(\Omega;\R^d),
\end{equation}
where $\langle\cdot,\cdot\rangle_{\Gamma}$ denotes the duality pairing of $H^{-1/2}(\Gamma;\R^d)\times H^{1/2}(\Gamma;\R^d)$. The operator $\gamma_{\mathrm{n}}:H(\diver;\Omega)\to H^{-1/2}(\Gamma;\R^d)$ is called the {\em normal trace operator} (see \cite[Lemma 3.1]{book}). In the smooth case, \eqref{eq:greenHdiv} reduces to
\eqref{eq:green_smooth}, so $\gamma_{\mathrm{n}}(\tau)$ coincides with the classical boundary function
$\tau\,\mathrm{n}$. For this reason we write suggestively
\begin{equation*}
	\gamma_{\mathrm{n}}(\tau)=\tau\,\mathrm{n}\in H^{-1/2}(\Gamma;\R^d).
\end{equation*}
Physically, $\tau\,\mathrm{n}$ is the traction vector (or surface force density) exerted across the boundary
with outward normal $\mathrm{n}$, and in particular $t(u):=\sigma(u)\mathrm{n}$ is the surface force induced
by the stress field $\sigma(u)$. Finally, since $\tau$ is symmetric, $\tau:\nabla v=\tau:\varepsilon(v)$ for all $v$, so \eqref{eq:greenHdiv} can be written in the symmetric form
\begin{equation}\label{eq:greenHdiv_sym}
	\langle \tau\,\mathrm{n},\,\gamma v\rangle_{\Gamma}
	= \int_\Omega \tau:\varepsilon(v)\;\dx + \int_\Omega (\diver\tau)\cdot v\;\dx
	\quad
	\forall (v,\tau)\in H^1(\Omega;\R^d)\times H(\diver;\Omega).
\end{equation}
Applying this to $\tau=\sigma(u)$ and assuming $\sigma(u)\in H(\diver;\Omega)$ provides
\begin{equation*}
	\langle\sigma(u)\mathrm{n},\gamma v\rangle_\Gamma = \int_\Omega \sigma(u):\varepsilon(v)\;\dx + \int_\Omega (\diver\sigma(u))\cdot v\;\dx\quad \forall v\in H^1(\Omega;\R^d).
\end{equation*}
which will be needed in the strong Signorini conditions.
