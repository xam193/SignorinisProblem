\section{Signorini's Problem}


\subsection{Physical and Geometric Setup of the Problem}

We consider an elastic body occupying a reference configuration $\Omega\subset\R^d,\,d\in\{2,3\}$.
Under the action of external forces and possible contact with a rigid foundation, the body deforms.
In the {\em small--strain} framework, the deformation is described by the {\em displacement field}
$u:\Omega\to\R^d$, and a material point $x\in\Omega$ is mapped to its deformed position
\begin{equation*}
	\varphi(x) := x + u(x),\qquad x\in\overline{\Omega}.
\end{equation*}
The aim of Signorini's problem is to determine an {\em equilibrium} displacement $u$ under
standard boundary conditions and {\em unilateral} contact constraints on the part of the boundary
that may touch the rigid foundation. Throughout the whole report, $\Omega\subset \R^d,\, d\in\{2,3\}$ denotes an open, bounded and connected {\em Lipschitz} domain. The later means, that the boundary $\Gamma:=\partial\Omega$ can locally be described as the graph of a Lipschitz continuous function. This ensures for example, that the outward normal vector $\mathrm{n}:\Gamma:\R^d$ is well--defined almost everywhere. The boundary is decomposed into three pairwise disjoint parts
\begin{equation*}
	\Gamma = \overline{\Gamma_D}\,\mathbin{\dot\cup}\,\overline{\Gamma_N}\,\mathbin{\dot\cup}\,\overline{\Gamma_C}
\end{equation*}
which are assumed to be open in $\Gamma$ and where each part is subject to different boundary conditions to model different physical interactions (See Figure \ref{fig:setup}). On $\Gamma_D$, we apply {\em homogeneous Dirichlet} boundary conditions. This forces the body to be {\em clamped} and this part, i.e.\ the displacement should not act on $\Gamma_D$,
\begin{equation*}
	\forall x\in\Gamma_D:\;x + u(x) = x \qquad\Longrightarrow \qquad(\gamma u)|_{\Gamma_D} = 0.
\end{equation*}
The part $\Gamma_N$ may be subject to a {\em Neumann} boundary condition, which means we impose a known {\em surface force density}. $\Gamma_C$ denotes the {\em potential contact} boundary. We assume that $\Gamma_C$ and $\Gamma_D$ have positive $(d-1)$--Lebesgue measure. The rigid foundation is  positioned so that any actual contact, both initially and in the deformed equilibrium configuration, may only occur on $\Gamma_C$. While $\Gamma_C$ is prescribed a priori, the {\em active contact zone} $\Gamma_{\mathrm{act}}(u)\subset\Gamma_C$ is unknown and depends on the solution $u$. Later on, the boundary conditions imposed on $\Gamma_C$ enforce {\em unilateral contact} through {\em nonpenetration} and {\em complementarity}.

\subsection{Weak Formulation}

\subsubsection{Admissible Displacements}
We start by introducing the energy space and the admissible set of displacements.
Homogeneous Dirichlet boundary conditions on $\Gamma_D$ are incorporated by
\begin{equation}\label{eq:energyspace}
	V:=\left\{ v\in H^1(\Omega;\R^d)\;\middle|\; \gamma v=0 \text{ on }\Gamma_D\right\}\subset H^1(\Omega;\R^d).
\end{equation}
For $v\in V$ we define the normal component of the boundary displacement on $\Gamma_C$ by
\begin{equation}\label{eq:usualnormal}
	v_{\mathrm n} := (\gamma v)\cdot \mathrm n \qquad \text{on }\Gamma_C,
\end{equation}
where $\mathrm n$ is the outward unit normal on $\Gamma$. The unilateral nature of contact is expressed by the {\em nonpenetration} constraint. The rigid foundation is located on the exterior side of $\Gamma_C$ in direction of $\mathrm n$. Hence the body is not allowed to move across $\Gamma_C$ into the obstacle. In the small--strain setting this yields that the solution must satisfy
\begin{equation*}
	u_{\mathrm n}\le 0 \qquad \text{a.e.\ on }\Gamma_C.
\end{equation*}
This motivates the nonempty, closed and convex cone of admissible displacements
\begin{equation}\label{eq:admissible}
	K := \left\{ v\in V\;\middle|\; v_{\mathrm n}\le 0 \text{ a.e.\ on }\Gamma_C\right\}\subset V.
\end{equation}

\begin{proposition}[$K$ is a closed convex cone]\label{pr:K}
	The set $K$ defined in \ref{eq:admissible} is a nonempty, closed and convex cone in $V$.
\end{proposition}
\begin{proof}
	$K$ is nonempty, since $0\in K$. Let $\lambda\in[0,\infty)$ and $v\in K\subset V$. Since $V$ is a vector space, we have $\lambda v\in V$. Furthermore,
	\begin{equation*}
		(\lambda v)_\mathrm{n} = \gamma(\lambda v)\cdot \mathrm{n} = (\lambda\gamma v)\cdot \mathrm{n} = \lambda(\gamma v\cdot n) = \lambda v_\mathrm{n}.
	\end{equation*}
	Since $v_\mathrm{n}\le 0$ a.e.\ on $\Gamma_C$ and $\lambda\ge 0$, we also have $\lambda v_\mathrm{n}\le 0$ a.e.\ on $\Gamma_C$, i.e.\ $\lambda v\in K$. Thus $K$ is a cone in $V$. Let $v,w\in K$ and $\theta\in[0,1]$ and set $u:= \theta v+(1-\theta)w\in V$. Then we have
	\begin{equation*}
		u_\mathrm{n} =(\gamma u)\cdot \mathrm{n} = (\theta\gamma v+(1-\theta)\gamma w)\cdot \mathrm{n} = \theta(\gamma v\cdot \mathrm{n})+(1-\theta)(\gamma w\cdot\mathrm{n}) = \theta v_\mathrm{n}+(1-\theta)w_\mathrm{n}
	\end{equation*}
	and $v_\mathrm{n},w_\mathrm{n}\le 0$ a.e.\ on $\Gamma_C$ and $\theta,(1-\theta)\ge 0$. Thus,
	\begin{equation*}
		\theta v_\mathrm{n}+(1-\theta)w_\mathrm{n}\le 0\qquad\text{a.e.\ on }\Gamma_C,
	\end{equation*}
	i.e.\ $u\in K$, which makes $K$ convex. To prove that $K$ is closed, let $(v_k)_{k\in\mathbb{N}}\subset K$ with
	$v_k\to v$ in $V$. Since the trace operator $\gamma:V\to L^2(\Gamma_C;\R^d)$
	is continuous, the mapping
	\begin{equation*}
		T:V\to L^2(\Gamma_C),\qquad T(w):=(\gamma w)\cdot \mathrm{n}=w_\mathrm{n},
	\end{equation*}
	is continuous as well. Hence,
	\begin{equation*}
		(v_k)_\mathrm{n}=T(v_k)\to T(v)=v_\mathrm{n}
		\qquad\text{in }L^2(\Gamma_C).
	\end{equation*}
	Since $(v_k)_\mathrm{n}\le 0$ a.e.\ on $\Gamma_C$ for all $k$, there exists a
	subsequence $((v_{k_j})_\mathrm{n})_{j\in\mathbb{N}}$ such that
	\begin{equation*}
		(v_{k_j})_\mathrm{n}(x)\to v_\mathrm{n}(x)
		\qquad\text{for a.e. }x\in\Gamma_C.
	\end{equation*}
	Passing to the limit in the inequality $(v_{k_j})_\mathrm{n}(x)\le 0$, we obtain
	\begin{equation*}
		v_\mathrm{n}(x)\le 0
		\qquad\text{for a.e. }x\in\Gamma_C.
	\end{equation*}
	Thus $v\in K$, so $K$ is closed.
\end{proof}

\subsubsection{Small Strain Elasticity}

We start by defining the {\em small strain tensor} for a displacement function $u\in H^1(\Omega;\R^d)$, which is similar to the symmetric part of square matrices but for the gradient $\nabla u$ of $u$ and through which the deformation of the body $\overline{\Omega}$ by $u$ is quantified.
\begin{definition}[Small Strain Tensor]
	The {\tt (small) strain tensor} $\varepsilon$ is defined by
	\begin{equation*}
		\varepsilon:H^1(\Omega;\R^d)\to L^2(\Omega;\mathrm{Sym}^2(\R^d)),\qquad u\mapsto \varepsilon(u):= \frac{1}{2}\left(\nabla u + \nabla u^\top\right).
	\end{equation*}
\end{definition}
In linear elasticity, stresses are assumed to depend linearly on strains. The material response is encoded in the fourth order {\em elasticity tensor} $\cA$, which maps a symmetric strain tensor to the corresponding {\em Cauchy stress tensor}. The tensor field $\cA$ is prescribed by the material and may vary in space, describing a {\em heterogeneous} body.
\begin{definition}[Elasticity tensor field]\label{def:elastfield}
	An {\tt elasticity tensor field} is a fourth order tensor
	\begin{equation*}
		\cA:\Omega\to \cL(\mathrm{Sym}^2(\R^d);\mathrm{Sym}^2(\R^d))
	\end{equation*}
	such that the coefficients satisfy the symmetries
	\begin{equation*}
		\cA_{ijkl}(x)=\cA_{jikl}(x)=\cA_{ijlk}(x)=\cA_{klij}(x),
		\qquad \forall\,1\le i,j,k,l\le d,
	\end{equation*}
	and the uniform ellipticity condition, i.e.\ there exists $\alpha_\cA>0$ such that for a.e.\ $x\in\Omega$
	\begin{equation*}
		s:\cA(x):s \ge \alpha_\cA\, (s:s)\qquad \forall\,s\in \mathrm{Sym}^2(\R^d).
	\end{equation*}
	Moreover, there exists $C_\cA>0$ such that for a.e.\ $x\in\Omega$
	\begin{equation*}
		|\cA_{ijkl}(x)|\le C_\cA \qquad \forall\,1\le i,j,k,l\le d.
	\end{equation*}
\end{definition}

The {\em internal} and {\em surface forces} in the body $\overline{\Omega}$ are described by the {\em Cauchy stress tensor}.

\begin{definition}[Cauchy stress tensor]
	Let $\cA$ be an elasticity tensor field. We define the
		{\tt (linear) Cauchy stress tensor} by
	\begin{equation*}
		\sigma: H^1(\Omega;\R^d)\to L^2(\Omega;\mathrm{Sym}^2(\R^d)),\qquad u\mapsto \cA : \varepsilon(u).
	\end{equation*}
\end{definition}
On $V$, we define the symmetric bilinear form
\begin{equation}\label{eq:symbil}
	a(u,v):= \int_\Omega \sigma(u):\varepsilon(v)\;\dx,\qquad u,v\in V
\end{equation}
representing the {\em virtual work of internal forces} and the linear bounded functional
\begin{equation}\label{eq:laodfunc}
	\ell(v):= \int_\Omega f\cdot v\;\dx  + \int_{\Gamma_N} F\cdot \gamma v\;\ds,\qquad  v\in V,
\end{equation}
with {\em volume force density} $f\in L^2(\Omega;\R^d)$ on the body and {\em surface loads} $F\in L^2(\Gamma_N;\R^d)$ on the Neumann boundary, representing the {\em virtual work of external forces}. In the following, we want to prove standard properties of $a$ and $\ell$. For this, we will need the following inequalities. The first inequality is a result of {\em Korn's second inequality} (\cite[Theorem 3.3]{book} and \cite[Section 3.5.2]{book}) and the {\em Peetre--Tartar} Lemma~(\cite[Lemma~3.5]{book}). Therefore only the proof of the second one is given.

\begin{lemma}\label{lem:korn}
	Assuming $|\Gamma_D|>0$, there exists a constant $C_K>0$ such that
	\begin{equation*}
		C_K^{-1}\,\|v\|_{1,\Omega} \le \|\varepsilon(v)\|_{0,\Omega}\le \|v\|_{1,\Omega} \qquad \forall\, v\in V.
	\end{equation*}
\end{lemma}
\begin{proof}
	We prove the second inequality. For all $v\in V$ we have
	\begin{equation*}
		|\varepsilon(v)(x)| = \left|\frac{1}{2}(\nabla v(x) +\nabla v^\top)\right|\le \frac{1}{2}(|\nabla v(x)|+|\nabla v(x)^\top|) = |\nabla v(x)|
	\end{equation*}
	and thus
	\begin{equation*}
		\|\varepsilon(v)\|_{0,\Omega}^2 =\int_\Omega |\varepsilon(v)(x)|^2\;\dx \le\int_\Omega |\nabla v(x)|^2\;\dx = \|\nabla v\|_{0,\Omega}^2\le \|v\|_{1,\Omega}^2,
	\end{equation*}
	where the last inequality holds due to
	\begin{equation*}
		\|v\|_{1,\Omega}^2 = \|v\|_{0,\Omega}^2 + \|\nabla v\|_{0,\Omega}^2.
	\end{equation*}
\end{proof}

\begin{proposition}\label{pr:aproperties}
	The symmetric bilinear form $a:V\times V\to \R$ defined in \eqref{eq:symbil} is
	\begin{itemize}
		\item[i)] {\em bounded}, i.e.\ for all $v,w\in V$ there exists $C_1<\infty$ such that
		      \begin{equation*}
			      |a(u,v)|\le C_1\,\|u\|_{1,\Omega}\,\|v\|_{1,\Omega}\qquad \text{with}\qquad C_1 = C_\cA,
		      \end{equation*}
		\item[ii)] $V${\em -elliptic}, i.e.\ for all $v\in V\setminus\{0\}$ there exist $C_2>0$ such that
		      \begin{equation*}
			      a(v,v)\ge C_2\,\|v\|_{1,\Omega}^2\qquad\text{with}\qquad C_2 = \frac{\alpha_\cA}{C_K^2}.
		      \end{equation*}
	\end{itemize}
	Moreover, the linear functional $\ell$ defined in \eqref{eq:laodfunc} is {\em bounded}, i.e.\ for all $v\in V$ there exists $C_3>0$ such that
	\begin{equation}\label{eq:ellbound}
		|\ell (v)|\le C_3\,\|v\|_{1,\Omega}\qquad\text{with}\qquad C_3 = \max\{1,C_\mathrm{tr}\}\left(\|f\|_{0,\Omega}+\|F\|_{0,\Gamma_N}\right)
	\end{equation}
\end{proposition}
\begin{proof}
	Let $v,w\in V$ be arbitrary. Then
	\begin{equation*}
		|a(v,w)|\le \|\cA:\varepsilon(v)\|_{0,\Omega}\,\|\varepsilon(w)\|_{0,\Omega}
	\end{equation*}
	by the Cauchy--Schwarz inequality. With Definition \ref{def:elastfield} there exists $C_\cA>0$ such that
	\begin{equation*}
		\|\cA:\varepsilon(v)\|_{0,\Omega}\le C_\cA\,\|\varepsilon(v)\|_{0,\Omega}.
	\end{equation*}
	Combining this with the second inequality form Lemma \ref{lem:korn} we have
	\begin{equation*}
		|a(v,w)|\le C_\cA\,\|\varepsilon(v)\|_{0,\Omega}\,\|\varepsilon(w)\|_{0,\Omega}\le C_\cA\,\|v\|_{1,\Omega}\,\|w\|_{1,\Omega}.
	\end{equation*}
	The ellipticity of $a$ follows by
	\begin{equation*}
		a(v,v) = \int_\Omega \varepsilon(v):\cA:\varepsilon(v)\;\dx \ge \alpha_\cA\int_\Omega\varepsilon(v):\varepsilon(v) = \alpha_\cA\,\|\varepsilon(v)\|^2_{0,\Omega},
	\end{equation*}
	where $\alpha_\cA>0$ exists due to Definition \ref{def:elastfield}. The first inequality of Lemma \ref{lem:korn} finally yields
	\begin{equation*}
		a(v,v)\ge \alpha_\cA\,\|\varepsilon(v)\|_{0,\Omega}^2\ge \alpha_\cA\,\left(C_K^{-1}\,\|v\|_{1,\Omega}\right)^2 = \frac{\alpha_\cA}{C_K^2}\,\|v\|_{1,\Omega}^2.
	\end{equation*}
	Again by Cauchy--Schwarz inequality,
	\begin{equation*}
		|\ell(v)|\le \|f\|_{0,\Omega}\,\|v\|_{0,\Omega}+\|F\|_{0,\Gamma_N}\,\|\gamma v\|_{0,\Gamma_N}\qquad \forall\, v\in V.
	\end{equation*}
	By the Trace Theorem \ref{thm:trace} there exists $C_\mathrm{tr}>0$ such that
	\begin{equation*}
		\|\gamma v\|_{0,\Gamma_N}\le \|\gamma v\|_{0,\Gamma}\le C_\mathrm{tr}\,\|v\|_{1,\Omega} \qquad \forall\, v\in V.
	\end{equation*}
	Finally,
	\begin{equation*}
		|\ell(v)|\le \left(\|f\|_{0,\Omega} + C_{\mathrm{tr}}\,\|F\|_{0,\Gamma_N}\right)\|v\|_{1,\Omega}\le \max\{1,C_\mathrm{tr}\}\,\left(\|f\|_{0,\Omega}+\|F\|_{0,\Gamma_N}\right)\|v\|_{1,\Omega}\qquad \forall\, v\in V.
	\end{equation*}
\end{proof}

\subsubsection{Energy Minimization and Well-posedness}

The total potential energy of the elastic body is given by the quadratic functional
\begin{equation}\label{eq:energyfunc}
	\cJ(v):=\frac{1}{2}\,a(v,v)-\ell(v),\qquad v\in V.
\end{equation}
In the presence of unilateral contact, the admissible displacements are restricted to the closed convex set $K\subset V$. We therefore consider the following constrained minimization problem.
\begin{problem}[Signorini's Problem -- Energy Minimization]\label{pr:signorini1}
Find $u\in K$ such that
\begin{equation*}
	\cJ(u)=\min_{v\in K}\cJ(v) = \min_{v\in K}\,\frac{1}{2}\,a(v,v)-\ell(v).
\end{equation*}
\end{problem}

We now what to prove well-posedness of the above Problem \ref{pr:signorini1}. This consist of existence and uniqueness of a solution and continuous dependency of the solution on the load functional $\ell$. We start by the uniqueness, which follows directly by the standard arguments of strict convexity of $\cJ$ and the convexity of $K$. The latter was already proven in Proposition \ref{pr:K}.

\begin{lemma}[Strict Convexity of $\cJ$]\label{lem:Jconvex}
	The Functional $\cJ$ defined in \eqref{eq:energyfunc} is strictly convex.
\end{lemma}
\begin{proof}
	For $u,v\in V$ and $t\in[0,1]$, we set $w_t:=(1-t)u+tv$. Then
	\begin{equation*}
		a(w_t,w_t) = (1-t)a(u,u)+ta(v,v)-t(1-t)a(u-v,u-v)
	\end{equation*}
	and
	\begin{equation*}
		\ell(w_t) = (1-t)\ell(u)+t\ell(v).
	\end{equation*}
	Together, we have
	\begin{equation*}
		\cJ(w_t) = (1-t)\cJ(u)+t\cJ(v)-\frac{1}{2}\,t(1-t)a(u-v,u-v).
	\end{equation*}
	If $u\neq v$ then $u-v\neq 0$ and thus by $V$-ellipticity of $a$ we have $a(u-v,u-v)>0$ and $t(1-t)>0$. This implies
	\begin{equation*}
		\cJ(w_t)<(1-t)\cJ(u)+t\cJ(v).
	\end{equation*}
\end{proof}

\begin{proposition}[Uniqueness of Minimizer of $\cJ$]\label{pr:Junique}
	If there exists a minimizer of the functional $\cJ$ defined in \eqref{eq:energyfunc}, then it is unique.
\end{proposition}
\begin{proof}
	Assume that there exist two minimizers $u_1,u_2\in K$ of $\cJ$ on $K$, i.e.
	\begin{equation}\label{eq:u1u2min}
		\cJ(u_1)=\min_{v\in K}\cJ(v)=\cJ(u_2).
	\end{equation}
	If $u_1=u_2$ there is nothing to show. Hence assume $u_1\neq u_2$.
	Since $K$ is convex, for every $t\in(0,1)$ the convex combination
	\begin{equation*}
		w_t := (1-t)u_1+tu_2
	\end{equation*}
	belongs to $K$. By Lemma \ref{lem:Jconvex}, $\cJ$ is strictly convex and therefore we have
	\begin{equation*}
		\cJ(w_t) < (1-t)\cJ(u_1)+t\cJ(u_2)\qquad \forall\, t\in(0,1).
	\end{equation*}
	Using \eqref{eq:u1u2min}, this yields
	\begin{equation*}
		\cJ(w_t) < (1-t)\min_{v\in K}\cJ(v)+t\min_{v\in K}\cJ(v)=\min_{v\in K}\cJ(v).
	\end{equation*}
	This contradicts the definition of the minimum, since $w_t\in K$.
	Therefore the assumption $u_1\neq u_2$ is false and we conclude $u_1=u_2$.
\end{proof}
It remains to show the existence of at least one minimizer. For this, we verify {\em coercivity} and {\em weakly lower semicontinuity} of the functional $\cJ$. The proof requires the following lemma. Weak convergence is denoted by $\rightharpoonup$.

\begin{lemma}[Weak lower semicontinuity of the norm]\label{lem:normwlsc}
	Let $(X,\|\cdot\|_X)$ be a normed space. Then $\|\cdot\|_X:X\to \R_{\ge 0}$ is a weakly lower semicontinuous functional, i.e.\ if $x_j\rightharpoonup x$ in $X$, then
	\begin{equation*}
		\|x\|_X \le \liminf_{j\to\infty}\|x_j\|_X.
	\end{equation*}
\end{lemma}
\begin{proof}
	Let $J:X\to X^{**}$ be the canonical embedding defined by
	\begin{equation*}
		(Jx)(\varphi):=\varphi(x)\qquad \forall\, \varphi\in X^\prime.
	\end{equation*}
	By the Hahn--Banach theorem, $J$ is an isometry, i.e.\ $\|Jx\|_{X^{**}}=\|x\|_X$ for all $x\in X$.
	Therefore,
	\begin{equation*}
		\|x\|_X
		=\|Jx\|_{X^{**}}
		=\sup_{\substack{\varphi\in X^\prime\\ \|\varphi\|_{X^\prime}\le 1}} |(Jx)(\varphi)|
		=\sup_{\substack{\varphi\in X^\prime\\ \|\varphi\|_{X^\prime}\le 1}} |\varphi(x)|.
	\end{equation*}
	If $x_j\rightharpoonup x$ in $X$, then $\varphi(x_j)\to\varphi(x)$ for all $\varphi\in X^\prime$ and hence
	\begin{equation*}
		\|x\|_X
		= \sup_{\substack{\varphi\in X^\prime\\ \|\varphi\|_{X^\prime}\le 1}} |\varphi(x)|
		\le \liminf_{j\to\infty}\sup_{\substack{\varphi\in X^\prime\\ \|\varphi\|_{X^\prime}\le 1}}|\varphi(x_j)|
		= \liminf_{j\to\infty}\|x_j\|_X.
	\end{equation*}
\end{proof}

\begin{lemma}[$\cJ$ is coercive and w.l.s.c.]\label{lem:Jcoercivewlsc}
	The functional $\cJ$ defined in \eqref{eq:energyfunc} is
	\begin{itemize}
		\item[i)]{\em coercive} on $V$, i.e.
		      \begin{equation*}
			      \cJ(v)\to\infty\qquad\text{as}\qquad\|v\|_{1,\Omega}\to\infty,
		      \end{equation*}
		\item[ii)] {\em weakly lower semicontinuous} on $V$, i.e.\ if $v_j\rightharpoonup v$ in $V$, then
		      \begin{equation*}
			      \cJ(v)\le \liminf_{j\to\infty}\cJ(v_j).
		      \end{equation*}
	\end{itemize}
\end{lemma}
\begin{proof}
	By boundedness of $\ell$ we have $|\ell(v)|\le \|\ell\|_{V^\prime}\|v\|_{1,\Omega}$. Hence
	\begin{equation*}
		\cJ(v)=\frac{1}{2}\,a(v,v)-\ell(v)
		\ge \frac{\alpha}{2}\,\|v\|_{1,\Omega}^2-\|\ell\|_{V^\prime}\,\|v\|_{1,\Omega}
		\xrightarrow{ \ \|v\|_{1,\Omega}\to\infty \ }\infty.
	\end{equation*}
	For $s\in L^2(\Omega;\mathrm{Sym}^2(\R^d))$ set
	\begin{equation*}
		\|s\|_\cA^2 := \int_\Omega s:\cA:s\;\dx.
	\end{equation*}
	By uniform ellipticity of $\cA$, the map $\|\cdot\|_\cA$ defines a norm on $L^2(\Omega;\mathrm{Sym}^2(\R^d))$
	equivalent to $\|\cdot\|_{0,\Omega}$. Moreover, $a(v,v)=\|\varepsilon(v)\|_\cA^2$.
	The mapping $v\mapsto \varepsilon(v)$ is linear and bounded from $V$ to
	$L^2(\Omega;\mathrm{Sym}^2(\R^d))$. Hence, if $v_j\rightharpoonup v$ in $V$, then
	\begin{equation*}
		\varepsilon(v_j)\rightharpoonup \varepsilon(v)
		\qquad\text{in } L^2(\Omega;\mathrm{Sym}^2(\R^d)).
	\end{equation*}
	Since norms are weakly lower semicontinuous by Lemma \ref{lem:normwlsc}, we obtain
	\begin{equation*}
		a(v,v)=\|\varepsilon(v)\|_\cA^2
		\le \liminf_{j\to\infty}\|\varepsilon(v_j)\|_\cA^2
		= \liminf_{j\to\infty} a(v_j,v_j).
	\end{equation*}
	Furthermore, $\ell\in V'$ is weakly continuous, hence $\ell(v_j)\to \ell(v)$.
	Therefore,
	\begin{equation*}
		\cJ(v)=\frac{1}{2}\,a(v,v)-\ell(v)
		\le \frac{1}{2}\,\liminf_{j\to\infty} a(v_j,v_j) - \lim_{j\to\infty}\ell(v_j)
		= \liminf_{j\to\infty}\cJ(v_j).
	\end{equation*}
\end{proof}

\begin{proposition}[Existence of Minimizer of $\cJ$]\label{pr:Jex}
	The functional $\cJ$ defined in \ref{eq:energyfunc} has at least one minimizer $u\in K$.
\end{proposition}
\begin{proof}
	Set
	\begin{equation*}
		M := \inf_{v\in K}\cJ(v)\in[-\infty,\infty).
	\end{equation*}
	Since $0\in K$ we have $M\le \cJ(0)=0$, hence $M$ is finite from above.
	Let $\{v_j\}_{j\in\N}\subset K$ be a minimizing sequence, i.e.\
	$\cJ(v_j)\to M$ as $j\to\infty$. By Lemma \ref{lem:Jcoercivewlsc} {\em i)}, $\cJ$ is coercive and thus, the sequence $\{v_j\}_{j\in\N}$ is bounded in $V$. Since $V$ is a Hilbert space, there exist a subsequence $\{v_{j_k}\}_{k\in\N}$ and some $u\in V$ such that
	\begin{equation*}
		v_{j_k} \rightharpoonup u \qquad\text{in } V.
	\end{equation*}
	As $K\subset V$ is closed and convex, it is weakly closed in $V$. Hence $u\in K$. By Lemma \ref{lem:Jcoercivewlsc} {\em ii)}, $\cJ$ is weakly lower semicontinuity and thus we obtain
	\begin{equation*}
		\cJ(u)\le \liminf_{k\to\infty}\cJ(v_{j_k})=M=\inf_{v\in K}\cJ(v),
	\end{equation*}
	so $u$ is a minimizer of $\cJ$ on $K$.
\end{proof}
We collect the previous results in the following corollary and further give a standard stability estimation for the norm of the unique solution.
\begin{corollary}[Well-posedness of Problem \ref{pr:signorini1}]\label{cor:wellposed}
	The functional $\cJ$ defined in \ref{eq:energyfunc} admits a unique minimizer $u\in K$. Moreover, the minimizer satisfies the a-priori bound
	\begin{equation}\label{eq:ustab}
		\|u\|_{1,\Omega} \le \frac{2\,C_K^2\,\max\{1,C_\mathrm{tr}\}}{\alpha_\cA}\,\left(\|f\|_{0,\Omega} +\|F\|_{0,\Gamma_N}\right).
	\end{equation}
	Consequently, Problem \ref{pr:signorini1} has a unique stable solution.
\end{corollary}
\begin{proof}
	By Proposition~\ref{pr:Jex} we have existence of a solution $u\in K$ to Problem~\ref{pr:signorini1} and by Proposition~\ref{pr:Junique} it is unique. It remains to prove the a-priori bound. Since $u$ is the unique minimizer, it holds
	\begin{equation*}
		\cJ(u)\le \cJ(0)=0\qquad\Longrightarrow\qquad \frac{1}{2}\,a(u,u)\le \ell(u).
	\end{equation*}
	By Proposition~\ref{pr:aproperties} {\em ii)} we have
	\begin{equation*}
		a(u,u)\ge \frac{\alpha_\cA}{C_K^2}\,\|u\|_{1,\Omega}^2
	\end{equation*}
	and thus
	\begin{equation*}
		\frac{\alpha_\cA}{2\,C_K^2}\,\|u\|_{1,\Omega}^2\le \frac{1}{2}\,a(u,u)\le |\ell(u)|\le  \max\{1,C_{\mathrm{tr}}\}\left(\|f\|_{0,\Omega} +\|F\|_{0,\Gamma_N}\right)\|u\|_{1,\Omega},
	\end{equation*}
	where the last inequality is \eqref{eq:ellbound}. Dividing both sides by $\|u\|_{1,\Omega}$ yields the desired estimate \eqref{eq:ustab}. For $u=0$ the estimate is trivial.
\end{proof}

\subsubsection{Variational inequality as an optimality condition}
Finally, we turn to the {\em weak formulation} of Signorini's problem. Unlike the weak formulation of conventional simple boundary value problems, here we do not end up with a variational equation, but with a variational inequality. This is derived by a first-order optimality condition on the functional $\cJ$, i.e. a condition on its {\em Fréchet derivative}.

\begin{definition}[Fréchet derivative]\label{def:frechetdiff}
	Let $X$ and $Y$ be normed vector spaces and let
	\begin{equation*}
		f:U\subseteq X\to Y
	\end{equation*}
	be a function defined on an open subset $U$ of $X$. Then $f$ is called {\tt Fréchet differentiable} at $x \in U$ if there exists a bounded linear operator $A(x):X\to Y$ such that
	\begin{equation}\label{eq:frechetquotient}
		\lim_{\|h\|_X\to 0} \frac{\|f(x+h)-f(x)-A(x)[h]\|_Y}{\|h\|_X}=0.
	\end{equation}
	If $A(x)$ exists, it is unique and is called the {\tt Fréchet derivative} of $f$ at $x$. We write
	\begin{equation*}
		f^\prime(x)[h]=A(x)[h]\qquad \forall\, h\in X.
	\end{equation*}
\end{definition}

\begin{lemma}[Fréchet differentiability of $\cJ$]\label{lem:Jdiff}
	The functional $\cJ$ defined in \eqref{eq:energyfunc} is Fréchet differentiable on $V$. Its derivative at $u\in V$ is given by
	\begin{equation}\label{eq:Jprime}
		\cJ'(u)[h] = a(u,h)-\ell(h)\qquad \forall\, h\in V.
	\end{equation}
\end{lemma}

\begin{proof}
	Let $u,k\in V$. Using bilinearity of $a$ and linearity of $\ell$, we obtain
	\begin{align*}
		\cJ(u+k)-\cJ(u)
		 & =\frac12\bigl(a(u+k,u+k)-a(u,u)\bigr)-\ell(k)            \\
		 & =\frac12\bigl(a(u,u)+2a(u,k)+a(k,k)-a(u,u)\bigr)-\ell(k) \\
		 & = a(u,k)-\ell(k)+\frac12\,a(k,k).
	\end{align*}
	Hence
	\begin{equation*}
		\cJ(u+k)-\cJ(u)-\bigl(a(u,k)-\ell(k)\bigr)
		=\frac12\,a(k,k).
	\end{equation*}
	Define $A(u):V\to\mathbb R$ by
	\begin{equation*}
		A(u)[h]:=a(u,h)-\ell(h).
	\end{equation*}
	Since $a$ and $\ell$ are bounded, $A(u)$ is linear and bounded. Moreover,
	\begin{equation*}
		\frac{\bigl|\cJ(u+k)-\cJ(u)-A(u)[k]\bigr|}{\|k\|_{1,\Omega}}
		=\frac{1}{2}\,\frac{|a(k,k)|}{\|k\|_{1,\Omega}}
		\le \frac12\,C_{\mathcal A}\,\|k\|_{1,\Omega}\xrightarrow{\ \|k\|_{1,\Omega}\to 0\ }0,
	\end{equation*}
	by boundedness of $a$, cf.\ Proposition \ref{pr:aproperties}. Therefore, $\cJ$ is Fréchet differentiable at $u$ with
	\begin{equation*}
		\cJ^\prime(u)[h]=A(u)[h]=a(u,h)-\ell(h)
		\qquad \forall\, h\in V.
	\end{equation*}
\end{proof}


\begin{proposition}[First-order optimality condition on convex sets]\label{pr:FOC}
	A function $u\in K$ minimizes $\cJ$ over $K$ if and only if it satisfies the variational inequality
	\begin{equation}\label{eq:firstorderopt}
		\cJ^\prime(u)[v-u]\ge 0\qquad\forall\, v\in K.
	\end{equation}
\end{proposition}
\begin{proof}
	Assume first that $u\in K$ minimizes $\cJ$ over $K$ and let $v\in K$.
	By convexity of $K$ we have $u+t(v-u)\in K$ for all $t\in[0,1]$.
	Define $\phi(t):=\cJ(u+t(v-u))$. Then $\phi$ attains its minimum at $t=0$, hence $\phi'(0^+)\ge 0$, where $\phi'(0^+)$ denotes the right-sided derivative at $t=0$. Using the chain rule and Fréchet differentiability of $\cJ$ we obtain
	\begin{equation*}
		\phi'(0^+)=\cJ'(u)[v-u]\ge 0.
	\end{equation*}
	Conversely, assume \eqref{eq:firstorderopt}. Let $v\in K$ be arbitrary and set $h:=v-u$. By Taylor expansion of the quadratic functional $\cJ$ we have
	\begin{equation*}
		\cJ(u+h)-\cJ(u)=\cJ'(u)[h]+\frac12\,a(h,h).
	\end{equation*}
	By \eqref{eq:firstorderopt}, $\cJ'(u)[h]\ge 0$, and since $a(h,h)\ge 0$ we conclude $\cJ(v)\ge \cJ(u)$ for all $v\in K$. Hence $u$ is a minimizer over $K$.
\end{proof}

Combining Lemma \ref{lem:Jdiff} and Proposition \ref{pr:FOC}, $u\in K$ solves Problem \ref{pr:signorini1} if and only if
\begin{equation*}
	\cJ^\prime(u)[v-u] = a(u,v-u)-\ell(v-u)\ge 0 \qquad \forall\, v\in K.
\end{equation*}
This is the first weak formulation of Signorini's problem as a variational inequality.

\begin{problem}[Signorini's Problem -- Weak Formulation]\label{pr:signorini2}
Find $u\in K$ such that
\begin{equation*}
	a(u,v-u)\ge \ell(v-u)\qquad \forall\, v\in K.
\end{equation*}
\end{problem}

Well-posedness of Problem \ref{pr:signorini2} is already ensured, since it is equivalent to the minimization Problem \ref{pr:signorini1} by Proposition \ref{pr:FOC} in the sense that each solution to the one of the problems is a solution to the other problem. The solution is unique due to Corollary \ref{cor:wellposed} with the stability estimate \eqref{eq:ustab}. Another useful equivalent formulation is the following.

\begin{problem}[Signorini's Problem -- Weak Formulation]\label{pr:signorini3}
Find $u\in K$ such that
\begin{equation*}
	\begin{cases}
		a(u,u) = \ell(u), &                   \\[2pt]
		a(u,v)\ge \ell(v) & \forall\, v\in K.
	\end{cases}
\end{equation*}
\end{problem}

\begin{proposition}[Equivalence of Problem \ref{pr:signorini2} and Problem \ref{pr:signorini3}]\label{pr:equivalence23} Any solution $u\in K$ to Problem \ref{pr:signorini2} is also a solution to Problem \ref{pr:signorini3} and vice versa.
\end{proposition}
\begin{proof}
	Let $u\in K$ solve Problem \ref{pr:signorini3}. Then
	\begin{equation*}
		a(u,v-u) = a(u,v)-a(u,u)  = a(u,v)-\ell(u)\ge \ell(v)-\ell(u) = \ell(v-u).
	\end{equation*}
	Assuming that $u\in K$ solves Problem \ref{pr:signorini2} and testing $v=0$ yields
	\begin{equation*}
		a(u,-u)\ge \ell(-u)\qquad\Longleftrightarrow\qquad -a(u,u)\ge -\ell(u),
	\end{equation*}
	which directly implies
	\begin{equation*}
		a(u,u)\le \ell(u).
	\end{equation*}
	However, testing with $v=2u$ yields $a(u,u)\ge \ell(u)$ and thus $a(u,u) = \ell(u)$. Furthermore, Problem \ref{pr:signorini2} reads
	\begin{equation*}
		a(u,v-u) = a(u,v)-a(u,u)\ge \ell(v-u) = \ell(v)-\ell(u),
	\end{equation*}
	which directly implies
	\begin{equation*}
		a(u,v)\ge a(u,u)+\ell(v)-\ell(u).
	\end{equation*}
	Plugging in $a(u,u) = \ell(u)$ yields $a(u,v)\ge \ell(v)$.
\end{proof}


We close this section by providing another stability result on the dependency of solutions to varying source terms as input data.

\begin{proposition}[Lipschitz dependence on the data]\label{pr:lipschitz}
	For $i\in\{1,2\}$, let $f_i\in L^2(\Omega;\mathbb R^d)$ and $F_i\in L^2(\Gamma_N;\mathbb R^d)$, and let $\ell_i\in V'$ be the corresponding load functional, i.e.
	\begin{equation*}
		\ell_i(v):=\int_\Omega f_i\cdot v\;\dx+\int_{\Gamma_N} F_i\cdot \gamma v\;\ds,
		\qquad v\in V.
	\end{equation*}
	Moreover, let $u_i\in K$ be the unique solution of Problem \ref{pr:signorini2} associated with $\ell_i$, i.e.
	\begin{equation*}
		a(u_i,v-u_i)\ge \ell_i(v-u_i)\qquad \forall\, v\in K.
	\end{equation*}
	Then the solution depends Lipschitz continuously on the load functional, i.e.
	\begin{equation}\label{eq:lipschitzabstract}
		\|u_1-u_2\|_{1,\Omega}\le \frac{C_K^2}{\alpha_\mathcal{A}}\,\|\ell_1-\ell_2\|_{V'}.
	\end{equation}
	In particular, it depends Lipschitz continuously on the data $f_i,F_i$ in the sense that
	\begin{equation}\label{eq:lipschitzloads}
		\|u_1-u_2\|_{1,\Omega}
		\le \frac{C_K^2\,\max\{1,C_\mathrm{tr}\}}{\alpha_\mathcal{A}}
		\Bigl(\|f_1-f_2\|_{0,\Omega}+\|F_1-F_2\|_{0,\Gamma_N}\Bigr).
	\end{equation}
\end{proposition}

\begin{proof}
	For $u_1$, choose $v=u_2\in K$ in Problem \ref{pr:signorini2}. Then
	\begin{equation*}
		a(u_1,u_2-u_1)\ge \ell_1(u_2-u_1).
	\end{equation*}
	Similarly, choosing $v=u_1\in K$ for $u_2$ yields
	\begin{equation*}
		a(u_2,u_1-u_2)\ge \ell_2(u_1-u_2).
	\end{equation*}
	Adding both inequalities and using symmetry of $a$, we obtain
	\begin{equation*}
		a(u_1-u_2,u_1-u_2)\le (\ell_1-\ell_2)(u_1-u_2).
	\end{equation*}
	By Proposition \ref{pr:aproperties}, the bilinear form $a$ is $V$-elliptic with constant $\alpha_\cA/C_K^2$. Hence
	\begin{equation*}
		\frac{\alpha_\mathcal{A}}{C_K^2}\,\|u_1-u_2\|_{1,\Omega}^2
		\le a(u_1-u_2,u_1-u_2)
		\le (\ell_1-\ell_2)(u_1-u_2)
		\le \|\ell_1-\ell_2\|_{V'}\,\|u_1-u_2\|_{1,\Omega}.
	\end{equation*}
	If $u_1=u_2$, there is nothing to prove. Otherwise, dividing by $\|u_1-u_2\|_{1,\Omega}$ yields \eqref{eq:lipschitzabstract}. üMoreover,
	\begin{equation*}
		(\ell_1-\ell_2)(v)
		=\int_\Omega (f_1-f_2)\cdot v\;\dx+\int_{\Gamma_N} (F_1-F_2)\cdot \gamma v\;\ds
		\qquad \forall\, v\in V.
	\end{equation*}
	Therefore, Proposition \ref{pr:aproperties} applied to the functional $\ell_1-\ell_2$ implies
	\begin{equation*}
		\|\ell_1-\ell_2\|_{V'}
		\le \max\{1,C_\mathrm{tr}\}
		\Bigl(\|f_1-f_2\|_{0,\Omega}+\|F_1-F_2\|_{0,\Gamma_N}\Bigr).
	\end{equation*}
	Combining this with \eqref{eq:lipschitzabstract}, we obtain \eqref{eq:lipschitzloads}.
\end{proof}

\subsection{Strong Formulation}
From the previous weak formulations of Signorini's problem, we know there exists a uniquely determined $u\in K$ that solves the problem. To derive a {\em strong} PDE formulation, we have to discuss what was $\sigma(u)\mathrm{n}$ means and in what sense the contact conditions on $\Gamma_C$ can be described as equalities and inequalities.
\subsubsection{Stress Regularity and Boundary Tractions}

In the weak formulation, the solution satisfies only $u\in K\subset H^1(\Omega;\R^d)$. Hence
\begin{equation*}
	\varepsilon(u)\in L^2(\Omega;\mathrm{Sym}^2(\R^d))
	\qquad\text{and}\qquad
	\sigma(u)=\mathcal A:\varepsilon(u)\in L^2(\Omega;\mathrm{Sym}^2(\R^d)).
\end{equation*}
This is sufficient to define the bilinear form
\begin{equation*}
	a(u,v)=\int_\Omega \sigma(u):\varepsilon(v)\;\dx,
\end{equation*}
but not yet enough to interpret the equilibrium equation $-\diver\sigma(u)=f$ in $L^2(\Omega;\R^d)$ or to define boundary tractions in a classical pointwise sense. To formulate a strong equilibrium equation and the corresponding boundary forces, we therefore assume the additional regularity
\begin{equation*}
	\sigma(u)\in H(\diver;\Omega).
\end{equation*}
By the general results from Section~\ref{subsec:Hdiv}, the normal trace operator
\begin{equation*}
	\gamma_{\mathrm n}:H(\diver;\Omega)\to H^{-1/2}(\Gamma;\R^d)
\end{equation*}
is well-defined. Accordingly, for $\tau\in H(\diver;\Omega)$, its boundary traction is denoted by
\begin{equation*}
	\tau\,\mathrm n:=\gamma_{\mathrm n}(\tau)\in H^{-1/2}(\Gamma;\R^d),
\end{equation*}
which agrees with the classical expression in the smooth case. Since every stress tensor $\tau\in H(\diver;\Omega)$ is symmetric, we have
\begin{equation*}
	\tau:\nabla v=\tau:\varepsilon(v)
	\qquad \forall\, v\in H^1(\Omega;\R^d).
\end{equation*}
Therefore, the Green formula \eqref{eq:greenHdiv} takes the specific form
\begin{equation}\label{eq:green}
	\langle \tau\,\mathrm n,\gamma v\rangle_\Gamma
	=
	\int_\Omega \tau:\varepsilon(v)\;\dx
	+
	\int_\Omega (\diver\tau)\cdot v\;\dx
	\qquad
	\forall\, v\in H^1(\Omega;\R^d).
\end{equation}
In particular, if $\sigma(u)\in H(\diver;\Omega)$, then
\begin{equation}\label{eq:green_sigma}
	\langle \sigma(u)\mathrm n,\gamma v\rangle_\Gamma
	=
	\int_\Omega \sigma(u):\varepsilon(v)\;\dx
	+
	\int_\Omega \diver\sigma(u)\cdot v\;\dx
	\qquad
	\forall\, v\in H^1(\Omega;\R^d).
\end{equation}
This identity will be used in the derivation of the strong Signorini conditions.


\subsubsection{A Hybrid Formulation}

The weak formulation in Problem \ref{pr:signorini2} encodes the equilibrium equation and the contact conditions only implicitly through the variational inequality. Under the additional stress regularity $\sigma(u)\in H(\diver;\Omega)$, these conditions can be separated into a bulk equation in $\Omega$ and boundary relations involving the traction $\sigma(u)\mathrm n$. This leads to the following hybrid formulation, which may be viewed as an intermediate form between the weak variational inequality and the strong Signorini conditions in the later subsections.

\begin{problem}[Signorini's Problem -- Hybrid Formulation]\label{pr:signorini4}
For given $f\in L^2(\Omega;\R^d)$ and $F\in L^2(\Gamma_N;\R^d)$, find $u\in K$ such that $\sigma(u)\in H(\diver;\Omega)$ and
\begin{subequations}\label{eq:hybrid}
	\begin{empheq}[left=\empheqlbrace]{alignat=2}
		-\diver\sigma(u) &= f
		&\qquad& \text{a.e.\ in }\Omega, \label{eq:hybrid_bulk}\\
		\langle \sigma(u)\mathrm n,\gamma u\rangle_\Gamma
		&= \mathrlap{\int_{\Gamma_N} F\cdot \gamma u\;\ds,}
		&& \label{eq:hybrid_eq}\\
		\langle \sigma(u)\mathrm n,\gamma v\rangle_\Gamma
		&\ge \int_{\Gamma_N} F\cdot \gamma v\;\ds
		&\qquad& \forall\, v\in K.
		\label{eq:hybrid_ineq}
	\end{empheq}
\end{subequations}
\end{problem}

We close this section by showing, that the hybrid formulation above is equivalent to the previous weak formulations in Problem \ref{pr:signorini2} in order to prove that it is indeed a valid formulation of the Signorini problem.

\begin{proposition}[Equivalence of hybrid and weak formulation]
	Assume that $\sigma(u)\in H(\diver;\Omega)$. Then $u\in K$ solves Problem \ref{pr:signorini2} if and only if it solves Problem \ref{pr:signorini4}.
\end{proposition}
\begin{proof}
	Assume first that $u\in K$ solves Problem \ref{pr:signorini2}, i.e.
	\begin{equation}\label{eq:weakineq_hybrid}
		a(u,v-u)\ge \ell(v-u)\qquad \forall\, v\in K.
	\end{equation}
	Let $\varphi\in\mathcal D(\Omega;\R^d)$. Since $\gamma\varphi=0$ on $\Gamma$, we have $u\pm t\varphi\in K$ for all sufficiently small $t>0$. Testing \eqref{eq:weakineq_hybrid} with $v=u\pm t\varphi$ yields
	\begin{equation*}
		a(u,\varphi)=\ell(\varphi).
	\end{equation*}
	Because $\varphi$ has compact support in $\Omega$, the boundary term in $\ell(\varphi)$ vanishes, and thus
	\begin{equation*}
		a(u,\varphi)=\int_\Omega f\cdot \varphi\;\dx.
	\end{equation*}
	Moreover, since $\sigma(u)$ is symmetric,
	\begin{equation*}
		a(u,\varphi)=\int_\Omega \sigma(u):\varepsilon(\varphi)\;\dx
		=\int_\Omega \sigma(u):\nabla\varphi\;\dx.
	\end{equation*}
	By definition of the distributional divergence, this means
	\begin{equation*}
		\langle-\diver\sigma(u),\,\varphi\rangle
		=
		\int_\Omega \sigma(u):\nabla\varphi\;\dx
		=
		\int_\Omega f\cdot\varphi\;\dx
		=
		\langle f,\,\varphi\rangle
		\qquad \forall\,\varphi\in\mathcal D(\Omega;\R^d).
	\end{equation*}
	Hence,
	\begin{equation*}
		-\diver\sigma(u)=f
		\qquad\text{in }\mathcal D'(\Omega;\R^d).
	\end{equation*}
	Since $\sigma(u)\in H(\diver;\Omega)$, we have $\diver\sigma(u)\in L^2(\Omega;\R^d)$, and therefore \eqref{eq:hybrid_bulk} holds. Next, let $v\in K$ and set $w:=v-u$. By \eqref{eq:green_sigma} and \eqref{eq:hybrid_bulk},
	\begin{equation*}
		a(u,w)
		=
		\langle \sigma(u)\mathrm n,\gamma w\rangle_\Gamma
		-\int_\Omega \diver\sigma(u)\cdot w\;\dx
		=
		\langle \sigma(u)\mathrm n,\gamma w\rangle_\Gamma
		+\int_\Omega f\cdot w\;\dx.
	\end{equation*}
	On the other hand, \eqref{eq:weakineq_hybrid} gives
	\begin{equation*}
		a(u,w)\ge \ell(w)=\int_\Omega f\cdot w\;\dx+\int_{\Gamma_N}F\cdot\gamma w\;\ds.
	\end{equation*}
	Subtracting the volume term yields
	\begin{equation*}
		\langle \sigma(u)\mathrm n,\gamma w\rangle_\Gamma
		\ge
		\int_{\Gamma_N}F\cdot\gamma w\;\ds,
	\end{equation*}
	which with $w= v-u$ is exactly
	\begin{equation*}
		\langle \sigma(u)\mathrm n,\gamma v\rangle_\Gamma
		-
		\langle \sigma(u)\mathrm n,\gamma u\rangle_\Gamma
		\ge
		\int_{\Gamma_N}F\cdot\gamma(v-u)\;\ds
		\qquad \forall\, v\in K.
	\end{equation*}
	It thus remains to prove \eqref{eq:hybrid_eq}. Then \eqref{eq:hybrid_ineq} holds automatically. Since $0\in K$ and $2u\in K$, testing \eqref{eq:weakineq_hybrid} with $v=0$ and $v=2u$ gives
	\begin{equation*}
		a(u,u)=\ell(u).
	\end{equation*}
	Applying \eqref{eq:green_sigma} with $v=u$ and using \eqref{eq:hybrid_bulk}, we obtain
	\begin{equation*}
		a(u,u)
		=
		\langle \sigma(u)\mathrm n,\gamma u\rangle_\Gamma
		+\int_\Omega f\cdot u\;\dx.
	\end{equation*}
	Together with
	\begin{equation*}
		\ell(u)=\int_\Omega f\cdot u\;\dx+\int_{\Gamma_N}F\cdot\gamma u\;\ds,
	\end{equation*}
	this yields \eqref{eq:hybrid_eq}. Consequently, the previous inequality becomes
	\begin{equation*}
		\langle \sigma(u)\mathrm n,\gamma v\rangle_\Gamma
		\ge
		\int_{\Gamma_N}F\cdot\gamma v\;\ds
		\qquad \forall\, v\in K,
	\end{equation*}
	which is exactly \eqref{eq:hybrid_ineq}. Thus $u$ solves Problem \ref{pr:signorini4}. Conversely, assume that $u\in K$ solves Problem \ref{pr:signorini4}. Let $v\in K$ and set $w:=v-u$. Using \eqref{eq:green_sigma} and \eqref{eq:hybrid_bulk}, we get
	\begin{equation*}
		a(u,w)
		=
		\langle \sigma(u)\mathrm n,\gamma w\rangle_\Gamma
		+\int_\Omega f\cdot w\;\dx.
	\end{equation*}
	Moreover, subtracting \eqref{eq:hybrid_eq} from \eqref{eq:hybrid_ineq} yields
	\begin{equation*}
		\langle \sigma(u)\mathrm n,\gamma w\rangle_\Gamma
		\ge
		\int_{\Gamma_N}F\cdot\gamma w\;\ds.
	\end{equation*}
	Hence
	\begin{equation*}
		a(u,w)
		\ge
		\int_\Omega f\cdot w\;\dx+\int_{\Gamma_N}F\cdot\gamma w\;\ds
		=
		\ell(w)
		=
		\ell(v-u).
	\end{equation*}
	Since $w=v-u$, this is exactly
	\begin{equation*}
		a(u,v-u)\ge \ell(v-u)\qquad \forall\, v\in K,
	\end{equation*}
	i.e.\ $u$ solves Problem \ref{pr:signorini2}.
\end{proof}


\subsubsection{Normal/Tangential Splitting and Contact Trace Spaces}
The hybrid formulation derived above still encodes the contact interaction through a single boundary traction functional acting on traces. In order to recover the Signorini contact conditions on $\Gamma_C$ in a meaningful weak sense, we now split traces and tractions into normal and tangential components and introduce the corresponding trace spaces on the contact boundary. Recall that for $v\in V$, the trace $\gamma v|_{\Gamma_C}$ is well-defined. We define the contact trace space by
\begin{equation*}
	W_C := \left\{\gamma v|_{\Gamma_C}\; \middle|\; v\in V\right\}\subset H^{1/2}(\Gamma_C;\R^d).
\end{equation*}
For $w\in W_C$, we define its normal and tangential components by
\begin{equation*}
	w_\mathrm{n} := w\cdot \mathrm n,
	\qquad
	w_t := w - (w\cdot \mathrm n)\mathrm n.
\end{equation*}
Accordingly, we introduce the normal and tangential trace spaces
\begin{equation*}
	W_{C,\mathrm{n}} := \left\{w\cdot \mathrm n\; \middle|\; w\in W_C\right\},
	\qquad
	W_{C,t} := \left\{w\in W_C\; \middle|\; w\cdot \mathrm n = 0\right\}.
\end{equation*}
Hence every $w\in W_C$ admits the decomposition
\begin{equation*}
	w = w_\mathrm{n}\mathrm n + w_t\in W_{C,\mathrm{n}}\,\mathrm{n}\oplus W_{C,t}
\end{equation*}
We endow $W_C$, $W_{C,\mathrm{n}}$ and $W_{C,t}$ with the quotient norms induced by $V$, namely
\begin{equation}\label{eq:norm_WC}
	\|w\|_{W_C}
	:=
	\inf\left\{\|v\|_{1,\Omega}\; \middle|\; v\in V,\ \gamma v|_{\Gamma_C}=w\right\},
	\qquad w\in W_C,
\end{equation}
and
\begin{equation}\label{eq:norm_WCn}
	\|\eta\|_{W_{C,\mathrm{n}}}
	:=
	\inf\left\{\|v\|_{1,\Omega}\; \middle|\; v\in V,\ (\gamma v|_{\Gamma_C})\cdot \mathrm n=\eta\right\},
	\qquad \eta\in W_{C,\mathrm{n}},
\end{equation}
and
\begin{equation}\label{eq:norm_WCt}
	\|\zeta\|_{W_{C,t}}
	:=
	\inf\left\{\|v\|_{1,\Omega}\; \middle|\; v\in V,\ (\gamma v|_{\Gamma_C})_t=\zeta\right\},
	\qquad \zeta\in W_{C,t}.
\end{equation}
We denote by $W_C'$, $W_{C,\mathrm{n}}^\prime$ and $W_{C,t}^\prime$ the corresponding dual spaces, equipped with the operator norms. Their duality pairings are denoted by $\langle \cdot,\cdot\rangle_{\Gamma_C}$. Whenever no confusion can arise, we use the same notation $\langle\cdot,\cdot\rangle_{\Gamma_C}$ for the pairings on
$W_C^\prime\times W_C$, $W_{C,\mathrm{n}}^\prime\times W_{C,\mathrm{n}}$ and $W_{C,t}^\prime\times W_{C,t}$. In the hybrid formulation, the boundary traction on $\Gamma_C$ is understood as a bounded linear functional on $W_C$.
Hence, for a contact traction $\tau\in W_C^\prime$, one can define normal and tangential components
\begin{equation}\label{eq:traction_split_dual}
	\tau_\mathrm{n}\in W_{C,\mathrm{n}}^\prime
	\quad\text{and}\quad
	\tau_t\in W_{C,t}^\prime,
\end{equation}
such that, for every $w\in W_C$ with decomposition $w=w_\mathrm{n}\mathrm n+w_t$,
\begin{equation}\label{eq:dual_split_pairing}
	\langle \tau,w\rangle_{\Gamma_C}
	=
	\langle \tau_\mathrm{n},w_\mathrm{n}\rangle_{\Gamma_C}
	+
	\langle \tau_t,w_t\rangle_{\Gamma_C}.
\end{equation}
To express the Signorini contact condition on the normal component in dual form, we introduce the cone of admissible normal traces
\begin{equation}\label{eq:cone_WCn_minus}
	W_{C,\mathrm{n}}^-
	:=
	\left\{\eta\in W_{C,\mathrm{n}}\; \middle|\; \eta\le 0 \text{ a.e.\ on }\Gamma_C\right\},
\end{equation}
and its dual cone of weakly nonpositive normal contact forces
\begin{equation}\label{eq:def_LambdaC}
	\Lambda_C
	:=
	\left\{\lambda\in W_{C,\mathrm{n}}'\; \middle|\; \langle \lambda,\eta\rangle_{\Gamma_C} \ge 0 \quad \forall\,\eta\in W_{C,\mathrm{n}}^-\right\}.
\end{equation}
With these notations, the boundary inequality in the hybrid formulation can be decomposed into normal and tangential contributions on $\Gamma_C$.
This allows us to state the Signorini contact conditions as a KKT-type system in the dual spaces $W_{C,\mathrm{n}}'$ and $W_{C,t}'$.

\subsubsection{KKT Contact Conditions}

Under the assumption of having no Neumann boundary part, $|\Gamma_N|$, we can make a strong formulation of Signorini's Problem. We denote the normal/tangential components of the boundary stress by
\begin{equation*}
	\sigma(u)\mathrm{n} = \sigma_\mathrm{n}(u)\mathrm{n}+\sigma_t(u).
\end{equation*}

\begin{problem}[Signorini's Problem -- Strong Formulation]\label{pr:signorini5}
For given $f\in L^2(\Omega;\R^d)$, find $u\in V$ such that
\begin{subequations}\label{eq:strong}
	\begin{empheq}[left=\empheqlbrace]{alignat=2}
		-\diver \sigma(u) &= f
		&\qquad& \text{a.e.\ in }\Omega, \label{eq:strong_bulk}\\
		\sigma_t(u) &= 0
		&\qquad& \text{in }W_{C,t}^\prime, \label{eq:strong_tangential}\\
		u_\mathrm{n} &\le 0
		&\qquad& \text{a.e.\ on }\Gamma_C, \label{eq:strong_nonpen}\\
		\sigma_\mathrm{n}(u) &\in \Lambda_C,
		&& \label{eq:strong_lambda}\\
		\langle\sigma_\mathrm{n}(u),\, u_\mathrm{n}\rangle_{\Gamma_C} &= 0.
		&& \label{eq:strong_comp}
	\end{empheq}
\end{subequations}
\end{problem}

\begin{proposition}[Equivalence of Strong and Weak Formulation]
	Suppose that $|\Gamma_N|= 0$. Then every solution to Problem \ref{pr:signorini5} is also a solution to Problem \ref{pr:signorini4} and vice versa.
\end{proposition}
\begin{proof}
	We prove directly the equivalence between Problems \ref{pr:signorini4} and \ref{pr:signorini5}. Assume first that $u\in V$ solves Problem \ref{pr:signorini5}. Since $u\in V$, we have
	\begin{equation*}
		\sigma(u)=\mathcal A:\varepsilon(u)\in L^2(\Omega;\mathrm{Sym}^2(\R^d)).
	\end{equation*}
	Moreover, \eqref{eq:strong_bulk} yields
	\begin{equation*}
		\diver\sigma(u)=-f\in L^2(\Omega;\R^d).
	\end{equation*}
	Hence $\sigma(u)\in H(\diver;\Omega)$. In particular, \eqref{eq:hybrid_bulk} holds. Now let $v\in K$. Since $|\Gamma_N|=0$ and $\gamma v=0$ on $\Gamma_D$, we have
	\begin{equation*}
		\langle \sigma(u)\mathrm n,\gamma v\rangle_\Gamma
		=
		\langle \sigma(u)\mathrm n,\gamma v\rangle_{\Gamma_C}.
	\end{equation*}
	Using the decomposition into normal and tangential components, we get
	\begin{equation*}
		\langle \sigma(u)\mathrm n,\gamma v\rangle_{\Gamma_C}
		=
		\langle \sigma_\mathrm n(u),v_\mathrm n\rangle_{\Gamma_C}
		+
		\langle \sigma_t(u),v_t\rangle_{\Gamma_C}.
	\end{equation*}
	By \eqref{eq:strong_tangential}, the second term vanishes. Therefore,
	\begin{equation*}
		\langle \sigma(u)\mathrm n,\gamma v\rangle_\Gamma
		=
		\langle \sigma_\mathrm n(u),v_\mathrm n\rangle_{\Gamma_C}.
	\end{equation*}
	Since $v\in K$, we have $v_\mathrm n\le 0$ a.e.\ on $\Gamma_C$, and by \eqref{eq:strong_lambda} we know that $\sigma_\mathrm n(u)\in\Lambda_C$. Hence
	\begin{equation*}
		\langle \sigma_\mathrm n(u),v_\mathrm n\rangle_{\Gamma_C}\ge 0,
	\end{equation*}
	which proves \eqref{eq:hybrid_ineq}, because $|\Gamma_N|=0$. Taking $v=u$ and using \eqref{eq:strong_comp}, we similarly obtain
	\begin{equation*}
		\langle \sigma(u)\mathrm n,\gamma u\rangle_\Gamma
		=
		\langle \sigma_\mathrm n(u),u_\mathrm n\rangle_{\Gamma_C}
		=0,
	\end{equation*}
	which is exactly \eqref{eq:hybrid_eq}. Thus $u$ solves Problem \ref{pr:signorini4}. Conversely, assume that $u\in K$ solves Problem \ref{pr:signorini4}. Then $u\in V$, and \eqref{eq:strong_nonpen} is already contained in the definition of $K$. Moreover, \eqref{eq:hybrid_bulk} is exactly \eqref{eq:strong_bulk}. We next prove \eqref{eq:strong_tangential}. Let $w_t\in W_{C,t}$. By definition of $W_{C,t}$, there exists $w\in V$ such that
	\begin{equation*}
		\gamma w|_{\Gamma_C}=w_t
		\qquad\text{and}\qquad
		w_\mathrm n=0.
	\end{equation*}
	Since $u\in K$ and $w_\mathrm n=0$, we have $u\pm w\in K$. Hence \eqref{eq:hybrid_ineq} applied to $v=u+w$ and $v=u-w$, together with \eqref{eq:hybrid_eq}, yields
	\begin{equation*}
		\langle \sigma(u)\mathrm n,\gamma w\rangle_\Gamma\ge 0
		\qquad\text{and}\qquad
		\langle \sigma(u)\mathrm n,\gamma w\rangle_\Gamma\le 0.
	\end{equation*}
	Therefore,
	\begin{equation*}
		\langle \sigma(u)\mathrm n,\gamma w\rangle_\Gamma=0.
	\end{equation*}
	Again using $|\Gamma_N|=0$ and $\gamma w=0$ on $\Gamma_D$, we obtain
	\begin{equation*}
		0=\langle \sigma(u)\mathrm n,\gamma w\rangle_{\Gamma_C}
		=\langle \sigma_\mathrm n(u),w_\mathrm n\rangle_{\Gamma_C}
		+\langle \sigma_t(u),w_t\rangle_{\Gamma_C}
		=\langle \sigma_t(u),w_t\rangle_{\Gamma_C}.
	\end{equation*}
	Since $w_t\in W_{C,t}$ was arbitrary, this proves \eqref{eq:strong_tangential}. We now prove \eqref{eq:strong_lambda}. Let $\eta\in W_C$ satisfy $\eta\le 0$ a.e.\ on $\Gamma_C$. By definition of $W_C$, there exists $v\in V$ such that $v_\mathrm n=\eta$. Because $\eta\le 0$, we have $v\in K$. Hence \eqref{eq:hybrid_ineq} gives
	\begin{equation*}
		\langle \sigma(u)\mathrm n,\gamma v\rangle_\Gamma\ge 0.
	\end{equation*}
	Using $|\Gamma_N|=0$, $\gamma v=0$ on $\Gamma_D$, and \eqref{eq:strong_tangential}, we infer
	\begin{equation*}
		0\le \langle \sigma(u)\mathrm n,\gamma v\rangle_{\Gamma_C}
		=\langle \sigma_\mathrm n(u),v_\mathrm n\rangle_{\Gamma_C}
		+\langle \sigma_t(u),v_t\rangle_{\Gamma_C}
		=\langle \sigma_\mathrm n(u),\eta\rangle_{\Gamma_C}.
	\end{equation*}
	This proves $\sigma_\mathrm n(u)\in\Lambda_C$, i.e.\ \eqref{eq:strong_lambda}. Finally, \eqref{eq:hybrid_eq}, together with $|\Gamma_N|=0$, $\gamma u=0$ on $\Gamma_D$, and \eqref{eq:strong_tangential}, yields
	\begin{equation*}
		0=\langle \sigma(u)\mathrm n,\gamma u\rangle_\Gamma
		=\langle \sigma(u)\mathrm n,\gamma u\rangle_{\Gamma_C}
		=\langle \sigma_\mathrm n(u),u_\mathrm n\rangle_{\Gamma_C}
		+\langle \sigma_t(u),u_t\rangle_{\Gamma_C}
		=\langle \sigma_\mathrm n(u),u_\mathrm n\rangle_{\Gamma_C},
	\end{equation*}
	which is exactly \eqref{eq:strong_comp}. Hence $u$ solves Problem \ref{pr:signorini5}.
\end{proof}

\subsubsection{Classical Strong Form under Additional Regularity}

Finally, under higher regularity assumptions on $u$, we can formulate Signorini's problem in a natural strong way. Here $|\Gamma_N|>0$ is allowed.
\begin{problem}[Signorini's Problem -- Classical Strong Formulation]\label{pr:signorini6}
For given $f\in L^2(\Omega;\R^d)$ and $F\in L^2(\Gamma_N;\R^d)$ find $u\in H^s(\Omega;\R^d)$ with $s>3/2$ such that the small strain elasticity equations
\begin{subequations}\label{eq:strong_classical}
	\begin{empheq}[left=\empheqlbrace]{alignat=2}
		-\diver\sigma(u) &= f
		&\qquad& \text{in }\Omega, \label{eq:strong_classical_bulk}\\
		\sigma(u) &= \mathcal A:\varepsilon(u)
		&\qquad& \text{in }\Omega, \label{eq:strong_classical_constitutive}\\
		u &= 0
		&\qquad& \text{on }\Gamma_D, \label{eq:strong_classical_dirichlet}\\
		\sigma(u)\mathrm n &= F
		&\qquad& \text{on }\Gamma_N, \label{eq:strong_classical_neumann}
	\end{empheq}
\end{subequations}
and the contact conditions
\begin{subequations}\label{eq:strong_contact}
	\begin{empheq}[left=\empheqlbrace]{alignat=2}
		u_\mathrm{n} &\le 0
		&\qquad& \text{on }\Gamma_C, \label{eq:strong_contact_nonpen}\\
		\sigma_\mathrm{n}(u) &\le 0
		&\qquad& \text{on }\Gamma_C, \label{eq:strong_contact_sigma_n}\\
		\sigma_\mathrm{n}(u)\,u_\mathrm{n} &= 0
		&\qquad& \text{on }\Gamma_C, \label{eq:strong_contact_comp}\\
		\sigma_t(u) &= 0
		&\qquad& \text{on }\Gamma_C \label{eq:strong_contact_tangential}
	\end{empheq}
\end{subequations}
hold.
\end{problem}


To prove equivalence to one of the previous formulations, we need the following Lemma.

\begin{lemma}[Localized boundary liftings]\label{lem:locboundlift}
	Assume the boundary decomposition is sufficiently regular. Then the following hold:
	\begin{itemize}
		\item[(i)] For every $g_N\in H^{1/2}(\Gamma_N;\R^d)$ there exists $w\in V$ such that
		      \begin{equation*}
			      \gamma w=g_N \text{ on }\Gamma_N,
			      \qquad
			      \gamma w=0 \text{ on }\Gamma_C.
		      \end{equation*}
		\item[(ii)] For every $\zeta\in W_{C,t}$ there exists $w\in V$ such that
		      \begin{equation*}
			      \gamma w=0 \text{ on }\Gamma_N,
			      \qquad
			      w_\mathrm{n}=0,\quad w_t=\zeta \text{ on }\Gamma_C.
		      \end{equation*}
		\item[(iii)] For every $\eta\in W_{C,\mathrm{n}}$ there exists $w\in V$ such that
		      \begin{equation*}
			      \gamma w=0 \text{ on }\Gamma_N,
			      \qquad
			      w_t=0,\quad w_\mathrm{n}=\eta \text{ on }\Gamma_C.
		      \end{equation*}
	\end{itemize}
\end{lemma}
\begin{proof}
	We repeatedly use the lifting operator on boundary parts provided by the trace and lifting theory from the mathematical framework. For (i), let $g_N\in H^{1/2}(\Gamma_N;\R^d)$. By the regularity assumption on the boundary decomposition, its zero extension
	\begin{equation*}
		\mathrm{Ext}_0(g_N)
		:=
		\begin{cases}
			g_N & \text{on }\Gamma_N,                \\
			0   & \text{on }\Gamma\setminus\Gamma_N,
		\end{cases}
	\end{equation*}
	belongs to $H^{1/2}(\Gamma;\R^d)$. Let $w\in H^1(\Omega;\R^d)$ be a lifting of $\mathrm{Ext}_0(g_N)$, i.e. $\gamma w=\mathrm{Ext}_0(g_N)$ on $\Gamma$. Then, by construction,
	\begin{equation*}
		\gamma w=g_N \text{ on }\Gamma_N,
		\qquad
		\gamma w=0 \text{ on }\Gamma_C,
	\end{equation*}
	and also $\gamma w=0$ on $\Gamma_D$. Hence $w\in V$. This proves (i). For (ii), let $\zeta\in W_{C,t}$. By definition of $W_{C,t}$, $\zeta$ is a tangential trace on $\Gamma_C$, hence
	\begin{equation*}
		\zeta\in H^{1/2}_{00}(\Gamma_C;\R^d)
		\qquad\text{and}\qquad
		\zeta\cdot \mathrm n=0 \text{ on }\Gamma_C.
	\end{equation*}
	Let $\mathrm{Ext}_0(\zeta)$ denote the zero extension of $\zeta$ from $\Gamma_C$ to $\Gamma$, and let $w\in H^1(\Omega;\R^d)$ be a lifting such that $\gamma w=\mathrm{Ext}_0(\zeta)$ on $\Gamma$. Then $\gamma w = 0$ on $\Gamma_N$ and on $\Gamma_C$ we have $\gamma w = \zeta$. Since $\zeta$ is tangential, it follows that
	\begin{equation*}
		w_{\mathrm n}
		=
		(\gamma w)\cdot \mathrm n
		=
		\zeta\cdot \mathrm n
		=
		0
		\qquad\text{on }\Gamma_C.
	\end{equation*}
	Moreover,
	\begin{equation*}
		w_t
		=
		\gamma w-w_{\mathrm n}\mathrm n
		=
		\zeta
		\qquad\text{on }\Gamma_C.
	\end{equation*}
	Finally, $\gamma w=0$ on $\Gamma_D$, so $w\in V$. This proves (ii). For (iii), let $\eta\in W_{C,\mathrm n}$. Then the vector-valued boundary function $g:=\eta\,\mathrm n$ belongs to $W_C\subset H^{1/2}_{00}(\Gamma_C;\R^d)$. Let $\mathrm{Ext}_0(g)$ denote its zero extension from $\Gamma_C$ to $\Gamma$, and let $w\in H^1(\Omega;\R^d)$ be a lifting such that $\gamma w = \mathrm{Ext}_0(g)$ on $\Gamma$. Then $\gamma 2 = 0$ on $\Gamma_N$ and on $\Gamma_C$ we have $\gamma w=\eta\,\mathrm n$. Hence
	\begin{equation*}
		w_{\mathrm n}
		=
		(\gamma w)\cdot \mathrm n
		=
		(\eta\,\mathrm n)\cdot \mathrm n
		=
		\eta
		\qquad\text{on }\Gamma_C,
	\end{equation*}
	while
	\begin{equation*}
		w_t
		=
		\gamma w-w_{\mathrm n}\mathrm n
		=
		\eta\,\mathrm n-\eta\,\mathrm n
		=
		0
		\qquad\text{on }\Gamma_C.
	\end{equation*}
	Again, $\gamma w=0$ on $\Gamma_D$, so $w\in V$. This proves (iii).
\end{proof}

\begin{proposition}[Equivalence of classical strong and hybrid formulation]
	Let $u\in H^s(\Omega;\R^d)$ with $s>3/2$. Then $u$ solves Problem \ref{pr:signorini6} if and only if it solves Problem \ref{pr:signorini4}.
\end{proposition}
\begin{proof}
	The additional regularity $u\in H^s(\Omega;\R^d)$ with $s>3/2$ is used precisely to interpret the boundary conditions in Problem \ref{pr:signorini6} in the classical sense. Indeed, by the trace theorem,
	\begin{equation*}
		\gamma u\in H^{s-1/2}(\Gamma;\R^d)\subset L^2(\Gamma;\R^d),
	\end{equation*}
	and therefore the normal and tangential components
	\begin{equation*}
		u_{\mathrm n}:=(\gamma u)\cdot \mathrm n,
		\qquad
		u_t:=\gamma u-u_{\mathrm n}\mathrm n
	\end{equation*}
	are $L^2$-functions on $\Gamma_C$. In particular, the contact conditions in Problem \ref{pr:signorini6} are genuine almost-everywhere conditions on $\Gamma_C$. Assume first that $u$ solves Problem \ref{pr:signorini6}. Since $-\diver \sigma(u) = f$ almost everywhere in $\Omega$, the Green formula yields
	\begin{equation*}
		\int_\Omega \sigma(u):\varepsilon(v)\;\dx
		=
		\int_\Omega f\cdot v\;\dx
		+
		\langle \sigma(u)\mathrm n,\gamma v\rangle_\Gamma\qquad \forall\, v\in H^1(\Omega;\R^d).
	\end{equation*}
	Because in the present regularity regime the boundary traction is a classical boundary function, the duality pairing can be written as a boundary integral. Hence
	\begin{equation*}
		a(u,v)
		=
		\int_\Omega f\cdot v\;\dx
		+
		\int_{\Gamma_N} \sigma(u)\mathrm n\cdot \gamma v\;\ds
		+
		\int_{\Gamma_C} \sigma(u)\mathrm n\cdot \gamma v\;\ds.
	\end{equation*}
	Using the Neumann condition in Problem \ref{pr:signorini6}, this becomes
	\begin{equation*}
		a(u,v)
		=
		\int_\Omega f\cdot v\;\dx
		+
		\int_{\Gamma_N} F\cdot \gamma v\;\ds
		+
		\int_{\Gamma_C} \sigma(u)\mathrm n\cdot \gamma v\;\ds.
	\end{equation*}
	Let now $v\in K$. On $\Gamma_C$, we decompose
	\begin{equation*}
		\sigma(u)\mathrm n\cdot \gamma v
		=
		\sigma_{\mathrm n}(u)\,v_{\mathrm n}
		+
		\sigma_t(u)\cdot v_t.
	\end{equation*}
	By the classical Signorini conditions \eqref{eq:strong_contact_nonpen},\eqref{eq:strong_contact_sigma_n} and \eqref{eq:strong_contact_tangential},
	\begin{equation*}
		\sigma_t(u)=0,
		\qquad
		\sigma_{\mathrm n}(u)\le 0,
		\qquad
		v_{\mathrm n}\le 0 \quad\text{a.e. on }\Gamma_C,
	\end{equation*}
	and therefore
	\begin{equation*}
		\int_{\Gamma_C} \sigma(u)\mathrm n\cdot \gamma v\;\ds
		=
		\int_{\Gamma_C} \sigma_{\mathrm n}(u)\,v_{\mathrm n}\;\ds
		\ge 0.
	\end{equation*}
	Thus,
	\begin{equation*}
		a(u,v)
		\ge
		\int_\Omega f\cdot v\;\dx
		+
		\int_{\Gamma_N} F\cdot \gamma v\;\ds
		=
		\ell(v)
		\qquad\forall\,v\in K.
	\end{equation*}
	This is exactly the hybrid inequality \eqref{eq:hybrid_ineq}. To prove the hybrid equality \eqref{eq:hybrid_eq}, we take $v=u$. Then we have
	\begin{equation*}
		\int_{\Gamma_C} \sigma(u)\mathrm n\cdot \gamma u\;\ds
		=
		\int_{\Gamma_C} \sigma_{\mathrm n}(u)\,u_{\mathrm n}\;\ds
		+
		\int_{\Gamma_C} \sigma_t(u)\cdot u_t\;\ds.
	\end{equation*}
	Again by the classical Signorini conditions \eqref{eq:strong_contact_tangential} and \eqref{eq:strong_contact_comp},
	\begin{equation*}
		\sigma_t(u)=0
		\qquad\text{and}\qquad
		\sigma_{\mathrm n}(u)\,u_{\mathrm n}=0
		\quad\text{a.e. on }\Gamma_C,
	\end{equation*}
	and thus
	\begin{equation*}
		\int_{\Gamma_C} \sigma(u)\mathrm n\cdot \gamma u\;\ds=0.
	\end{equation*}
	Hence
	\begin{equation*}
		a(u,u)
		=
		\int_\Omega f\cdot u\;\dx
		+
		\int_{\Gamma_N} F\cdot \gamma u\;\ds
		=
		\ell(u),
	\end{equation*}
	which is the hybrid equality \eqref{eq:hybrid_eq}. Therefore $u$ solves Problem \ref{pr:signorini4}. Conversely, assume that $u$ solves Problem \ref{pr:signorini4}. The bulk equation
	\begin{equation*}
		-\diver\sigma(u)=f
		\qquad\text{a.e. in }\Omega
	\end{equation*}
	is already contained in the hybrid formulation. Moreover, since $u\in K\subset V$, we also obtain
	\begin{equation*}
		u=0 \text{ on }\Gamma_D
		\qquad\text{and}\qquad
		u_{\mathrm n}\le 0 \text{ on }\Gamma_C,
	\end{equation*}
	which is exactly \eqref{eq:strong_classical_dirichlet} and \eqref{eq:strong_contact_nonpen}. It remains to prove the Neumann condition on $\Gamma_N$ and the remaining contact conditions on $\Gamma_C$. We first prove the Neumann condition. Let $g_N\in H^{1/2}(\Gamma_N;\R^d)$ and choose $w\in V$ as in Lemma~\ref{lem:locboundlift}(i), i.e.
	\begin{equation*}
		\gamma w=g_N \text{ on }\Gamma_N,
		\qquad
		\gamma w=0 \text{ on }\Gamma_C.
	\end{equation*}
	Since $w_{\mathrm n}=0$ on $\Gamma_C$, both $w$ and $-w$ belong to $K$. Applying the hybrid inequality to $w$ and to $-w$, we get
	\begin{equation*}
		\langle \sigma(u)\mathrm n,\gamma w\rangle_\Gamma
		=
		\int_{\Gamma_N} F\cdot \gamma w\;\ds.
	\end{equation*}
	Because $\gamma w=0$ on $\Gamma_C$, this reads
	\begin{equation*}
		\int_{\Gamma_N} \sigma(u)\mathrm n\cdot g_N\;\ds
		=
		\int_{\Gamma_N} F\cdot g_N\;\ds
		\qquad\forall\,g_N\in H^{1/2}(\Gamma_N;\R^d).
	\end{equation*}
	Since both $\sigma(u)\mathrm n$ and $F$ are $L^2$-functions on $\Gamma_N$, it follows that
	\begin{equation*}
		\sigma(u)\mathrm n = F
		\qquad\text{a.e. on }\Gamma_N.
	\end{equation*}
	This proves the Neumann boundary condition \eqref{eq:strong_classical_neumann}. Next, we prove the tangential contact condition \eqref{eq:strong_contact_tangential}. Let $\zeta\in W_{C,t}$ and choose $w\in V$ according to Lemma~\ref{lem:locboundlift}(ii), i.e.
	\begin{equation*}
		\gamma w=0 \text{ on }\Gamma_N,
		\qquad
		w_{\mathrm n}=0,\quad w_t=\zeta \text{ on }\Gamma_C.
	\end{equation*}
	Again, both $w$ and $-w$ belong to $K$, and therefore the hybrid inequality applied to $\pm w$ yields
	\begin{equation*}
		\langle \sigma(u)\mathrm n,\gamma w\rangle_\Gamma = 0.
	\end{equation*}
	Because $\gamma w=0$ on $\Gamma_N$ and $w_{\mathrm n}=0$ on $\Gamma_C$, we obtain
	\begin{equation*}
		0
		=
		\int_{\Gamma_C} \sigma(u)\mathrm n\cdot w_t\;\ds
		=
		\int_{\Gamma_C} \sigma_t(u)\cdot \zeta\;\ds
		\qquad\forall\,\zeta\in W_{C,t}.
	\end{equation*}
	Hence $\sigma_t(u)=0$ almost everywhere on $\Gamma_C$ which is \eqref{eq:strong_contact_tangential}. We now prove the normal sign condition \eqref{eq:strong_contact_sigma_n}. Let $\eta\in W_{C,\mathrm n}$ satisfy $\eta\le 0$ almost everywhere on $\Gamma_C$, and choose $w\in V$ as in Lemma~\ref{lem:locboundlift}(iii), i.e.
	\begin{equation*}
		\gamma w=0 \text{ on }\Gamma_N,
		\qquad
		w_t=0,\quad w_{\mathrm n}=\eta \text{ on }\Gamma_C.
	\end{equation*}
	Then $w\in K$, and the hybrid inequality gives
	\begin{equation*}
		0
		\le
		\langle \sigma(u)\mathrm n,\gamma w\rangle_\Gamma
		=
		\int_{\Gamma_C} \sigma_{\mathrm n}(u)\,\eta\;\ds
		\qquad\forall\,\eta\in W_{C,\mathrm n},\ \eta\le 0.
	\end{equation*}
	Therefore, $\sigma_{\mathrm n}(u)\le 0$ almost everywhere on $\Gamma_C$ which is exactly \eqref{eq:strong_contact_sigma_n}. Finally, we prove the complementarity condition \eqref{eq:strong_contact_comp}. The hybrid equality \eqref{eq:hybrid_eq} gives
	\begin{equation*}
		\langle \sigma(u)\mathrm n,\gamma u\rangle_\Gamma
		=
		\int_{\Gamma_N} F\cdot \gamma u\;\ds.
	\end{equation*}
	Using the Neumann condition \eqref{eq:strong_classical_neumann} already proved on $\Gamma_N$, we obtain
	\begin{equation*}
		\int_{\Gamma_C} \sigma(u)\mathrm n\cdot \gamma u\;\ds = 0.
	\end{equation*}
	Since $\sigma_t(u)=0$ on $\Gamma_C$ by \eqref{eq:strong_contact_tangential}, this reduces to
	\begin{equation*}
		\int_{\Gamma_C} \sigma_{\mathrm n}(u)\,u_{\mathrm n}\;\ds = 0.
	\end{equation*}
	Now by \eqref{eq:strong_contact_nonpen} and \eqref{eq:strong_contact_sigma_n} it is
	\begin{equation*}
		u_{\mathrm n}\le 0
		\qquad\text{and}\qquad
		\sigma_{\mathrm n}(u)\le 0
		\qquad\text{a.e. on }\Gamma_C,
	\end{equation*}
	so the product $\sigma_{\mathrm n}(u)\,u_{\mathrm n}$ is nonnegative almost everywhere on $\Gamma_C$. Since its integral is zero, it follows that
	\begin{equation*}
		\sigma_{\mathrm n}(u)\,u_{\mathrm n}=0
		\qquad\text{a.e. on }\Gamma_C.
	\end{equation*}
	Thus all conditions in Problem \ref{pr:signorini6} hold, and $u$ solves the classical strong formulation.
\end{proof}

\subsection{Mixed Formulation}

In the primal setting, Signorini's problem is posed as a variational inequality on the convex set $K\subset V$.
In the mixed approach we introduce an additional unknown $\lambda$ on the contact boundary, which represents the weak normal contact traction.
This leads to a saddlepoint structure in the variables $(u,\lambda)$, where the nonpenetration constraint is encoded by the cone condition $\lambda\in\Lambda_C$.

\subsubsection{Mixed Weak Formulation as a Saddlepoint Problem}

We keep the energy functional
\begin{equation*}
	\cJ(v)=\frac12 a(v,v)-\ell(v),\qquad v\in V
\end{equation*}
from \eqref{eq:energyfunc} and use the normal trace space $W_{C,\mathrm n}$ and the dual cone $\Lambda_C\subset W_{C,\mathrm n}'$ introduced in \eqref{eq:def_LambdaC}. For $v\in V$ and $\lambda\in W_{C,\mathrm n}'$ we define the coupling form
\begin{equation}\label{eq:def_b_mixed}
	b(v,\lambda):=\left\langle \lambda,\,(\gamma v|_{\Gamma_C})\cdot \mathrm n\right\rangle_{\Gamma_C}.
\end{equation}
We then introduce the Lagrangian on $V\times\Lambda_C$ by
\begin{equation}\label{eq:def_Lagrangian_mixed}
	\cL(v,\lambda):=\cJ(v)-b(v,\lambda).
\end{equation}

\begin{problem}[Signorini's Problem -- Mixed weak formulation]\label{pr:signorini_mixed}
Find $(u,\lambda)\in V\times \Lambda_C$ such that

\begin{equation*}
	\begin{cases}
		a(u,v)-b(v,\lambda)=\ell(v) & \forall\,v\in V,           \\[3pt]
		b(u,\mu-\lambda)\ge 0       & \forall\,\mu\in \Lambda_C.
	\end{cases}
\end{equation*}
\end{problem}

The first condition \eqref{eq:mixed_eq1} corresponds to bulk equilibrium with an additional boundary reaction term encoded by $\lambda$.
The second condition \eqref{eq:mixed_eq2} is a variational inequality on the cone $\Lambda_C$ and represents the complementarity structure of frictionless contact.

\begin{remark}[Saddlepoint property]\label{rem:saddlepoint_property}
	A pair $(u,\lambda)\in V\times\Lambda_C$ solves Problem~\ref{pr:signorini_mixed} if and only if it satisfies the saddlepoint inequality
	\begin{equation}\label{eq:saddlepoint}
		\cL(u,\mu)\le \cL(u,\lambda)\le \cL(v,\lambda)\quad \forall \,(v,\mu)\in V\times\Lambda_C.
	\end{equation}
\end{remark}


\subsubsection{Well--posedness and inf--sup condition}

We briefly explain the role of the inf--sup condition for the coupling form $b(\cdot,\cdot)$ in \eqref{eq:def_b_mixed}.
Define the operators $A:V\to V'$ and $B:V\to W_{C,\mathrm n}$ by
\begin{equation}\label{eq:def_operators_AB}
	(Au)(v):=a(u,v),\quad Bv:= (\gamma v|_{\Gamma_C})\cdot \mathrm n.
\end{equation}
Then \eqref{eq:mixed_eq1} can be written as
\begin{equation}\label{eq:first_equation_operator}
	Au-B^*\lambda=\ell \quad \text{in }V',
\end{equation}
where $B^*:W_{C,\mathrm n}'\to V'$ is the adjoint operator given by
\begin{equation}\label{eq:def_B_adjoint_mixed}
	(B^*\lambda)(v)=b(v,\lambda)\quad \forall\, v\in V.
\end{equation}

In the purely linear saddlepoint setting (without the cone constraint $\lambda\in\Lambda_C$), well--posedness of the block system is governed by the inf--sup condition
\begin{equation}\label{eq:inf_sup_mixed}
	\inf_{\lambda\in W_{C,\mathrm n}'\setminus\{0\}}
	\sup_{v\in V\setminus\{0\}}
	\frac{b(v,\lambda)}{\|v\|_{1,\Omega}\,\|\lambda\|_{W_{C,\mathrm n}'}}
	\ge \beta>0.
\end{equation}
Equivalently, \eqref{eq:inf_sup_mixed} can be written as
\begin{equation}\label{eq:inf_sup_equiv}
	\|B^*\lambda\|_{V'}\ge \beta\,\|\lambda\|_{W_{C,\mathrm n}'}\quad \forall\, \lambda\in W_{C,\mathrm n}'.
\end{equation}
This expresses stability of the coupling between $u$ and $\lambda$ and, in particular, implies uniqueness of the multiplier once $u$ is fixed.

\begin{proposition}[Uniqueness of the multiplier]\label{prop:unique_multiplier}
	Assume that \eqref{eq:inf_sup_mixed} holds.
	If $(u,\lambda_1)$ and $(u,\lambda_2)$ satisfy \eqref{eq:mixed_eq1} with the same $u\in V$, then $\lambda_1=\lambda_2$ in $W_{C,\mathrm n}'$.
\end{proposition}
\begin{proof}
	Subtracting \eqref{eq:mixed_eq1} for $\lambda_1$ and $\lambda_2$ yields
	\begin{equation}\label{eq:bdiff_zero}
		b(v,\lambda_1-\lambda_2)=0\quad \forall\, v\in V.
	\end{equation}
	Hence $(B^*(\lambda_1-\lambda_2))(v)=0$ for all $v\in V$, i.e. $B^*(\lambda_1-\lambda_2)=0$ in $V'$.
	Using \eqref{eq:inf_sup_equiv} gives $\lambda_1-\lambda_2=0$.
\end{proof}

Finally, note that well--posedness of the mixed Signorini formulation also follows from its equivalence to the primal (or hybrid) formulation established earlier.  Nevertheless, the inf--sup viewpoint is useful since it explains the saddlepoint structure and the stability of the multiplier.
