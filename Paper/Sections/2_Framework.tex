\section{Mathematical Framework}
This section introduces the basic notation used throughout the mathematical framework and the later analysis of Signorini's problem. Our goal is to fix a consistent language for bulk quantities, boundary quantities, and tensor-valued fields before introducing weak derivatives and Sobolev spaces. Unless stated otherwise, let $\Omega\subset\R^d,\,d\ge 1$, be a nonempty open set. Whenever trace operators or boundary spaces are used, we additionally assume that $\Omega$ is a bounded Lipschitz domain (see e.g.\ \cite[Section 2.3]{book}) and write $\Gamma:=\partial\Omega$.

\subsection{Scalars, Vectors and Matrices}\label{sec:SVM}
The purpose of this subsection is mainly to introduce the notation used throughout this report. For $m,n\ge 1$, we write $\R^{m\times n}$ for the space of real $m\times n$ matrices and for vectors $u,v\in\R^m$, we denote by
\begin{equation*}
	u\cdot v := \sum_{j=1}^m u_j\,v_j,
	\qquad
	|u| := \sqrt{u\cdot u}
\end{equation*}
the Euclidean inner product and norm. For matrices $A,B\in\R^{m\times n}$, we use the Frobenius product
\begin{equation*}
	A:B := \sum_{j=1}^m\sum_{k=1}^n A_{jk}B_{jk},
	\qquad
	|A| := \sqrt{A:A}.
\end{equation*}
The space $\mathrm{Sym}^2(\R^d)$ of symmetric second order tensors is
\begin{equation*}
	\mathrm{Sym}^2(\R^d):= \left\{ s\in\R^d\otimes\R^d\;\middle|\; s^\top = s\right\}.
\end{equation*}
We will interpret these tensors as symmetric matrices in $\R^{d\times d}$ in the canonical basis of $\R^d$. Indeed,
\begin{equation*}
	\mathrm{Sym}^2(\R^d)\cong \left\{S\in\R^{d\times d}\;\middle|\; S^\top = S\right\}.
\end{equation*}
Furthermore, if $\cA:\Omega\to\cL(\mathrm{Sym}^2(\R^d);\mathrm{Sym}^2(\R^d))$ is a fourth order tensor and $s\in\mathrm{Sym}^2(\R^d)$, we use the {\em double contraction}
\begin{equation*}
	(\cA(x):s)_{ij}:=\sum_{k = 1}^d \sum_{l = 1}^d \cA_{ijkl}(x)s_{kl},
	\qquad
	s:\cA(x):s:=\sum_{i = 1}^d \sum_{j = 1}^d \sum_{k = 1}^d \sum_{l = 1}^d\cA_{ijkl}(x)s_{ij}s_{kl}.
\end{equation*}
\subsection{\texorpdfstring{$L^p$}{Lp}-Spaces}
Even assuming the reader is already familiar with the concept of $L^p$-spaces, the definitions and notations are listed below. We begin with the $L^p$-spaces on $\Omega$ as spaces of equivalence classes of $p$-integrable functions whose difference vanishes almost everywhere on $\Omega$. Although this is the most correct way to define $L^p$-spaces, we will subsequently always assume an actual function by choosing an arbitrary representative of a class. The assumed measure on $\Omega$ here is the Lebesgue measure in $\R^d$. We omit the case $p = \infty$.

\begin{definition}[$L^p$-Spaces]\label{sec:Lpspaces}
	Let $m\ge 1$ and $1\le p<\infty$. We define
	\begin{equation*}
		L^p(\Omega;\R^m)
		:=
		\left\{
		u:\Omega\to\R^m\text{\em\enspace measurable }\;\middle|\; \int_\Omega |u|^p\;\dx<\infty
		\right\}\Big/\sim,
	\end{equation*}
	where $u\sim v$ if and only if $u=v$ almost everywhere in $\Omega$. For some equivalence class $[u]\in L^p(\Omega;\R^m)$ we take an arbitrary representative and write $u = [u]$. If $m = 1$, we abbreviate $L^p(\Omega):= L^p(\Omega;\R^1)$.
	Moreover, we equip $L^p(\Omega;\R^m)$ with the norm
	\begin{equation*}
		\|u\|_{0,p,\Omega} := \left(\int_\Omega |u|^p\;\dx\right)^{1/p},\qquad u\in L^p(\Omega;\R^m).
	\end{equation*}
\end{definition}

The notation $\|\cdot\|_{0,p,\Omega}$ will be more clear after Sobolev spaces are introduced. Next, it is necessary to define these spaces for boundary parts of $\Omega$ as well. This is done analogously, with the difference that we now consider a surface or boundary integral and accordingly also the Lebesgue measure on $\R^{d-1}$.

\begin{definition}[$L^p$-Spaces on Boundary Parts]
	Assume that $\Omega$ is a bounded Lipschitz domain and let $\Gamma_0\subset\Gamma$ be measurable with respect to the surface measure on $\Gamma$. For $m\ge 1$ and $1\le p<\infty$, we define
	\begin{equation*}
		L^p(\Gamma_0;\R^m)
		:=
		\left\{
		g:\Gamma_0\to\R^m \text{\em\enspace  measurable }
		\;\middle|\;
		\int_{\Gamma_0}|g|^p\;\ds<\infty
		\right\}\Big/\sim,
	\end{equation*}
	where $g\sim h$ if and only if $g = h$ almost everywhere on $\Gamma_0$. For some equivalence class $[g]\in L^p(\Gamma_0;\R^m)$ we take an arbitrary representative and write $g = [g]$. If $m = 1$, we abbreviate $L^p(\Gamma_0):= L^p(\Gamma_0;\R^1)$. Moreover, we equip $L^p(\Gamma_0;\R^m)$ with the norm
	\begin{equation*}
		\|g\|_{0,p,\Gamma_0}:= \int_{\Gamma_0} |g|^p\;\dx,\qquad g\in L^p(\Gamma_0;\R^m).
	\end{equation*}
\end{definition}

Moreover, we also need an $L^p$-notation for matrix-valued fields $A:\Omega\to\mathrm{Sym}^2(\R^d)$.

\begin{definition}[$L^p$-Spaces for Matrix-Valued Fields]
	Let $1\le p<\infty$. We define
	\begin{equation*}
		L^p(\Omega;\mathrm{Sym}^2(\R^d)):=\left\{ A:\Omega\to\mathrm{Sym}^2(\R^d)\text{\em\enspace measurable}\;\middle|\; \int_\Omega |A|^p\;\dx<\infty\right\}\Big/\sim,
	\end{equation*}
	where $A\sim B$ if and only if $A = B$ almost everywhere on $\Omega$. For some equivalence class $[A]\in L^p(\Omega;\mathrm{Sym}^2(\R^d))$, we take an arbitrary representative and write $A=[A]$. Moreover, we equip $L^p(\Omega;\mathrm{Sym}^2(\R^d))$ with the norm
	\begin{equation*}
		\|A\|_{0,p,\Omega}:=\int_\Omega |A|^p\;\dx,\qquad A\in L^p(\Omega;\mathrm{Sym}^2(\R^d)).
	\end{equation*}
\end{definition}

Finally, one particular $L^p$-space is of special interest because it is the only one that, together with its classical inner product, defines a Hilbert space. This is the case $p=2$. We record this fact in the following proposition and simultaneously introduce a meaningful notation for its inner product and its norm.

\begin{proposition}[$L^2$ is a Hilbert Space]\label{pr:L2hilbert}
	For every $m\ge 1$, the space $L^2(\Omega;\R^m)$ is a Hilbert space with inner product
	\begin{equation*}
		(u,\,v)_{0,\Omega}
		:=
		\int_\Omega u\cdot v\;\dx
		\qquad u,v\in L^2(\Omega;\R^m),
	\end{equation*}
	and the induced norm is $\|u\|_{0,\Omega}:= \|u\|_{0,2,\Omega} = (u,u)_{0,\Omega}^{1/2}$. If $\Omega$ is a bounded Lipschitz domain and $\Gamma_0\subset\Gamma$ is measurable, $L^2(\Gamma_0,\R^m)$ is a Hilbert space and for $g,h\in L^2(\Gamma_0,\R^m)$ we analogously write
	\begin{equation*}
		(g,\,h)_{0,\Gamma_0}
		:=
		\int_{\Gamma_0} g\cdot h\;\ds,
		\qquad
		\|g\|_{0,\Gamma_0}
		:=
		\|g\|_{0,2,\Gamma_0} = (g,\,g)_{0,\Gamma_0}^{1/2}.
	\end{equation*}
	Moreover, $L^2(\Omega;\mathrm{Sym}^2(\R^d))$ together with the inner product
	\begin{equation*}
		(A,\,B)_{0,\Omega}:= \int_\Omega A:B\;\dx,\qquad A,B\in L^2(\Omega;\mathrm{Sym}^2(\R^d))
	\end{equation*}
	is a Hilbert space and the induced norm is
	\begin{equation*}
		\|A\|_{0,\Omega}:= \|A\|_{0,2,\Omega} = (A,\,A)_{0,\Omega}^{1/2}.
	\end{equation*}
\end{proposition}

For later purpose, we also need to consider the subspace of locally integrable functions on $\Omega$.

\begin{definition}[Locally Integrable Functions]
	The space of {\tt locally integrable} functions on $\Omega$ is defined as
	\begin{equation*}
		L_\loc^1 (\Omega;\R^m):= \left\{u:\Omega\to\R^m\text{\em\enspace measurable}\;\middle|\;u|_K\in L^1(K;\R^m)\;\forall K\subset\Omega,\,K\textit{\em\enspace compact}\right\}.
	\end{equation*}
	For $m=1$, we denote $L_\loc^1(\Omega):=L_\loc^1(\Omega;\R^1)$ and the space $L^1_\loc(\Omega;\mathrm{Sym}^2(\R^d))$ is defined analogously.
\end{definition}

\subsection{Test Functions \& Distributional Derivatives}

Before introducing Sobolev spaces, we first formalize a weak notion of differentiation. The reason is that the functions arising in variational formulations of boundary value problems are typically not smooth enough to admit classical derivatives in the pointwise sense. Nevertheless, one still wants to differentiate such functions in a meaningful way, at least under an integral sign. The key observation is that in integration by parts, derivatives can be shifted from an unknown function to a smooth test function with compact support. This suggests defining derivatives indirectly through their action on test functions. The resulting objects are distributions, and distributional derivatives extend classical derivatives whenever the latter exist. For the sake of completeness, we begin with the definitions of the classic notions of differentiation, which should already be familiar to the reader.
\begin{definition}[Partial Derivative, Gradient, Jacobian and Divergence]\label{def:classicaldiffs}
	Let $u:\Omega\to\R$ and $F:\Omega\to\R^m$ and $A:\Omega\to \R^{m\times d}$ be sufficiently regular maps.
	\begin{itemize}
		\item[i)] The {\tt partial derivative} of $u$ with respect to $\alpha\in\N_0^d$ is
		      \begin{equation*}
			      \Diff^\alpha u := \frac{\partial^{|\alpha|}u}{\partial x_1^{\alpha_1}\cdots\partial x_d^{\alpha_d}}:\Omega\to\R.
		      \end{equation*}

		\item[ii)] The {\tt gradient} of $u$ is
		      \begin{equation*}
			      \nabla u := \left(\frac{\partial u}{\partial x_1},\ldots,\frac{\partial u}{\partial x_d}\right)^\top:\Omega\to\R^d.
		      \end{equation*}

		\item[iii)] The {\tt Jacobian} of $F=(F_1,\ldots,F_m)^\top$ is
		      \begin{equation*}
			      \Diff F := \left(\frac{\partial F_j}{\partial x_k}\right)_{1\le j\le m}^{1\le k\le d}:\Omega\to\R^{m\times d}.
		      \end{equation*}

		\item[iv)] For $m = d$, the {\tt divergence} of $F=(F_1,\ldots,F_d)^\top$ is
		      \begin{equation*}
			      \diver F := \sum_{j=1}^d \frac{\partial F_j}{\partial x_j}:\Omega\to\R.
		      \end{equation*}

		\item[v)] The {\tt divergence} of $A = (A_{jk})_{1\le j\le m}^{1\le k\le d}$ is
		      \begin{equation*}
			      \diver A
			      :=
			      \left(\sum_{k=1}^d \frac{\partial A_{jk}}{\partial x_k}\right)_{1\le j\le m}
			      :\Omega\to\R^m.
		      \end{equation*}
	\end{itemize}
\end{definition}

We collect continuous functions with continuous partial derivatives of the same order in the following well-known function spaces.

\begin{definition}[Spaces of Continuously Differentiable Functions]
	We denote by $\cC^0(\Omega;\R^m)$ the space of continuous functions $u:\Omega\to\R^m$ and by
	\begin{equation*}
		\cC^k(\Omega;\R^m):= \left\{u\in \cC^0(\Omega;\R^m)\;\middle|\; |\alpha|\le k\;\Rightarrow\;\Diff^\alpha u\in\cC^0(\Omega;\R^m)\right\},\qquad k\ge 0,
	\end{equation*}
	the space of such functions that are continuously differentiable up to order $k\ge 0$. Moreover,
	\begin{equation*}
		\cC^\infty(\Omega;\R^m) := \bigcap_{k\ge 0}\cC^k(\Omega;\R^m)
	\end{equation*}
	denotes the space of infinitely continuously differentiable functions. For $m=1$, we abbreviate
	\begin{equation*}
		\cC^k(\Omega):= \cC^k(\Omega;\R^1),\qquad \forall\,0\le k\le \infty
	\end{equation*}
	and note that
	\begin{equation*}
		\cC^k(\Omega;\R^m) = (\cC^k(\Omega))^m,\quad\forall\,0\le k\le\infty,\, m\ge 1.
	\end{equation*}
	Finally, for $0\le k\le\infty$, we define
	\begin{equation*}
		\cC^k(\Omega;\mathrm{Sym}^2(\R^d)):= \left\{ A=(A_{ij})_{i,j=1}^d:\Omega\to\mathrm{Sym}^2(\R^d)\;\middle|\; A_{ij}\in \cC^k(\Omega)\;\forall 1\le i,j\le d\right\}.
	\end{equation*}
\end{definition}

A particular subspace of the $k=\infty$ case is of special interest to us and the concept of distributional derivatives. This is the space of infinitely often continuously differentiable functions with compact support, also called {\em test functions}.

\begin{definition}[Test Function]\label{def:testfunctions}
	The space $\cD(\Omega;\R^m)$ contains all infinitely often continuously differentiable functions with compact support in $\Omega$, i.e.\
	\begin{equation*}
		\cD(\Omega;\R^m) = \left\{\varphi\in\cC^\infty(\Omega;\R^m)\;\middle|\;\supp(\varphi)\subset\Omega\text{\em\enspace compact}\right\}
	\end{equation*}
	For $m=1$, we abbreviate $\cD(\Omega):= \cD(\Omega;\R^1)$ and note that
	\begin{equation*}
		\cD(\Omega;\R^m) = (\cD(\Omega))^m,\qquad \forall\,m\ge 1.
	\end{equation*}
	The space $\cD(\Omega;\mathrm{Sym}^2(\R^d))$ is defined analogously and functions of these spaces are called {\tt test functions}.
\end{definition}
The space of test functions allows us to encode differentiation through integration by parts.
This leads to distributions, which extend classical derivatives to nonsmooth functions and, more generally, to generalized functions.
\begin{definition}[Distributions]
	The space of scalar-valued {\tt distributions} on $\Omega$ is the topological dual space
	\begin{equation*}
		\cD^\prime(\Omega):=\left(\cD(\Omega)\right)^\prime.
	\end{equation*}
	Thus, a scalar-valued distribution is a linear continuous functional $T:\cD(\Omega)\to\R$.
	For $T\in\cD^\prime(\Omega)$ and $\varphi\in\cD(\Omega)$, we denote the duality pairing by $\langle T,\,\varphi\rangle$. For $m\ge 1$, we define the space of vector-valued distributions componentwise by
	\begin{equation*}
		\cD^\prime(\Omega;\R^m):=\left(\cD^\prime(\Omega)\right)^m.
	\end{equation*}
	For $T=(T_1,\ldots,T_m)\in\cD^\prime(\Omega;\R^m)$ and $\varphi=(\varphi_1,\ldots,\varphi_m)\in\cD(\Omega;\R^m)$, we use the duality pairing
	\begin{equation*}
		\langle T,\,\varphi\rangle := \sum_{j=1}^m \langle T_j,\,\varphi_j\rangle.
	\end{equation*}

	Analogously, the space $\cD^\prime(\Omega;\mathrm{Sym}^2(\R^d))$ is defined componentwise, and for
	\begin{equation*}
		T=(T_{jk})_{j,k=1}^d\in\cD^\prime(\Omega;\mathrm{Sym}^2(\R^d)),
		\qquad
		\varphi=(\varphi_{jk})_{j,k=1}^d\in\cD(\Omega;\mathrm{Sym}^2(\R^d)),
	\end{equation*}
	we use the duality pairing
	\begin{equation*}
		\langle T,\,\varphi\rangle := \sum_{j=1}^d\sum_{k=1}^{d}\langle T_{jk},\,\varphi_{jk}\rangle.
	\end{equation*}
\end{definition}

\begin{definition}[Distributional Derivative]
	Let $T\in\cD^\prime(\Omega)$ and $\alpha\in\N_0^d$. The {\tt distributional derivative} $\Diff^\alpha T\in\cD^\prime(\Omega)$ is defined by
	\begin{equation*}
		\langle \Diff^\alpha T,\,\varphi\rangle
		:=
		(-1)^{|\alpha|}\,\langle T,\,\Diff^\alpha\varphi\rangle,
		\qquad \forall\,\varphi\in\cD(\Omega).
	\end{equation*}
	Distributional derivatives of vector-valued and matrix-valued distributions are defined componentwise.
	In particular, for $T=(T_1,\ldots,T_m)\in\cD^\prime(\Omega;\R^m)$, we define
	\begin{equation*}
		\Diff^\alpha T := (\Diff^\alpha T_1,\ldots,\Diff^\alpha T_m)\in\cD^\prime(\Omega;\R^m).
	\end{equation*}
	Analogously, for $T=(T_{jk})_{j,k=1}^d\in\cD^\prime(\Omega;\mathrm{Sym}^2(\R^d))$, we define
	\begin{equation*}
		\Diff^\alpha T := (\Diff^\alpha T_{jk})_{j,k=1}^d\in\cD^\prime(\Omega;\mathrm{Sym}^2(\R^d)).
	\end{equation*}
\end{definition}

Using the componentwise definition of distributional derivatives, we can now extend the classical differential operators introduced above to distributions. For $1\le k\le d$, let $e_k\in\N_0^d$ denote the $k$-th unit multi-index, i.e.
\begin{equation*}
	e_k := (0,\ldots,0,1,0,\ldots,0)^\top,
\end{equation*}
where the entry $1$ is in the $k$-th position. Then $\Diff^{e_k}$ in the distributional sense corresponds to the first-order derivative in the $x_k$-direction.

\begin{definition}[Distributional Gradient, Jacobian and Divergence]
	Let $T\in\cD^\prime(\Omega)$ and $U\in\cD^\prime(\Omega;\R^m)$ and $\Sigma\in\cD^\prime(\Omega;\mathrm{Sym}^2(\R^d))$.
	\begin{itemize}
		\item[i)] The {\tt distributional gradient} of $T$ is
		      \begin{equation*}
			      \nabla T:= \left(\Diff^{e_1}T,\ldots,\Diff^{e_d}T\right)^\top \in \cD^\prime(\Omega;\R^d).
		      \end{equation*}
		\item[ii)] The {\tt distributional Jacobian} of $U = (U_1,\ldots, U_m)^\top$ is
		      \begin{equation*}
			      \Diff U := \left(\Diff^{e_k}  U_j\right)_{1\le j\le m}^{1\le k\le d}\in\cD^\prime(\Omega;\R^{m\times d}).
		      \end{equation*}
		\item[iii)] For $d = m$, the {\tt distributional divergence} of $U =(U_1,\ldots, U_d)^\top$ is
		      \begin{equation*}
			      \diver U:= \sum_{k = 1}^d\Diff^{e_k} U_k\in\cD^\prime(\Omega).
		      \end{equation*}
		\item[iv)] The {\tt distributional divergence} of $\Sigma = (\Sigma_{jk})_{j,k = 1}^d$ is
		      \begin{equation*}
			      \diver \Sigma:= \left(\sum_{k=1}^d\Diff^{e_k}\Sigma_{jk}\right)_{1\le j\le d}\in \cD^\prime(\Omega;\R^d).
		      \end{equation*}
	\end{itemize}
\end{definition}

At this stage, the above definitions are purely distributional and do not require any pointwise differentiability.
In a later subsection, we will identify locally integrable functions with regular distributions and show that weak derivatives are precisely those distributional derivatives that are again induced by functions.

\subsection{Regular Distributions \& Weak Derivatives}

The distributional framework introduced above becomes particularly useful once distributions are induced by locally integrable functions. These so-called regular distributions provide the link between generalized differentiation and the weak derivatives used later in Sobolev spaces and variational formulations. From now on let $X$ denote one of the spaces $\R^m,\,m\ge 1$ or $\mathrm{Sym}^2(\R^d)\,d\ge 2$ to save some writing work and recall that $(\cdot,\,\cdot)_{0,\Omega}$ denotes the inner product on each of the spaces $L^2(\Omega;X)$ as introduced in Proposition \ref{pr:L2hilbert}. On $X$ we use the notation of $(\cdot,\,\cdot)_X$ as the inner product, i.e.\ Euclidean or Frobenius, respectively, as stated in Section~\ref{sec:SVM}.

\begin{definition}[Regular and Singular Distributions]
	A {\tt regular distribution} is a distribution $T_f\in\cD^\prime(\Omega;X)$ induced by a locally integrable function $f\in L^1_{\loc}(\Omega;X)$ by
	\begin{equation*}
		\langle T_f,\,\varphi\rangle := \int_\Omega (f,\,\varphi)_X\;\dx ,\qquad \forall\,\varphi\in\cD(\Omega;X).
	\end{equation*}
	In particular, $L^1_\loc(\Omega;X)$ is embedded into $\cD^\prime(\Omega;X)$ via $f\mapsto T_f$ and its image is the space of regular distributions. Distributions that are not regular are called {\tt singular}.
\end{definition}

The concept of distributional derivatives applied to regular distributions is exactly the concept of weak derivatives.

\begin{definition}[Weak Derivative]
	Let $f\in L^1_{\loc}(\Omega;X)$ and $\alpha\in\N_0^d$.
	A locally integrable function $g\in L^1_{\loc}(\Omega;X)$ is called the {\tt weak derivative} of $f$ of order $\alpha$ if
	\begin{equation*}
		\int_\Omega (g,\,\varphi)_X\;\dx =
		(-1)^{|\alpha|}\int_\Omega(f,\,\Diff^\alpha\varphi)_X \;\dx\qquad \forall\,\varphi\in\cD(\Omega;X).
	\end{equation*}
	In this case, we write $g=\Diff^\alpha f$.
\end{definition}

\begin{proposition}
	Let $f,g\in L^1_\loc(\Omega;X)$. Then $g$ is the weak derivative $\Diff^\alpha f$ of $f$ if and only if
	\begin{equation*}
		\langle \Diff^\alpha T_f,\,\varphi\rangle = \langle T_g,\,\varphi\rangle\qquad\forall\,\varphi\in\cD(\Omega;X),
	\end{equation*}
	or in other words $\Diff^\alpha T_f = T_g$ in $\cD^\prime(\Omega;X)$.
\end{proposition}
\begin{proof}
	Assume first that $g=\Diff^\alpha f$ in the weak sense. Then for every $\varphi\in\cD(\Omega;X)$, by definition of the distributional derivative and the weak derivative,
	\begin{equation*}
		\langle \Diff^\alpha T_f,\,\varphi\rangle = (-1)^{|\alpha|}\,\langle T_f,\,\Diff^\alpha\varphi\rangle  = (-1)^{|\alpha|}\int_\Omega (f,\,\Diff^\alpha\varphi)_X\;\dx = \int_\Omega (g,\,\varphi)_X\;\dx  = \langle T_g,\,\varphi\rangle.
	\end{equation*}
	Hence $\Diff^\alpha T_f = T_g$ in $\cD^\prime(\Omega;X)$. Conversely, assume that $\Diff^\alpha T_f = T_g$ in $\cD^\prime(\Omega;X)$. Then for every test function $\varphi\in\cD(\Omega;X)$,
	\begin{equation*}
		\int_\Omega (g,\,\varphi)_X\;\dx = \langle T_g,\,\varphi\rangle  =\langle \Diff^\alpha T_f,\,\varphi\rangle  =(-1)^{|\alpha|}\,\langle T_f,\,\Diff^\alpha\varphi\rangle  = (-1)^{|\alpha|}\int_\Omega (f,\,\Diff^\alpha\varphi)_X\;\dx.
	\end{equation*}
	Thus, $g$ is the weak derivative $\Diff^\alpha f$ of $f$.
\end{proof}

\begin{proposition}[Differential Operators on Regular Distributions]\label{pr:ops_regular_distributions}
	Let $u\in L^1_\loc(\Omega)$ and $F \in L^1_\loc(\Omega;\R^d)$ and $A \in L^1_\loc(\Omega;\mathrm{Sym}^2(\R^d))$.

	\begin{itemize}
		\item[i)] For every $\varphi\in\cD(\Omega;\R^d)$,
		      \begin{equation*}
			      \langle \nabla T_u,\,\varphi\rangle = -\langle T_u,\,\diver\varphi\rangle = -\int_\Omega u\,\diver \varphi\;\dx,
		      \end{equation*}
		      where $\diver\varphi$ is understood in the classical sense. If the weak gradient $\nabla u$ exists in $L^1_\loc(\Omega;\R^d)$, then
		      \begin{equation*}
			      \nabla T_u = T_{\nabla u}\qquad\text{in }\cD^\prime(\Omega;\R^d).
		      \end{equation*}

		\item[ii)] For every $\varphi\in\cD(\Omega)$,
		      \begin{equation*}
			      \langle \diver T_F,\,\varphi\rangle
			      = -\langle T_F,\,\nabla\varphi\rangle
			      = -\int_\Omega F\cdot\nabla\varphi\;\dx,
		      \end{equation*}
		      where $\nabla \varphi$ is understood in the classical sense. If, in addition, the weak divergence $\diver F$ exists in $L^1_\loc(\Omega)$, then
		      \begin{equation*}
			      \diver T_F = T_{\diver F}\qquad\text{in }\cD^\prime(\Omega).
		      \end{equation*}

		\item[iii)] For every $\varphi\in\cD(\Omega;\R^d)$,
		      \begin{equation*}
			      \langle \diver T_A,\,\varphi\rangle
			      = -\int_\Omega A:\Diff \varphi\;\dx,
		      \end{equation*}
		      where $\Diff \varphi$ is the Jacobian in the classical sense. If the weak divergence $\diver A$ exists in $L^1_\loc(\Omega;\R^d)$, then
		      \begin{equation*}
			      \diver T_A = T_{\diver A}\qquad\text{in }\cD^\prime(\Omega;\R^d).
		      \end{equation*}
	\end{itemize}
\end{proposition}

\begin{proof}
	For i), let $\varphi=(\varphi_1,\ldots,\varphi_d)^\top\in\cD(\Omega;\R^d)$. By the componentwise definition of the pairing and of the distributional gradient, we obtain
	\begin{equation*}
		\langle \nabla T_u,\,\varphi\rangle
		=
		\sum_{k=1}^d \langle \Diff^{e_k}T_u,\,\varphi_k\rangle.
	\end{equation*}
	Using the definition of the distributional derivative, this yields
	\begin{equation*}
		\langle \nabla T_u,\,\varphi\rangle
		=
		-\sum_{k=1}^d \langle T_u,\,\Diff^{e_k}\varphi_k\rangle
		=
		-\left\langle T_u,\,\sum_{k=1}^d \Diff^{e_k}\varphi_k\right\rangle
		=
		-\langle T_u,\,\diver\varphi\rangle.
	\end{equation*}
	Since $T_u$ is the regular distribution induced by $u$, we further get
	\begin{equation*}
		\langle \nabla T_u,\,\varphi\rangle
		=
		-\int_\Omega u\,\diver\varphi\;\dx.
	\end{equation*}
	Now assume that the weak gradient $\nabla u$ exists in $L^1_\loc(\Omega;\R^d)$. Then each weak derivative $\Diff^{e_k}u$ exists in $L^1_\loc(\Omega)$, and by the definition of weak derivative we have
	\begin{equation*}
		\langle \Diff^{e_k}T_u,\,\psi\rangle
		=
		\langle T_{\Diff^{e_k}u},\,\psi\rangle
		\qquad \forall\,\psi\in\cD(\Omega),\quad 1\le k\le d.
	\end{equation*}
	Hence
	\begin{equation*}
		\nabla T_u
		=
		\left(\Diff^{e_1}T_u,\ldots,\Diff^{e_d}T_u\right)^\top
		=
		\left(T_{\Diff^{e_1}u},\ldots,T_{\Diff^{e_d}u}\right)^\top
		=
		T_{\nabla u}
		\qquad\text{in }\cD^\prime(\Omega;\R^d).
	\end{equation*}
	For ii), let $\varphi\in\cD(\Omega)$. By the definition of the distributional divergence and the distributional derivative,
	\begin{equation*}
		\langle \diver T_F,\,\varphi\rangle
		=
		\sum_{k=1}^d \langle \Diff^{e_k}(T_F)_k,\,\varphi\rangle
		=
		-\sum_{k=1}^d \langle (T_F)_k,\,\Diff^{e_k}\varphi\rangle.
	\end{equation*}
	By the componentwise pairing, this is
	\begin{equation*}
		\langle \diver T_F,\,\varphi\rangle
		=
		-\langle T_F,\,\nabla\varphi\rangle.
	\end{equation*}
	Since $T_F$ is the regular distribution induced by $F$, we obtain
	\begin{equation*}
		\langle \diver T_F,\,\varphi\rangle
		=
		-\int_\Omega F\cdot\nabla\varphi\;\dx.
	\end{equation*}
	If, in addition, the weak divergence $\diver F$ exists in $L^1_\loc(\Omega)$, then by the definition of weak divergence
	\begin{equation*}
		\int_\Omega (\diver F)\,\varphi\;\dx
		=
		-\int_\Omega F\cdot\nabla\varphi\;\dx
		\qquad \forall\,\varphi\in\cD(\Omega).
	\end{equation*}
	Together with the previous identity, this implies
	\begin{equation*}
		\langle \diver T_F,\,\varphi\rangle
		=
		\langle T_{\diver F},\,\varphi\rangle
		\qquad \forall\,\varphi\in\cD(\Omega),
	\end{equation*}
	and therefore
	\begin{equation*}
		\diver T_F = T_{\diver F}\qquad\text{in }\cD^\prime(\Omega).
	\end{equation*}
	For iii), let $\varphi=(\varphi_1,\ldots,\varphi_d)^\top\in\cD(\Omega;\R^d)$. By the componentwise definition of the pairing and of the distributional divergence of matrix-valued distributions,
	\begin{equation*}
		\langle \diver T_A,\,\varphi\rangle
		=
		\sum_{j=1}^d \left\langle \sum_{k=1}^d \Diff^{e_k}(T_A)_{jk},\,\varphi_j\right\rangle.
	\end{equation*}
	Using the definition of the distributional derivative, we get
	\begin{equation*}
		\langle \diver T_A,\,\varphi\rangle
		=
		-\sum_{j=1}^d\sum_{k=1}^d \langle (T_A)_{jk},\,\Diff^{e_k}\varphi_j\rangle.
	\end{equation*}
	Since $T_A$ is induced by $A$, this becomes
	\begin{equation*}
		\langle \diver T_A,\,\varphi\rangle
		=
		-\sum_{j=1}^d\sum_{k=1}^d \int_\Omega A_{jk}\,\Diff^{e_k}\varphi_j\;\dx
		=
		-\int_\Omega A:\Diff\varphi\;\dx.
	\end{equation*}
	If the weak divergence $\diver A$ exists in $L^1_\loc(\Omega;\R^d)$, then by the componentwise weak definition we have
	\begin{equation*}
		\int_\Omega (\diver A,\varphi)\;\dx
		=
		-\int_\Omega A:\Diff\varphi\;\dx
		\qquad \forall\,\varphi\in\cD(\Omega;\R^d).
	\end{equation*}
	Hence
	\begin{equation*}
		\langle \diver T_A,\,\varphi\rangle
		=
		\langle T_{\diver A},\,\varphi\rangle
		\qquad \forall\,\varphi\in\cD(\Omega;\R^d),
	\end{equation*}
	which means
	\begin{equation*}
		\diver T_A = T_{\diver A}\qquad\text{in }\cD^\prime(\Omega;\R^d).
	\end{equation*}
\end{proof}


\subsection{Sobolev Spaces}

Before defining Sobolev spaces, we connect the previously introduced distributional framework with the $L^p$-spaces from Section~\ref{sec:Lpspaces}. Since Sobolev spaces are defined by requiring distributional derivatives to belong again to some $L^p$-space, it is essential that $L^p$-functions can be interpreted as regular distributions. The following proposition shows that every $L^p$-function is locally integrable and therefore canonically induces a distribution.

\begin{lemma}[Hölder's and Cauchy-Schwarz Inequality]
	Let $1<p<\infty$ and let $p^\prime=p/(p-1)$ be the conjugate exponent. Then for all
	$f\in L^p(\Omega;X)$ and $g\in L^{p^\prime}(\Omega;X)$ Hölder's inequality
	\begin{equation*}
		\left|\int_\Omega (f,\, g)_X\;\dx\right|
		\le
		\|f\|_{0,p,\Omega}\,\|g\|_{0,p^\prime,\Omega}
	\end{equation*}
	holds. The special case $p = p^\prime = 2$ yields the Cauchy-Schwarz inequality
	\begin{equation*}
		(f,\,g)_{0,\Omega}\le \|f\|_{0,\Omega}\|g\|_{0,\Omega}\qquad\forall\,f,g\in L^2(\Omega;X).
	\end{equation*}
	\eop
\end{lemma}

\begin{proposition}[$L^p$-Functions are Locally Integrable]\label{pr:Lp_locally_integrable}
	It holds that $L^p(\Omega;X)\subset L^1_\loc(\Omega;X)$ for all $1\le p<\infty$. In particular, every function$f\in L^p(\Omega;X)$ induces a regular distribution $T_f\in\cD^\prime(\Omega;X)$ and admits distributional derivatives of arbitrary order.
\end{proposition}

\begin{proof}
	Let $f\in L^p(\Omega;X)$ and let $K\subset\Omega$ be compact. The case $p=1$ is trivial. Let now $p>1$ and let $p^\prime=p/(p-1)$ be the conjugate exponent. Since $K$ has finite Lebesgue measure, Hölder's inequality yields
	\begin{equation*}
		\int_K |f|\;\dx
		\le
		\|f\|_{0,p,\Omega}\,\|1\|_{0,p^\prime,\Omega}
		=
		\|f\|_{0,p,K}\,|K|^{1-1/p}
		\le
		\|f\|_{0,p,\Omega}\,|K|^{1-1/p}
		<\infty.
	\end{equation*}
	Hence $f|_K\in L^1(K;X)$. Since $K\subset\Omega$ was arbitrary compact, we conclude that $f\in L^1_\loc(\Omega;X)$. The remaining statement follows from the embedding $L^1_\loc(\Omega;X)\hookrightarrow\cD^\prime(\Omega;X)$ via $f\mapsto T_f$ and the fact that $\Diff^\alpha T_f$ is defined for every $\alpha\in\N_0^d$.
\end{proof}

By Proposition~\ref{pr:Lp_locally_integrable}, every $f\in L^p(\Omega;X)$ can be identified with the regular distribution $T_f\in\cD^\prime(\Omega;X)$ given by
\begin{equation*}
	\langle T_f,\,\varphi\rangle = \int_\Omega (f,\,\varphi)_X\;\dx,
	\qquad \forall\,\varphi\in\cD(\Omega;X).
\end{equation*}
In the following, we will use this identification implicitly whenever no confusion can arise. In the Hilbert case $p=2$, this pairing is consistent with the $L^2$-inner product introduced in Proposition~\ref{pr:L2hilbert}. Indeed, since $\varphi\in\cD(\Omega;X)$ has compact support, we have $\varphi\in L^2(\Omega;X)$, and therefore
\begin{equation*}
	\langle T_f,\,\varphi\rangle
	=
	\int_\Omega (f,\,\varphi)_X\;\dx
	=
	(f,\,\varphi)_{0,\Omega},
	\qquad f\in L^2(\Omega;X),\ \varphi\in\cD(\Omega;X).
\end{equation*}
Thus, on test functions, the duality pairing of the regular distribution $T_f$ coincides with the $L^2$-inner product. In the Sobolev setting, we therefore use the terms distributional derivative and weak derivative interchangeably whenever the derivative is represented by an $L^p$-function. We continue with the definition of Sobolev spaces of integer order. For now, we only need the case $X = \R^m,\,m\ge 1$

\begin{definition}[Sobolev Spaces of Integer Order]
	Let $m\ge 1,\, k\in\N_0$ and $1\le p<\infty$.
	The {\tt Sobolev space} $W^{k,p}(\Omega;\R^m)$ is defined by
	\begin{equation*}
		W^{k,p}(\Omega;\R^m)
		:=\left\{u\in L^p(\Omega;\R^m)\;\middle|\; \Diff^\alpha u\in L^p(\Omega;\R^m)\ \forall\,\alpha\in\N_0^d,\ |\alpha|\le k\right\},
	\end{equation*}
	where $\Diff^\alpha u$ is understood in the weak sense. If $m = 1$, we abbreviate $W^{k,p}(\Omega):= W^{k,p}(\Omega;\R^1)$ and we adopt the usual conventions
	\begin{equation*}
		H^k(\Omega;\R^m):=W^{k,2}(\Omega;\R^m)\quad\text{and}\quad W^{0,p}(\Omega;\R^m):= L^p(\Omega;\R^m).
	\end{equation*}
	Moreover, we equip $W^{k,p}(\Omega;\R^m)$ with the norm
	\begin{equation*}
		\|u\|_{k,p,\Omega}
		:=
		\left(
		\sum_{|\alpha|\le k}\|\Diff^\alpha u\|_{0,p,\Omega}^p
		\right)^{1/p},
		\qquad u\in W^{k,p}(\Omega;\R^m).
	\end{equation*}
	For $k\ge 1$, we also introduce the semi-norm
	\begin{equation*}
		|u|_{k,p,\Omega}
		:=
		\left(
		\sum_{|\alpha|=k}\|\Diff^\alpha u\|_{0,p,\Omega}^p
		\right)^{1/p},
		\qquad u\in W^{k,p}(\Omega;\R^m).
	\end{equation*}
\end{definition}

Again, the special case $p = 2$ provides us with a Hilbert space and we introduce the following notation for the corresponding inner product.

\begin{proposition}[$H^k(\Omega;\R^m)$ is a Hilbert space]
	The Sobolev space $H^k(\Omega;\R^m)$ together with the inner product
	\begin{equation*}
		(u,\, v)_{k,\Omega}:= \sum_{|\alpha|\le k}(\Diff^\alpha u,\, \Diff^\alpha v)_{0,\Omega},\qquad u,v\in H^k(\Omega;\R^m)
	\end{equation*}
	is a Hilbert space and it induces the norm
	\begin{equation*}
		\|u\|_{k,\Omega} = \left(\sum_{|\alpha|\le k}\|\Diff^\alpha u\|_{0,\Omega}^2\right)^{1/2}<\infty,\qquad u\in H^k(\Omega;\R^m).
	\end{equation*}\eop
\end{proposition}

After the integer-order spaces $W^{k,p}(\Omega;\R^m)$, we also need Sobolev spaces of non-integer order. The main reason is that trace operators naturally lead to boundary spaces of fractional order. In particular, traces of $H^1(\Omega;\R^m)$-functions are, in general, not measured in another integer-order Sobolev space, but in a space of order $1/2$ on the boundary and similarly on boundary parts $\Gamma_0\subset\Gamma$. For this reason, it is convenient to introduce fractional Sobolev spaces already at this stage. For integer orders, regularity is encoded by requiring weak derivatives $\Diff^\alpha u$ to belong to $L^p$. For a non-integer order $s=k+\theta$ with $\theta\in(0,1)$, the remaining fractional part $\theta$ is measured by difference quotients. More precisely, one measures how strongly a function oscillates between pairs of points $x,y\in\Omega$, with a weight that becomes singular as $|x-y|\to 0$. This leads to the following quantity.
\begin{definition}[Slobodeckij Seminorm]
	Let $m\ge 1$, $1\le p<\infty$, and $\theta\in(0,1)$. For a measurable function $u:\Omega\to\R^m$, we define the {\tt Slobodeckij semi-norm}
	\begin{equation*}
		|u|_{\theta,p,\Omega}
		:=
		\left(\int_\Omega\int_\Omega \frac{|u(x)-u(y)|^p}{|x-y|^{d+p\theta}}\;\dx\;\dy\right)^{1/p},
	\end{equation*}
	whenever the right-hand side is finite.
\end{definition}

The quantity $|u|_{\theta,p,\Omega}$ is only a semi-norm, since it vanishes on constant functions. Combined with an $L^p$-term, however, it yields a genuine norm.

\begin{definition}[Sobolev Spaces of Fractional Order]\label{def:fractionalsobolev}
	Let $m\ge 1$, $1\le p<\infty$, and let $s=k+\theta$ with $k\in\N_0$ and $\theta\in(0,1)$. The {\tt Sobolev space} of fractional order $s$ is defined by
	\begin{equation*}
		W^{s,p}(\Omega;\R^m)
		:=
		\left\{
		u\in W^{k,p}(\Omega;\R^m)\;\middle|\; |\Diff^\alpha u|_{\theta,p,\Omega}<\infty \ \forall\,\alpha\in\N_0^d,\ |\alpha|=k
		\right\},
	\end{equation*}
	where $\Diff^\alpha u$ is understood in the weak sense. If $m=1$, we abbreviate $W^{s,p}(\Omega):=W^{s,p}(\Omega;\R^1)$ and we adopt the usual convention
	\begin{equation*}
		H^s(\Omega;\R^m):=W^{s,2}(\Omega;\R^m).
	\end{equation*}
	Moreover, we define the fractional Sobolev semi-norm
	\begin{equation*}
		|u|_{s,p,\Omega}
		:=
		\left(
		\sum_{|\alpha|=k} |\Diff^\alpha u|_{\theta,p,\Omega}^p
		\right)^{1/p},
		\qquad u\in W^{s,p}(\Omega;\R^m),
	\end{equation*}
	and equip $W^{s,p}(\Omega;\R^m)$ with the norm
	\begin{equation*}
		\|u\|_{s,p,\Omega}
		:=
		\left(
		\|u\|_{k,p,\Omega}^p + |u|_{s,p,\Omega}^p
		\right)^{1/p},
		\qquad u\in W^{s,p}(\Omega;\R^m).
	\end{equation*}
\end{definition}

 For general non-integer order $s=k+\theta$, one applies this semi-norm to all weak derivatives of order $k$. As before, the Hilbert case $p=2$ is of particular importance for the subsequent variational analysis. In particular, the case $s=1/2$ will later arise naturally in connection with trace operators and boundary spaces.

\begin{proposition}[$H^s(\Omega;\R^m)$ is a Hilbert Space]
	Let $m\ge 1$ and let $s=k+\theta$ with $k\in\N_0$ and $\theta\in(0,1)$. Then $H^s(\Omega;\R^m)$ is a Hilbert space with inner product
	\begin{equation*}
		(u,\,v)_{s,\Omega}
		:=
		(u,\,v)_{k,\Omega}
		+
		\sum_{|\alpha|=k}
		\int_\Omega\int_\Omega
		\frac{\big(\Diff^\alpha u(x)-\Diff^\alpha u(y)\big)\cdot\big(\Diff^\alpha v(x)-\Diff^\alpha v(y)\big)}
		{|x-y|^{d+2\theta}}\;\dx\;\dy
	\end{equation*}
	for $u,v\in H^s(\Omega;\R^m)$. The induced norm is
	\begin{equation}\label{eq:slobodeckijnorm}
		\|u\|_{s,\Omega}
		=
		\left(
		\|u\|_{k,\Omega}^2
		+
		\sum_{|\alpha|=k} |\Diff^\alpha u|_{\theta,2,\Omega}^2
		\right)^{1/2},
		\qquad u\in H^s(\Omega;\R^m).
	\end{equation}
\end{proposition}

\subsection{Trace Operators \& Boundary Spaces}

In variational formulations, boundary conditions and boundary forces are imposed on traces of bulk functions. For functions in $H^1(\Omega;\R^m)$, pointwise boundary values are in general not defined. The trace theorem (see \cite[Theorem 2.6]{book}) provides a canonical replacement by a continuous linear trace operator taking values in a fractional-order Sobolev space on the boundary. Throughout this subsection, we assume that $\Omega\subset\R^d$ is a bounded Lipschitz domain and write $\Gamma:=\partial\Omega$.

\begin{theorem}[Trace Theorem]\label{th:tracetheorem}
	Let $m\ge 1$. There exists a unique linear continuous operator
	\begin{equation*}
		\gamma : H^1(\Omega;\R^m)\to H^{1/2}(\Gamma;\R^m)
	\end{equation*}
	which coincides with the classical restriction on sufficiently regular functions. Moreover, there exists a constant $C_{\mathrm{tr}}>0$ such that
	\begin{equation*}
		\|\gamma u\|_{\tfrac12,\Gamma}\le C_{\mathrm{tr}}\|u\|_{1,\Omega}
		\qquad\forall\,u\in H^1(\Omega;\R^m).
	\end{equation*}
	Furthermore, $\gamma$ is surjective and admits a linear continuous right inverse
	\begin{equation*}
		\cL : H^{1/2}(\Gamma;\R^m)\to H^1(\Omega;\R^m)
	\end{equation*}
	such that
	\begin{equation*}
		\gamma(\cL g)=g
		\qquad\forall\,g\in H^{1/2}(\Gamma;\R^m).
	\end{equation*}
	The operator $\cL$ is called the lifting operator.
\end{theorem}

The trace theorem motivates the following definition of boundary spaces.

\begin{definition}[Boundary Sobolev Space of Order $1/2$]
	We define equip the space
	\begin{equation*}
		H^{1/2}(\Gamma;\R^m)=\gamma\big(H^1(\Omega;\R^m)\big).
	\end{equation*}
	with the quotient norm
	\begin{equation}\label{eq:quotientnorm}
		\|g\|_{\tfrac12,\Gamma}
		:=
		\inf\left\{\|u\|_{1,\Omega}\;\middle|\;u\in H^1(\Omega;\R^m),\ \gamma u=g\right\},\qquad g\in H^{1/2}(\Gamma;\R^d)
	\end{equation}
\end{definition}

Notice, how the notation of \eqref{eq:quotientnorm} and \eqref{eq:slobodeckijnorm} coincide, which is due to equivalency of both of these norms on $H^{1/2}(\Gamma;\R^m)$ (see \cite[Proposition 2.14]{book}). Theorem \ref{th:tracetheorem} therefore holds for both of these norms and if used later on, we will opt for the second option.

\begin{definition}[Dual Boundary Space]
	The dual boundary space of order $-1/2$ is defined by
	\begin{equation*}
		H^{-1/2}(\Gamma;\R^m):=\big(H^{1/2}(\Gamma;\R^m)\big)^\prime.
	\end{equation*}
	For $\lambda\in H^{-1/2}(\Gamma;\R^m)$ and $g\in H^{1/2}(\Gamma;\R^m)$, we denote the duality pairing by $\langle \lambda,\,g\rangle_{\Gamma}$.
\end{definition}

In applications, boundary conditions are often prescribed only on a measurable boundary part $\Gamma_0\subset\Gamma$. We therefore also use trace spaces on boundary parts.

\begin{definition}[Trace Spaces on Boundary Parts]
	Let $\Gamma_0\subset\Gamma$ be measurable.
	We define
	\begin{equation*}
		H^{1/2}(\Gamma_0;\R^m)
		:=
		\left\{ g:\Gamma_0\to\R^m\;\middle|\;\exists\,\widetilde g\in H^{1/2}(\Gamma;\R^m)\text{ with }\widetilde g|_{\Gamma_0}=g\right\},
	\end{equation*}
	with norm
	\begin{equation*}
		\|g\|_{\tfrac12,\Gamma_0}
		:=
		\inf\left\{\|\widetilde g\|_{\tfrac12,\Gamma}\;\middle|\;\widetilde g\in H^{1/2}(\Gamma;\R^m),\ \widetilde g|_{\Gamma_0}=g\right\}.
	\end{equation*}
	If $m=1$, we abbreviate $H^{1/2}(\Gamma_0):=H^{1/2}(\Gamma_0;\R^1)$.
\end{definition}

\begin{definition}[Dual Boundary Space on Boundary Parts]
	Let $\Gamma_0\subset\Gamma$ be measurable. We define
	\begin{equation*}
		H^{-1/2}(\Gamma_0;\R^m):=\big(H^{1/2}(\Gamma_0;\R^m)\big)^\prime,
	\end{equation*}
	and denote the duality pairing by
	\begin{equation*}
		\langle \lambda,\,g\rangle_{\Gamma_0},
		\qquad
		\lambda\in H^{-1/2}(\Gamma_0;\R^m),\ g\in H^{1/2}(\Gamma_0;\R^m).
	\end{equation*}
\end{definition}

For later use in Signorini-type boundary conditions, we note that for $u\in H^1(\Omega;\R^d)$ the trace $\gamma u$ belongs to $H^{1/2}(\Gamma;\R^d)$ and can therefore be restricted to boundary parts such as a contact boundary $\Gamma_C\subset\Gamma$ or a Dirichlet boundary $\Gamma_D\subset\Gamma$. Boundary data will be measured in the corresponding spaces $H^{1/2}(\Gamma_0;\R^d)$, while boundary functionals act naturally in the dual spaces $H^{-1/2}(\Gamma_0;\R^d)$.
