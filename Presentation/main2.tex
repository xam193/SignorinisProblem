\section{Signorini's Problem}

\subsection{Geometrical and Physical Setup}

\begin{frame}{Boundary Decomposition and Preview on the Boundary Conditions}
	\begin{itemize}
		\item Decompose $\Gamma = \partial\Omega$ into three relatively open and pairwise disjoint parts $\Gamma = \overline{\Gamma_D}\,\mathbin{\dot\cup}\,\overline{\Gamma_N}\,\mathbin{\dot\cup}\,\overline{\Gamma_C}$ with $|\Gamma_D|,|\Gamma_C|>0$.
		\item Each part has different boundary conditions:
		      \begin{itemize}
			      \item $\Gamma_D$: Homogeneous Dirichlet. (Clamped boundary part).
			      \item $\Gamma_N$: Neumann conditions. (We impose known surface force density).
			      \item $\Gamma_C$ is the {\em potential} contact boundary, i.e. $\Gamma_C$ is prescribed a priori but the active contact zone $\Gamma_{\mathrm{act}}(u)\subset\Gamma_C$ is unknown. On $\Gamma_C$: Contact conditions. (Nonpenetration, Complementarity, ...)
		      \end{itemize}
		\item Rigid foundation can be describe by gap-function $g:\Gamma_C\to\R$. We assume $g = 0$.
	\end{itemize}
	\begin{figure}[!h]
		\centering
		\input{Figures/setup.tex}
	\end{figure}
\end{frame}

\begin{frame}{Small Strain Elasticity}
	\begin{itemize}
		\item {\em Symmetric second order tensors}:
		      \begin{equation*}
			      \S^d:=\left\{s\in\R^d\oplus\R^d\;\middle|\;s^T = s\right\}
		      \end{equation*}
		\item {\em Stresses} are described by fields $\tau:\Omega\to\S^d$
		\item We identify $\S^d$ with symmetric $d\times d$ matrices.
		\item On $\R^{d\times d}$, we use the {\em Frobenius product}
		      \begin{equation*}
			      A:B := \sum_{j=1}^m\sum_{k=1}^n A_{jk}B_{jk},
			      \qquad
			      |A| := \sqrt{A:A}.
		      \end{equation*}
		\item If $\cA:\Omega\to\cL(\S^d;\S^d)$ is a fourth order tensor and $s\in\S^d$, we use the {\em double contraction}
		      \begin{equation*}
			      (\cA(x):s)_{ij}:=\sum_{k = 1}^d \sum_{l = 1}^d \cA_{ijkl}(x)s_{kl},
			      \qquad
			      s:\cA(x):s:=\sum_{i = 1}^d \sum_{j = 1}^d \sum_{k = 1}^d \sum_{l = 1}^d\cA_{ijkl}(x)s_{ij}s_{kl}.
		      \end{equation*}
	\end{itemize}
\end{frame}

\begin{frame}{Small Strain Elasticity}
	\begin{itemize}
		\item {\em Small strain tensor}: $\ \varepsilon:H^1(\Omega;\R^d)\to L^2(\Omega;\S^d),\qquad u\mapsto \varepsilon(u):= \frac{1}{2}\left(\nabla u + \nabla u^\top\right)$.

		\item {\em Elasticity} of $\Omega$ is describes by a {\em fourth order tensor} $\cA:\Omega\to\cL(\S^d;\S^d)$.

		\item $\cA$ is an {\em elasticity tensor field}, i.e.\
		      \begin{itemize}
			      \item $\cA$ is {\em symmetric} in the sense that
			            \begin{equation*}
				            \cA_{ijkl}(x)=\cA_{jikl}(x)=\cA_{ijlk}(x)=\cA_{klij}(x),
				            \qquad \forall\,1\le i,j,k,l\le d,
			            \end{equation*}
			      \item $\cA$ is {\em uniformly elliptic}, i.e.\ $\exists\,\alpha_{\cA}>0$ such that for a.e. $x\in\Omega$
			            \begin{equation*}
				            s:\cA(x):s \ge \alpha_\cA\, (s:s)\qquad \forall\,s\in \S^d.
			            \end{equation*}
			      \item $\cA$ is {\em bounded}, i.e.\ $\exists\,C_{\cA}>0$ such that for a.e. $x\in\Omega$
			            \begin{equation*}
				            |\cA_{ijkl}(x)|\le C_\cA \qquad \forall\,1\le i,j,k,l\le d.
			            \end{equation*}
		      \end{itemize}
		\item {\em (Cauchy) stress tensor}: $\ \sigma: H^1(\Omega;\R^d)\to L^2(\Omega;\S^d),\qquad u\mapsto \cA : \varepsilon(u)$.
	\end{itemize}
\end{frame}

\begin{frame}{External and Internal Forces}
	\begin{itemize}
		\item {\em Volume force density}: $\ f\in L^2(\Omega;\R^d)$.
		\item {\em Surface load} on Neumann part: $\ F\in L^2(\Gamma_N;\R^d)$.
		\item {\em Work = force $\cdot$ displacement}: The linear functional
		      \begin{equation*}
			      \ell(v)
			      :=
			      \int_\Omega f\cdot v\;\dx
			      +
			      \int_{\Gamma_N} F\cdot (\gamma v)\;\ds,\qquad v\in H^1(\Omega;\R^d)
		      \end{equation*}
		      describes the {\em virtual work of external forces}.
		\item {\em Internal forces} are described by the {\em stress} $\sigma(u) = \cA:\varepsilon(u)$.
		\item {\em Work = force $\cdot$ displacement}: The symmetric bilinear form
		      \begin{equation*}
			      a(u,v)
			      :=
			      \int_\Omega \sigma(u):\varepsilon(v)\;\dx,\qquad u,v\in H^1(\Omega;\R^d).
		      \end{equation*}
		      describes the {\em virtual work of internal forces}.
		\item {\em Equilibrium} between both: $\ a(u,v) = \ell(v)\qquad\forall\,v\in H^1(\Omega;\R^d)$.
	\end{itemize}
\end{frame}

\subsection{Weak Formulations}

\begin{frame}{Notations for Sobolev Spaces}
	\begin{itemize}
		\item On $H^k(\Omega;\R^d),\, k\in \N_0$, we denote the inner product and norm by
		      \begin{equation*}
			      (u,\, v)_{k,\Omega}:= \sum_{|\alpha|\le k}(\Diff^\alpha u,\, \Diff^\alpha v)_{0,\Omega},\qquad \|u\|_{k,\Omega}
			      :=
			      \left(
			      \sum_{|\alpha|\le k}\|\Diff^\alpha u\|_{0,\Omega}^2
			      \right)^{1/2}.
		      \end{equation*}
		\item $H^0(\Omega;\R^d) = L^2(\Omega;\R^d)$.
		\item On $H^s(\Omega;\R^d)$ for $s= k+\theta$ with $k\in \N_0$ and $\theta\in(0,1)$ we use
		      \begin{equation*}
			      (u,\,v)_{s,\Omega}
			      :=
			      (u,\,v)_{k,\Omega}
			      +
			      \sum_{|\alpha|=k}
			      \int_\Omega\int_\Omega
			      \frac{\big(\Diff^\alpha u(x)-\Diff^\alpha u(y)\big)\cdot\big(\Diff^\alpha v(x)-\Diff^\alpha v(y)\big)}
			      {|x-y|^{d+2\theta}}\;\dx\;\dy
		      \end{equation*}
		      and induce norm $\|\cdot\|_{s,\Omega}$.
	\end{itemize}
\end{frame}

\begin{frame}{The Trace Theorem and Boundary Spaces}
	\begin{itemize}
		\item We define the {\em classical trace operator} by
		      \begin{equation*}
			      \gamma_0:\cC^\infty(\overline{\Omega};\R^d)\to L^2(\Gamma;\R^d),
			      \qquad
			      \gamma_0 u:=u|_\Gamma.
		      \end{equation*}
		\item For $s>1/2$ we define the boundary space
		      \begin{equation*}
			      H^{s-1/2}(\Gamma;\R^d):= \gamma_0(\cC^\infty(\overline{\Omega};\R^d))
		      \end{equation*}
		      and equip it with the norm
		      \begin{equation*}
			      \|g\|_{s-\tfrac12,\Gamma}
			      :=
			      \inf\left\{
			      \|u\|_{s,\Omega}
			      \;\middle|\;
			      u\in \cC^\infty(\overline{\Omega};\R^d),\ \gamma_0 u=g
			      \right\}.
		      \end{equation*}
		\item {\bf Trace Theorem: } $\gamma_0$ extends uniquely to a continuous surjective linear operator
		      \begin{equation*}
			      \gamma: H^s(\Omega;\R^d)\to H^{s-1/2}(\Gamma;\R^d).
		      \end{equation*}
		\item For $s=1$ we have define $\ker(\gamma)=:H_0^1(\Omega;\R^d)$.
	\end{itemize}
\end{frame}

\begin{frame}{Admissible Displacements and the Nonpenetration Constraint}
	\begin{itemize}
		\item {\em Homogeneous Dirichlet} boundary conditions on $\Gamma_D$ are incorporated in the underlying Hilbert space
		      \begin{equation*}
			      V:=\left\{ v\in H^1(\Omega;\R^d)\;\middle|\; (\gamma v)|_{\Gamma_D}=0\right\}\subset H^1(\Omega;\R^d).
		      \end{equation*}
		\item {\em Outward normal}: $\ \mathrm{n}\in L^\infty(\Gamma;\R^d)$.
		\item {\em Normal Component}: $\ u_\mathrm{n}:= (\gamma u)\cdot \mathrm{n},\qquad u\in V$
		\item {\em Nonpenetration constraint}:
		      \begin{equation*}
			      u_\mathrm{n}\le g = 0\qquad\text{a.e. on }\Gamma_C
		      \end{equation*}
		\item Set of {\em admissible displacements}:
		      \begin{equation*}
			      K := \left\{ v\in V\;\middle|\; v_{\mathrm n}\le 0 \text{ a.e.\ on }\Gamma_C\right\}\subset V.
		      \end{equation*}
		\item $K$ is closed and convex in $V$.
		\item $K$ is a {\em cone}, i.e.\ if $v\in K$ then $\lambda v\in K$ for all $\lambda\in [0,\infty)$.
	\end{itemize}
\end{frame}

\begin{frame}{Energy Minimization}
	\begin{itemize}
		\item {\em Equilibrium}:
		      \begin{equation*}
			      a(u,v) = \int_\Omega \sigma(u):\varepsilon(v)\;\dx = \int_\Omega f\cdot v\;\dx +\int_{\Gamma_N} F\cdot(\gamma v)\;\ds,\qquad \forall\, v\in V
		      \end{equation*}
		\item The {\em potential energy} of the elastic body is given by the quadratic functional
		      \begin{equation*}
			      \cJ(v):= \frac{1}{2}a(v,v)-\ell(v),\qquad v\in V.
		      \end{equation*}
		\item {\em Nonpenetration}: Minimize $\cJ(v)$ over all {\em admissible} displacements $v\in K$.
	\end{itemize}

	\begin{problem}[Signorini's Problem -- Energy Minimization]
	\begin{equation*}
		\text{Find }u\in K\text{ such that:}\qquad u\in \argmin_{v\in K}\cJ(v)
	\end{equation*}
	\end{problem}
\end{frame}

\begin{frame}{Standard Properties of $a$ and $\ell$}
	\begin{proposition}
		The symmetric bilinear form $a:V\times V\to \R$ defined above is
		\begin{itemize}
			\item[i)] {\em bounded}, i.e.\ for all $v,w\in V$ there exists $C_1<\infty$ such that
			      \begin{equation*}
				      |a(u,v)|\le C_1\,\|u\|_{1,\Omega}\,\|v\|_{1,\Omega}\qquad \text{with}\qquad C_1 = C_\cA,
			      \end{equation*}
			\item[ii)] $V${\em -elliptic}, i.e.\ for all $v\in V\setminus\{0\}$ there exist $C_2>0$ such that
			      \begin{equation*}
				      a(v,v)\ge C_2\,\|v\|_{1,\Omega}^2\qquad\text{with}\qquad C_2 = \frac{\alpha_\cA}{C_K^2}.
			      \end{equation*}
		\end{itemize}
		Moreover, the linear functional $\ell$ defined above is {\em bounded}, i.e.\ for all $v\in V$ there exists $C_3>0$ such that
		\begin{equation*}
			|\ell (v)|\le C_3\,\|v\|_{1,\Omega}\qquad\text{with}\qquad C_3 = \max\{1,C_\mathrm{tr}\}\left(\|f\|_{0,\Omega}+\|F\|_{0,\Gamma_N}\right)
		\end{equation*}
	\end{proposition}
\end{frame}

\begin{frame}{Existence of Minimizer}
	\begin{lemma}[$\cJ$ is Coercive and w.l.s.c. on $V$]
		\begin{itemize}
			\item[i)] $\cJ$ is {\em coercive} on $V$, i.e.\ $\cJ(v)\to \infty$ as $\|v\|_{1,\Omega}\to\infty$.
			\item[ii)] $\cJ$ is {\em weakly lower semicontinuous} on $V$, i.e.\ if $v_j\rightharpoonup v$ in $V$, then
			      \begin{equation*}
				      \cJ(v)\le \liminf_{j\to\infty}\cJ(v_j).
			      \end{equation*}
		\end{itemize}
	\end{lemma}
	\textbf{\em Proof: }
	\begin{itemize}
		\item By boundedness of $\ell$ we have $|\ell(v)|\le \|\ell\|_{V^\prime}\|v\|_{1,\Omega}$. Hence
		      \begin{equation*}
			      \cJ(v)=\frac{1}{2}\,a(v,v)-\ell(v)
			      \ge \frac{\alpha}{2}\,\|v\|_{1,\Omega}^2-\|\ell\|_{V^\prime}\,\|v\|_{1,\Omega}
			      \xrightarrow{ \ \|v\|_{1,\Omega}\to\infty \ }\infty.
		      \end{equation*}
	\end{itemize}
\end{frame}

\begin{frame}{Existence of Minimizer}
	\begin{lemma}[$\cJ$ is Coercive and w.l.s.c. on $V$]
		\begin{itemize}
			\item[i)] $\cJ$ is {\em coercive} on $V$, i.e.\ $\cJ(v)\to \infty$ as $\|v\|_{1,\Omega}\to\infty$.
			\item[ii)] $\cJ$ is {\em weakly lower semicontinuous} on $V$, i.e.\ if $v_j\rightharpoonup v$ in $V$, then
			      \begin{equation*}
				      \cJ(v)\le \liminf_{j\to\infty}\cJ(v_j).
			      \end{equation*}
		\end{itemize}
	\end{lemma}
	\begin{lemma}[W.l.s.c. of Norms]
		Let $(X,\|\cdot\|_X)$ be a normed space. Then $\|\cdot\|_X:X\to \R_{\ge 0}$ is a weakly lower semicontinuous functional, i.e.\ if $x_j\rightharpoonup x$ in $X$, then
		\begin{equation*}
			\|x\|_X \le \liminf_{j\to\infty}\|x_j\|_X.
		\end{equation*}
	\end{lemma}
\end{frame}

\begin{frame}{Existence of Minimizer}
	\begin{lemma}[$\cJ$ is Coercive and w.l.s.c. on $V$]
		\begin{itemize}
			\item[i)] $\cJ$ is {\em coercive} on $V$, i.e.\ $\cJ(v)\to \infty$ as $\|v\|_{1,\Omega}\to\infty$.
			\item[ii)] $\cJ$ is {\em weakly lower semicontinuous} on $V$, i.e.\ if $v_j\rightharpoonup v$ in $V$, then
			      \begin{equation*}
				      \cJ(v)\le \liminf_{j\to\infty}\cJ(v_j).
			      \end{equation*}
		\end{itemize}
	\end{lemma}
	\textbf{\em Proof: }
	\begin{itemize}
		\item On $L^2(\Omega;\S^d)$, we define the norm
		      \begin{equation*}
			      \|s\|_\cA^2:= \int_\Omega s:\cA:s\;\dx,\qquad s\in L^2(\Omega;\S^d).
		      \end{equation*}
		\item We have $a(v,v) = \|\varepsilon(v)\|_\cA^2$.
		\item Since $\varepsilon$ is linear and bounded, we have $\varepsilon(v_j)\rightharpoonup\varepsilon(v)$ in $L^2(\Omega;\S^d)$.
	\end{itemize}
\end{frame}

\begin{frame}{Existence of Minimizer}
	\begin{lemma}[$\cJ$ is Coercive and w.l.s.c. on $V$]
		\begin{itemize}
			\item[i)] $\cJ$ is {\em coercive} on $V$, i.e.\ $\cJ(v)\to \infty$ as $\|v\|_{1,\Omega}\to\infty$.
			\item[ii)] $\cJ$ is {\em weakly lower semicontinuous} on $V$, i.e.\ if $v_j\rightharpoonup v$ in $V$, then
			      \begin{equation*}
				      \cJ(v)\le \liminf_{j\to\infty}\cJ(v_j).
			      \end{equation*}
		\end{itemize}
	\end{lemma}
	\textbf{\em Proof: }
	\begin{itemize}
		\item Since norms are w.l.s.c., we have
		      \begin{equation*}
			      a(v,v)=\|\varepsilon(v)\|_\cA^2
			      \le \liminf_{j\to\infty}\|\varepsilon(v_j)\|_\cA^2
			      = \liminf_{j\to\infty} a(v_j,v_j).
		      \end{equation*}
		\item $\ell\in V^\prime$ and thus $\ell(v_j)\to\ell(v)$.
		\item Therefore,
		      \begin{equation*}
			      \cJ(v)=\frac{1}{2}\,a(v,v)-\ell(v)
			      \le \frac{1}{2}\,\liminf_{j\to\infty} a(v_j,v_j) - \lim_{j\to\infty}\ell(v_j)
			      = \liminf_{j\to\infty}\cJ(v_j).
		      \end{equation*}\qed
	\end{itemize}
\end{frame}

\begin{frame}{Existence of Minimizer}
	\begin{proposition}[Existence of Minimizer of $\cJ$]
		The functional $\cJ$ has at least one minimizer $u\in K$.
	\end{proposition}
	\textbf{\em Proof:}
	\begin{itemize}
		\item Set
		      \begin{equation*}
			      M := \inf_{v\in K}\cJ(v)\in[-\infty,\infty).
		      \end{equation*}
		\item Since $0\in K$ we have $M\le \cJ(0)=0$, hence $M$ is finite from above.
		\item Let $\{v_j\}_{j\in\N}\subset K$ such that $\cJ(v_j)\to M$ as $j\to\infty$.
		\item $\{v_j\}_{j\in\N}\subset K$ is bounded since $\cJ$ is coercive.
		\item Since $V$ is Hilbert space there exists subsequence $\{v_{j_k}\}_{k\in\N}$ and some $u\in V$ such that $v_{j_k} \rightharpoonup u$ in $V$.
		\item $u\in K$ since $K$ is (weakly) closed.
		\item By w.l.s.c. of $\cJ$, $u\in K$ is minimizer of $\cJ$ because
		      \begin{equation*}
			      \cJ(u)\le \liminf_{k\to\infty}\cJ(v_{j_k})=M=\inf_{v\in K}\cJ(v).
		      \end{equation*}\eop
	\end{itemize}
\end{frame}

\begin{frame}{Uniqueness of Minimizer}
	\begin{proposition}[Uniqueness of a Minimizer of $\cJ$]
		If there exists a minimizer of the functional $\cJ$, it is unique.
	\end{proposition}
	\textbf{\em Proof: }
	\begin{itemize}
		\item $\cJ$ is strictly convex on $V$ and thus on the convex set $K$. Thus, a minimizer $u\in K$ is unique.\eop
	\end{itemize}
\end{frame}

\begin{frame}{A-Priori Stability Bound}
	\begin{proposition}[Stability estimate for the minimizer]
		The Unique minimizer $u\in K$ of $\cJ$ satisfies the a-priori bound
		\begin{equation*}
			\|u\|_{1,\Omega} \le \frac{2\,C_K^2\,\max\{1,C_\mathrm{tr}\}}{\alpha_\cA}\,\left(\|f\|_{0,\Omega} +\|F\|_{0,\Gamma_N}\right).
		\end{equation*}
	\end{proposition}
	\textbf{\em Proof:}
	\begin{itemize}
		\item Since $u\in K$ is the unique minimizer, we have $\cJ(u)\le\cJ(0) = 0$ and thus $\tfrac12a(u,u)\le \ell(u)$.
		\item For $u = 0$: trivial. Thus, assume $u\neq 0$.
		\item Recall: $a$ is $V$-elliptic with $a(u,u)\ge \alpha_\cA/C_K^2\,\|u\|_{1,\Omega}^2$ and $\ell$ is bounded with
		      \begin{equation*}
			      |\ell(u)|\le  \max\{1,C_{\mathrm{tr}}\}\left(\|f\|_{0,\Omega} +\|F\|_{0,\Gamma_N}\right)\|u\|_{1,\Omega}.
		      \end{equation*}
		\item Finally,
		      \begin{equation*}
			      \frac{\alpha_\cA}{2\,C_K^2}\,\|u\|_{1,\Omega}^2\le \frac{1}{2}\,a(u,u)\le |\ell(u)|\le  \max\{1,C_{\mathrm{tr}}\}\left(\|f\|_{0,\Omega} +\|F\|_{0,\Gamma_N}\right)\|u\|_{1,\Omega}.
		      \end{equation*}\eop
	\end{itemize}
\end{frame}

\begin{frame}{Variational Inequality as an Optimality Condition}
	\begin{proposition}[First-order optimality condition on convex sets]
		A function $u\in K$ minimizes $\cJ$ over $K$ if and only if it satisfies the variational inequality
		\begin{equation*}
			\cJ^\prime(u)[v-u] =  a(u,v-u)-\ell(v-u) \ge 0\qquad\forall\, v\in K.
		\end{equation*}
	\end{proposition}
	\textbf{\em Proof:} $\ \Longrightarrow:$
	\begin{itemize}
		\item Let $u\in K$ minimize $\cJ$ over $K$ and let $v\in K$.
		\item $K$ is convex and thus $u+t(v-u)\in K$ for all $t\in[0,1]$.
		\item $\phi(t):= \cJ(u+t(v-u))$ attains minimum at $t = 0$, i.e.\ $\phi^\prime(0^+)\ge 0$.
		\item By chainrule, $\phi'(0^+)=\cJ'(u)[v-u]\ge 0$.
	\end{itemize}
\end{frame}

\begin{frame}{Variational Inequality as an Optimality Condition}
	\begin{proposition}[First-order optimality condition on convex sets]
		A function $u\in K$ minimizes $\cJ$ over $K$ if and only if it satisfies the variational inequality
		\begin{equation*}
			\cJ^\prime(u)[v-u] =  a(u,v-u)-\ell(v-u) \ge 0\qquad\forall\, v\in K.
		\end{equation*}
	\end{proposition}
	\textbf{\em Proof:} $\ \Longleftarrow:$
	\begin{itemize}
		\item Let $u\in K$ such that  $\cJ^\prime(u)[v-u]\ge 0$ for all $v\in K$ and set $h:= v-u$.
		\item Taylor-expansion:
		      \begin{equation*}
			      \cJ(u+h)-\cJ(u)=\underbrace{\cJ'(u)[h]}_{\ge 0}+\underbrace{\frac12\,a(h,h)}_{\ge 0} \ge 0.
		      \end{equation*}
		\item Thus, $\cJ(u+h)-\cJ(u) = \cJ(v)-\cJ(u)\ge 0$ for all $v\in K$.
		\item $u\in K$ minimizes $\cJ$ over $K$.\eop
	\end{itemize}
\end{frame}

\begin{frame}{Weak Formulations / Variational Inequality}
	\begin{problem}[Signorini's Problem -- Weak Formulation 1]
	\begin{equation*}
		\text{Find }u\in K\text{ such that:}\qquad a(u,v-u)\ge \ell(v-u)\qquad \forall\, v\in K.
	\end{equation*}
	\end{problem}

	\begin{problem}[Signorini's Problem -- Weak Formulation 2]
	Find $u\in K$ such that
	\begin{equation*}
		\begin{cases}
			a(u,u) = \ell(u), &                   \\[2pt]
			a(u,v)\ge \ell(v) & \forall\, v\in K.
		\end{cases}
	\end{equation*}
	\end{problem}
\end{frame}

\subsection{Strong Formulations}

\begin{frame}{Normal Traces and Boundary Tractions}
	\begin{itemize}
		\item For $v\in H^1(\Omega;\R^d)$, we have $\gamma v\in H^{1/2}(\Gamma;\R^d)\subset L^2(\Gamma;\R^d)$.
		\item Therefore $v_\mathrm{n} = (\gamma v)\cdot \mathrm{n} \in L^2(\Gamma;\R)$.
		\item For our $u\in V$, we have $\sigma(u)\in L^2(\Omega;\S^d)$, i.e.\ trace theorem does not apply to $\sigma(u)$.
		\item If we assume $\sigma(u)\in H^1(\Omega;\S^d)$, then $\gamma\sigma(u)\in L^2(\Gamma;S^d)$ and $(\gamma\sigma(u))\cdot\mathrm{n}\in L^2(\Gamma;\R^d)$.
		\item This assumptions is much stronger than needed!
		\item In strong formulations, we need an equilibrium equation $-\diver \sigma(u) = f$ in $L^2(\Omega;\R^d)$.
		\item Therefore, we assume $\sigma(u)\in H(\diver;\Omega)$, where
		      \begin{equation*}
			      H(\diver;\Omega):= \left\{\tau\in L^2(\Omega;\S^d)\;\middle|\;\diver\tau\in L^2(\Omega;\R^d)\right\},
		      \end{equation*}
		      where $\diver \tau$ is understood row-wise in the weak sense.
		\item Normal traces on $H(\diver;\Omega)$ are understood as follows.
	\end{itemize}
\end{frame}

\begin{frame}{Normal Traces and Boundary Tractions}
	\begin{itemize}
		\item We denote by $H^{-1/2}(\Gamma;\R^d)$ the dual space of $H^{1/2}(\Gamma;\R^d)$.
		\item On $H(\diver;\Omega)$, we define a normal trace $\gamma_{\mathrm n}(\tau)\in H^{-1/2}(\Gamma;\R^d)$ via integration by parts.
		\item For smooth $\tau$, Green's identity identifies the traction $\tau\,\mathrm n$ through its boundary work:
		      \begin{equation*}
			      \int_\Omega \tau:\nabla v\;\dx+\int_\Omega (\diver\tau)\cdot v\;\dx
			      =
			      \int_\Gamma (\tau\,\mathrm n)\cdot (\gamma v)\;\ds,
		      \end{equation*}
		      i.e.\ only via testing against $\gamma v$.
		\item The left-hand side remains well-defined for $\tau\in H(\diver;\Omega)$ and $v\in H^1(\Omega;\R^d)$.
		\item Define the functional
		      \begin{equation*}
			      F_\tau(v):=\int_\Omega \tau:\nabla v\;\dx+\int_\Omega (\diver\tau)\cdot v\;\dx,
			      \qquad v\in H^1(\Omega;\R^d).
		      \end{equation*}
		\item $F_\tau$ depends only on the trace $\gamma v$, since it vanishes on $H^1_0(\Omega;\R^d)$.
		\item Therefore, $F_\tau$ induces a continuous linear functional on $H^{1/2}(\Gamma;\R^d)$:
		      \begin{equation*}
			      \langle \gamma_{\mathrm n}(\tau),\,\gamma v\rangle_\Gamma := F_\tau(v),
			      \qquad \gamma_{\mathrm n}(\tau)\in H^{-1/2}(\Gamma;\R^d).
		      \end{equation*}
		\item In the smooth case, $\gamma_{\mathrm n}(\tau)$ coincides with the classical traction $\tau\,\mathrm n$. We thus write $\tau\,\mathrm{n} = \gamma_{\mathrm n}(\tau)$.
	\end{itemize}
\end{frame}

\begin{frame}{Stress Regularity Assumption}
	\begin{itemize}
		\item Thus, we further assume $\sigma(u)\in H(\diver;\Omega)$.
		\item {\em Boundary Traction}: $\ \sigma(u)\mathrm{n}:= \gamma_\mathrm{n}(\sigma(u))\in H^{-1/2}(\Gamma;\R^d)$.
		\item Since $\sigma(u)$ is symmetric, we have
		      \begin{equation*}
			      \sigma(u):\nabla v = \sigma(u):\varepsilon(v) \qquad\forall\, v\in V.
		      \end{equation*}
		\item Therefore
		      \begin{equation*}
			      \langle \sigma(u)\mathrm n,\gamma v\rangle_\Gamma
			      =
			      \int_\Omega \sigma(u):\varepsilon(v)\;\dx
			      +
			      \int_\Omega \diver\sigma(u)\cdot v\;\dx
			      \qquad
			      \forall\, v\in V.
		      \end{equation*}
		\item {\em Useful identity}:
		      \begin{equation*}
			      a(u,v)-\ell(v) = \langle \sigma(u)\mathrm n,\gamma v\rangle_\Gamma-\int_{\Gamma_N} F\cdot (\gamma v)\;\ds,\qquad \forall\, v\in V.
		      \end{equation*}
	\end{itemize}
\end{frame}

\begin{frame}{A Hybrid Formulation}
	\begin{problem}[Signorini's Problem -- Hybrid Formulation]
	For given $f\in L^2(\Omega;\R^d)$ and $F\in L^2(\Gamma_N;\R^d)$, find $u\in K$ such that $\sigma(u)\in H(\diver;\Omega)$ and
	\begin{subequations}\label{eq:hybrid}
		\begin{empheq}[left=\empheqlbrace]{alignat=2}
			-\diver\sigma(u) &= f
			&\qquad& \text{a.e.\ in }\Omega, \label{eq:hybrid_bulk}\\
			\langle \sigma(u)\mathrm n,\gamma u\rangle_\Gamma
			&= \mathrlap{\int_{\Gamma_N} F\cdot (\gamma u)\;\ds,}
			&& \label{eq:hybrid_eq}\\
			\langle \sigma(u)\mathrm n,\gamma v\rangle_\Gamma
			&\ge \int_{\Gamma_N} F\cdot (\gamma v)\;\ds
			&\qquad& \forall\, v\in K.
			\label{eq:hybrid_ineq}
		\end{empheq}
	\end{subequations}
	\end{problem}
\end{frame}

\begin{frame}{Equivalence of Hybrid and Weak Formulation}
	\begin{proposition}[Equivalence of Hybrid and Weak Formulation]
		Assume $\sigma(u)\in H(\diver;\Omega)$. Then $u\in K$ solves the hybrid problem above if and only if it solves the weak problem
		\begin{equation*}
			\text{Find }u\in K\text{ such that}\qquad a(u,v-u)\ge\ell(v-u)\qquad \forall v\in K.
		\end{equation*}
	\end{proposition}
	\textbf{\em Proof:} $\ \textit{Hybrid }\Rightarrow\textit{ Weak}$:
	\begin{itemize}
		\item Subtract \eqref{eq:hybrid_eq} from \eqref{eq:hybrid_ineq}:
		      \begin{equation*}
			      \langle \sigma(u)\mathrm n,\gamma (v-u)\rangle_\Gamma - \int_{\Gamma_N}F\cdot \gamma (v-u)\;\ds \ge 0\qquad\forall\, v\in K.
		      \end{equation*}
		\item Insert the identity from the above:
		      \begin{equation*}
			      a(u,v-u)-\ell(v-u)\ge 0
			      \qquad\forall\,v\in K,
		      \end{equation*}
		      i.e.\ the weak formulation.
	\end{itemize}
\end{frame}

\begin{frame}{Equivalence of Hybrid and Weak Formulation}
	\begin{proposition}[Equivalence of Hybrid and Weak Formulation]
		Assume $\sigma(u)\in H(\diver;\Omega)$. Then $u\in K$ solves the hybrid problem above if and only if it solves the weak problem
		\begin{equation*}
			\text{Find }u\in K\text{ such that}\qquad a(u,v-u)\ge\ell(v-u)\qquad \forall v\in K.
		\end{equation*}
	\end{proposition}
	\textbf{\em Proof:} $\ \textit{Weak }\Rightarrow\textit{ Hybrid (Bulk Equation)}$:
	\begin{itemize}
		\item Let $\varphi\in \cD(\Omega;\R^d)$. Then $u\pm t\varphi\in K$ for $|t|\ll 1$, hence the weak inequality implies
		      \begin{equation*}
			      a(u,\varphi)=\ell(\varphi)=\int_\Omega f\cdot \varphi\;\dx = \langle f,\,\varphi\rangle\qquad \forall\,\varphi\in \cD(\Omega;\R^d).
		      \end{equation*}
		\item The left hand side is
		      \begin{equation*}
			      a(u,\varphi) = \int_\Omega\sigma(u):\varepsilon(\varphi) = \int_\Omega\sigma(u):\nabla\varphi\;\dx = \langle -\diver\sigma(u),\,\varphi\rangle.
		      \end{equation*}
		\item Thus, $-\diver\sigma(u) = f$ in $\cD^\prime(\Omega;\R^d)$ and by $\sigma(u)\in H(\diver;\Omega)$, it holds in $L^2(\Omega;\R^d)$ as well.
	\end{itemize}
\end{frame}

\begin{frame}{Equivalence of Hybrid and Weak Formulation}
	\begin{proposition}[Equivalence of Hybrid and Weak Formulation]
		Assume $\sigma(u)\in H(\diver;\Omega)$. Then $u\in K$ solves the hybrid problem above if and only if it solves the weak problem
		\begin{equation*}
			\text{Find }u\in K\text{ such that}\qquad a(u,v-u)\ge\ell(v-u)\qquad \forall v\in K.
		\end{equation*}
	\end{proposition}
	\textbf{\em Proof:} $\ \textit{Weak }\Rightarrow\textit{ Hybrid (Hybrid Equality)}$:
	\begin{itemize}
		\item Since $K$ is a cone, we have $0,2u\in K$.
		\item Testing both in the variational inequality yields $a(u,u)\le \ell(u)$ and $a(u,u)\ge \ell(u)$ and thus $a(u,u)= \ell(u)$.
		\item The identity earlier yields
		      \begin{equation*}
			      0 = a(u,u)-\ell(u) = \langle\sigma(u)\mathrm{n},\,\gamma u\rangle_\Gamma -\int_{\Gamma_N}F\cdot\gamma u\;\ds.
		      \end{equation*}
		\item This is exactly \eqref{eq:hybrid_eq}.
	\end{itemize}
\end{frame}

\begin{frame}{Equivalence of Hybrid and Weak Formulation}
	\begin{proposition}[Equivalence of Hybrid and Weak Formulation]
		Assume $\sigma(u)\in H(\diver;\Omega)$. Then $u\in K$ solves the hybrid problem above if and only if it solves the weak problem
		\begin{equation*}
			\text{Find }u\in K\text{ such that}\qquad a(u,v-u)\ge\ell(v-u)\qquad \forall v\in K.
		\end{equation*}
	\end{proposition}
	\textbf{\em Proof:} $\ \textit{Weak }\Rightarrow\textit{ Hybrid (Hybrid Inequality)}$:
	\begin{itemize}
		\item With the identity earlier, we have for all $v\in V$ that
		      \begin{equation*}
			      a(u,v-u) = \langle \sigma(u)\mathrm{n},\, \gamma(v-u)\rangle_\Gamma + \int_\Omega f\cdot(v-u)\;\dx \ge \ell(v-u)=\int_\Omega f\cdot(v-u)\;\dx +\int_{\Gamma_N}F\cdot\gamma(v-u)\;\ds
		      \end{equation*}
		\item Subtracting volume term yields: $\ \displaystyle  \langle \sigma(u)\mathrm n,\gamma(v-u)\rangle_\Gamma \ge\int_{\Gamma_N}F\cdot\gamma(v-u)\;\ds\qquad \forall\,v\in V$.
		\item This means
		      \begin{equation*}
			      \langle\sigma(u)\mathrm{n}\,\gamma v\rangle_\Gamma\ge \int_{\Gamma_N}F\cdot \gamma v\;\ds + \underbrace{\left[\langle\sigma(u)\mathrm{n}\,\gamma u\rangle_\Gamma-\int_{\Gamma_N}F\cdot\gamma u\;\ds\right]}_{ = 0}\qquad \forall\,v\in V.
		      \end{equation*}
	\end{itemize}
\end{frame}

\begin{frame}{Boundary Spaces on Boundary Parts}
	\begin{itemize}
		\item Recall: $\gamma: H^s(\Omega;\R^d)\to H^{s-1/2}(\Gamma;\R^d)$
		\item Let $\Gamma_0\subset\Gamma$ be open in $\Gamma$.
		\item We define
		      \begin{equation*}
			      H^{s-1/2}(\Gamma_0;\R^m)
			      :=
			      \left\{g:\Gamma_0\to \R^m\;\middle|\;\exists\,\widetilde g\in H^{s-1/2}(\Gamma;\R^m)\text{\em\enspace with }\widetilde g|_{\Gamma_0}=g\right\}.
		      \end{equation*}
		\item Moreover, we define
		      \begin{equation*}
			      H^{s-1/2}_{00}(\Gamma_0;\R^m)
			      :=
			      \left\{g\in H^{s-1/2}(\Gamma_0;\R^m)\;\middle|\;\mathrm{Ext}_0(g)\in H^{s-1/2}(\Gamma;\R^m)\right\},
		      \end{equation*}
		      where $\mathrm{Ext}_0(g)$ denotes the {\em zero extension} of $g$ from $\Gamma_0$ to $\Gamma$.
		\item We define
		      \begin{equation*}
			      H^{-1/2}(\Gamma_0;\R^m):=\big(H^{1/2}_{00}(\Gamma_0;\R^m)\big)^\prime.
		      \end{equation*}
		      with duality pairing $\langle\cdot,\,\cdot\rangle_{\Gamma_0}$.
	\end{itemize}
\end{frame}

\begin{frame}{Normal/Tangential Splitting and Contact Trace Spaces}
	\begin{itemize}
		\item For {\em stronger} boundary conditions, we need to split functions into their {\em normal} and {\em tangential} component.
		\item {\em Contact trace space}:
		      \begin{equation*}
			      W_C := \left\{\gamma v|_{\Gamma_C}\; \middle|\; v\in V\right\}\subset H^{1/2}(\Gamma_C;\R^d).
		      \end{equation*}
		\item For $w\in W_C$, we define its {\em normal} and {\em tangential component} by
		      \begin{equation*}
			      w_\mathrm{n} := w\cdot \mathrm n
			      \qquad\text{and}\qquad
			      w_t := w - (w\cdot \mathrm n)\mathrm n.
		      \end{equation*}
		\item {\em Normal and tangential trace spaces}:
		      \begin{equation*}
			      W_{C,\mathrm{n}} := \left\{w\cdot \mathrm n\; \middle|\; w\in W_C\right\}
			      \qquad\text{and}\qquad
			      W_{C,t} := \left\{w\in W_C\; \middle|\; w\cdot \mathrm n = 0\right\}.
		      \end{equation*}
		\item Every $w\in W_C$ admits the decomposition
		      \begin{equation*}
			      w = w_\mathrm{n}\mathrm n + w_t\in W_{C,\mathrm{n}}\,\mathrm{n}\oplus W_{C,t} = W_C
		      \end{equation*}
		\item Dual Spaces $W^\prime_C,\, W_{C,\mathrm{n}}^\prime,\,W_{C,t}^\prime$ and duality pairing $\langle\cdot,\,\cdot\rangle_{\Gamma_C}$.
	\end{itemize}
\end{frame}

\begin{frame}{Normal Traces on Boundary Parts}
	\begin{itemize}
		\item Let $\Gamma_0\subset\Gamma$ be open in $\Gamma$.
		\item The normal trace of $\tau\in H(\diver;\Omega)$ on $\Gamma_0$ is defined as the unique element $\gamma_{\mathrm{n},0}(\tau)\in H^{-1/2}(\Gamma_0;\R^d)$ given by
		      \begin{equation*}
			      \langle \gamma_{\mathrm{n},0}(\tau),\,g\rangle_{\Gamma_0}
			      :=
			      \langle \gamma_{\mathrm{n}}(\tau),\,\mathrm{Ext}_0 (g)\rangle_{\Gamma},
			      \qquad g\in H^{1/2}_{00}(\Gamma_0;\R^d),
		      \end{equation*}
		      where $\mathrm{Ext}_0 (g)$ denotes the zero extension of $g$ from $\Gamma_0$ to $\Gamma$.
		\item We have $H^{1/2}_{00}(\Gamma_C;\R^d)\subset W_C$ and thus $W_C^\prime\subset H^{-1/2}(\Gamma_C;\R^d)$.
		\item Therefore the normal trace of $\tau\in H(\diver;\Omega)$ on $\Gamma_C$ is the unique element $\gamma_n,C\in W_C^\prime$ given by $\langle \gamma_{\mathrm{n},0}(\tau),\,g\rangle_{\Gamma_C}$
	\end{itemize}
\end{frame}

\begin{frame}{Normal Traces on Boundary Parts}
	\begin{itemize}
		\item We have $H^{1/2}_{00}(\Gamma_C;\R^d)\subset W_C$, hence every
		      $\lambda_C\in W_C^\prime$ defines by restriction an element of
		      $H^{-1/2}(\Gamma_C;\R^d)=\big(H^{1/2}_{00}(\Gamma_C;\R^d)\big)^\prime$.
		\item For $\tau\in H(\diver;\Omega)$, the local normal trace on $\Gamma_C$ is
		      the element $\gamma_{\mathrm{n},0}(\tau)\in H^{-1/2}(\Gamma_C;\R^d)$
		      defined by
		      \begin{equation*}
			      \langle \gamma_{\mathrm{n},0}(\tau),g\rangle_{\Gamma_C}
			      :=
			      \langle \gamma_{\mathrm n}(\tau),\mathrm{Ext}_0(g)\rangle_\Gamma,
			      \qquad
			      g\in H^{1/2}_{00}(\Gamma_C;\R^d).
		      \end{equation*}
		\item In general, $\gamma_{\mathrm{n},0}(\tau)$ need not extend continuously to
		      all of $W_C$.
	\end{itemize}
\end{frame}

\begin{frame}{Classical Strong Formulation under Additional Regularity}
	\begin{problem}[Signorini's Problem -- Classical Strong Formulation]
	For given $f\in L^2(\Omega;\R^d)$ and $F\in L^2(\Gamma_N;\R^d)$ find $u\in H^s(\Omega;\R^d)$ with $s>3/2$ such that the {\em small strain elasticity equations}
	\begin{subequations}\label{eq:strong_classical}
		\begin{empheq}[left=\empheqlbrace]{alignat=2}
			-\diver\sigma(u) &= f
			&\qquad& \text{in }\Omega, \label{eq:strong_classical_bulk}\\
			\sigma(u) &= \mathcal A:\varepsilon(u)
			&\qquad& \text{in }\Omega, \label{eq:strong_classical_constitutive}\\
			u &= 0
			&\qquad& \text{on }\Gamma_D, \label{eq:strong_classical_dirichlet}\\
			\sigma(u)\mathrm n &= F
			&\qquad& \text{on }\Gamma_N, \label{eq:strong_classical_neumann}
		\end{empheq}
	\end{subequations}
	and the {\em contact conditions}
	\begin{subequations}\label{eq:strong_contact}
		\begin{empheq}[left=\empheqlbrace]{alignat=2}
			u_\mathrm{n} &\le 0
			&\qquad& \text{on }\Gamma_C, \label{eq:strong_contact_nonpen}\\
			\sigma_\mathrm{n}(u) &\le 0
			&\qquad& \text{on }\Gamma_C, \label{eq:strong_contact_sigma_n}\\
			\sigma_\mathrm{n}(u)\,u_\mathrm{n} &= 0
			&\qquad& \text{on }\Gamma_C, \label{eq:strong_contact_comp}\\
			\sigma_t(u) &= 0
			&\qquad& \text{on }\Gamma_C \label{eq:strong_contact_tangential}
		\end{empheq}
	\end{subequations}
	hold.
	\end{problem}
\end{frame}
