\section{Introduction}
\begin{frame}{Introduction}
	\begin{itemize}
		\item Body $\Omega\subset\R^d,\,d\in\{2,3\}$ on {\em rigid} foundation subject to elasticity properties and forces.
		\item No friction between $\Omega$ and foundation.
		\item Not in equilibrium configuration.

		\item Complicated part: Contact Boundary.
		\item Rigid foundation can be described by gap-function $g$ but we assume a flat foundation, i.e.\ $g = 0$.
	\end{itemize}

\section{Sobolev Spaces}

\subsection{Integer Order Sobolev Spaces}

\begin{frame}{Integer Order Sobolev Spaces}
	\begin{itemize}
		\item Let $\Omega\subset\R^d,\,d\ge 1$ be nonempty, open and bounded.
		\item Integer order Sobolev space:
		      \begin{equation*}
			      H^k(\Omega;\R^m)
			      :=
			      \left\{u\in L^2(\Omega;\R^m)\;\middle|\; \Diff^\alpha u\in L^2(\Omega;\R^m)\ \forall\,\alpha\in\N_0^d,\ |\alpha|\le k\right\},\qquad k\in\N_0.
		      \end{equation*}
		\item Set $H^0(\Omega;\R^m) = L^2(\Omega;\R^m)$ and denote $L^2$-Product/Norm by $(\cdot,\,\cdot)_{0,\Omega}$ and $\|\cdot\|_{0,\Omega}$.
		\item $H^k(\Omega;\R^m)$ is Hilbert space with
		      \begin{equation*}
			      (u,\, v)_{k,\Omega}:= \sum_{|\alpha|\le k}(\Diff^\alpha u,\, \Diff^\alpha v)_{0,\Omega},\qquad \|u\|_{k,\Omega}
			      :=
			      \left(
			      \sum_{|\alpha|\le k}\|\Diff^\alpha u\|_{0,\Omega}^2
			      \right)^{1/2}
		      \end{equation*}
		\item For $k=1$ we get
		      \begin{equation*}
			      \|u\|_{1,\Omega} = \left(\|u\|_{0,\Omega}^2 + \|\nabla u\|_{0,\Omega}^2\right)^{1/2},\qquad u\in H^1(\Omega;\R^m).
		      \end{equation*}
	\end{itemize}
\end{frame}

\subsection{Fractional Order Sobolev Spaces}

\begin{frame}{Fractional Order Sobolev Spaces}
	\begin{itemize}
		\item For $k\in\N_0$ and $\theta\in (0,1)$ let $s:= k+\theta$.
		\item Slobodeckij semi-norm:
		      \begin{equation*}
			      |u|_{\theta,\Omega}
			      :=
			      \left(\int_\Omega\int_\Omega \frac{|u(x)-u(y)|^2}{|x-y|^{d+2\theta}}\;\dx\;\dy\right)^{1/2}.
		      \end{equation*}

		\item Fractional order Sobolev space:
		      \begin{equation*}
			      H^s(\Omega;\R^m)
			      :=
			      \left\{
			      u\in H^k(\Omega;\R^m)\;\middle|\; |\Diff^\alpha u|_{\theta,\Omega}<\infty \ \forall\,\alpha\in\N_0^d,\ |\alpha|=k
			      \right\}.
		      \end{equation*}
		\item $H^s(\Omega;\R^m)$ is Hilbert space with
		      \begin{equation*}
			      (u,\,v)_{s,\Omega}
			      :=
			      (u,\,v)_{k,\Omega}
			      +
			      \sum_{|\alpha|=k}
			      \int_\Omega\int_\Omega
			      \frac{\big(\Diff^\alpha u(x)-\Diff^\alpha u(y)\big)\cdot\big(\Diff^\alpha v(x)-\Diff^\alpha v(y)\big)}
			      {|x-y|^{d+2\theta}}\;\dx\;\dy
		      \end{equation*}
		      and $\|u\|_{s,\Omega}
			      :=
			      \left(
			      \|u\|_{k,\Omega}^2 + |u|_{s,\Omega}^2
			      \right)^{1/2}$
		\item Boundary traces of $H^k$-functions live in $H^s$ spaces.
	\end{itemize}
\end{frame}

\subsection{Trace Operators and Boundary Spaces}

\begin{frame}{The Classical Trace Operator}
	\begin{itemize}
		\item Let $\Omega$ have additionally Lipschitz boundary $\Gamma:= \partial\Omega$.
		\item Classical Trace Operator (restriction to boundary):
		      \begin{equation*}
			      \gamma_0:\cC^\infty(\overline{\Omega};\R^m)\to L^2(\Gamma;\R^m),
			      \qquad
			      \gamma_0 u:=u|_\Gamma
		      \end{equation*}
		\item For $s>1/2$ we define the boundary space
		      \begin{equation*}
			      H^{s-1/2}(\Gamma;\R^m):= \gamma_0(\cC^\infty(\overline{\Omega};\R^m))
		      \end{equation*}
		      and equip it with the norm
		      \begin{equation*}
			      \|g\|_{s-\tfrac12,\Gamma}
			      :=
			      \inf\left\{
			      \|u\|_{s,\Omega}
			      \;\middle|\;
			      u\in \cC^\infty(\overline{\Omega};\R^m),\ \gamma_0 u=g
			      \right\}.
		      \end{equation*}
		\item Same set for all $s>1/2$ but different norms.
	\end{itemize}
\end{frame}

\begin{frame}{The Trace Theorem}
	\begin{theorem}[Trace Theorem]
		Let $1/2< s\le 1$. Then the classical trace operator $\gamma_0$ extends uniquely to a continuous surjective linear operator
		\begin{equation*}
			\gamma:H^s(\Omega;\R^m)\to H^{s-1/2}(\Gamma;\R^m).
		\end{equation*}
		In particular, there exists a constant $C_{\tr,s}>0$ such that
		\begin{equation*}
			\|\gamma u\|_{s-\tfrac12,\Gamma}\le C_{\tr,s}\|u\|_{s,\Omega}
			\qquad\forall\,u\in H^s(\Omega;\R^m).
		\end{equation*}
		In the special case $s=1$, one has
		\begin{equation*}
			\ker(\gamma)=\overline{\cD(\Omega;\R^m)}^{\|\cdot\|_{1,\Omega}}=:H_0^1(\Omega;\R^m).
		\end{equation*}
	\end{theorem}
\end{frame}

\begin{frame}{Boundary Spaces on Boundary Parts}
	\begin{itemize}
		\item Let $\Gamma_0\subset\Gamma$ be open in $\Gamma$.
		\item We define
		      \begin{equation*}
			      H^{s-1/2}(\Gamma_0;\R^m)
			      :=
			      \left\{g:\Gamma_0\to \R^m\;\middle|\;\exists\,\widetilde g\in H^{s-1/2}(\Gamma;\R^m)\text{\em\enspace with }\widetilde g|_{\Gamma_0}=g\right\},
		      \end{equation*}
		      with norm
		      \begin{equation*}
			      \|g\|_{s-\tfrac12,\Gamma_0}
			      :=
			      \inf\left\{
			      \|\widetilde g\|_{s-\tfrac12,\Gamma}
			      \;\middle|\;
			      \widetilde g\in H^{s-1/2}(\Gamma;\R^m),\ \widetilde g|_{\Gamma_0}=g
			      \right\}.
		      \end{equation*}
		\item Moreover, we define
		      \begin{equation*}
			      H^{s-1/2}_{00}(\Gamma_0;\R^m)
			      :=
			      \left\{g\in H^{s-1/2}(\Gamma_0;\R^m)\;\middle|\;\mathrm{Ext}_0(g)\in H^{s-1/2}(\Gamma;\R^m)\right\},
		      \end{equation*}
		      where $\mathrm{Ext}_0(g)$ denotes the zero extension of $g$ from $\Gamma_0$ to $\Gamma$ with the norm
		      \begin{equation*}
			      \|g\|_{s-\tfrac12,00,\Gamma_0}:=\|\mathrm{Ext}_0(g)\|_{s-\tfrac12,\Gamma}.
		      \end{equation*}
	\end{itemize}
\end{frame}

\begin{frame}{Dual Spaces of Boundary Spaces}
	\begin{itemize}
		\item We define
		      \begin{equation*}
			      H^{-1/2}(\Gamma;\R^m):=\big(H^{1/2}(\Gamma;\R^m)\big)^\prime,\qquad  H^{-1/2}(\Gamma_0;\R^m):=\big(H^{1/2}_{00}(\Gamma_0;\R^m)\big)^\prime.
		      \end{equation*}
		      with duality pairings $\langle\cdot,\,\cdot\rangle_\Gamma$ and $\langle\cdot,\,\cdot\rangle_{\Gamma_0}$, respectively.
	\end{itemize}
\end{frame}

\begin{frame}{Lifting Operators}
	\begin{theorem}[Lifting Operator]
		Let $\Omega\subset\R^d$ be a bounded Lipschitz domain and let $\Gamma_0\subset\Gamma$ be relatively open.

		\begin{itemize}
			\item[i)] There exists a bounded linear operator $\cL:H^{1/2}(\Gamma;\R^d)\to H^1(\Omega;\R^d)$ such that
			      \begin{equation*}
				      \gamma(\cL g)=g
				      \qquad\forall\,g\in H^{1/2}(\Gamma;\R^d).
			      \end{equation*}

			\item[ii)] There exists a bounded linear operator $ \cL:H^{1/2}_{00}(\Gamma_0;\R^d)\to H^1(\Omega;\R^d)$
			      such that for every $g\in H^{1/2}_{00}(\Gamma_0;\R^d)$,
			      \begin{equation*}
				      \gamma(\cL g)|_{\Gamma_0}=g
				      \qquad\text{and}\qquad
				      \gamma(\cL g)=0\ \text{on }\Gamma\setminus\Gamma_0.
			      \end{equation*}
		\end{itemize}
	\end{theorem}
\end{frame}

\subsection{\texorpdfstring{$H(\diver;\Omega)$}{Hdiv} and Normal Traces}

\begin{frame}{$H(\diver;\Omega)$ for Matrix-Valued Fields}
	\begin{itemize}
		\item The space $\mathrm{Sym}^2(\R^d)$ of symmetric second order tensors is
		      \begin{equation*}
			      \mathrm{Sym}^2(\R^d):= \left\{ s\in\R^d\otimes\R^d\;\middle|\; s^\top = s\right\}.
		      \end{equation*}
		\item We identify $\mathrm{Sym}^2(\R^d)$ with symmetric matrices in $\R^{d\times d}$.
		\item In elasticity, stresses are fields $\tau:\Omega\to\mathrm{Sym}^2(\R^d)$.
		\item On $\R^{d\times d}$, we use the Frobenius product
		      \begin{equation*}
			      A:B := \sum_{j=1}^m\sum_{k=1}^n A_{jk}B_{jk},
			      \qquad
			      |A| := \sqrt{A:A}.
		      \end{equation*}
		\item If $\cA:\Omega\to\cL(\mathrm{Sym}^2(\R^d);\mathrm{Sym}^2(\R^d))$ is a fourth order tensor and $s\in\mathrm{Sym}^2(\R^d)$, we use the {\em double contraction}
		      \begin{equation*}
			      (\cA(x):s)_{ij}:=\sum_{k = 1}^d \sum_{l = 1}^d \cA_{ijkl}(x)s_{kl},
			      \qquad
			      s:\cA(x):s:=\sum_{i = 1}^d \sum_{j = 1}^d \sum_{k = 1}^d \sum_{l = 1}^d\cA_{ijkl}(x)s_{ij}s_{kl}.
		      \end{equation*}
		\item We define
		      \begin{equation*}
			      H(\diver;\Omega)
			      :=
			      \left\{\tau\in L^2(\Omega;\mathrm{Sym}^2(\R^d))\;\middle|\;\diver\tau\in L^2(\Omega;\R^d)\right\},
		      \end{equation*}
		      where $\diver\tau$ is understood row-wise in the weak sense.
	\end{itemize}
\end{frame}


\begin{frame}{Normal Traces and Boundary Tractions}
	\begin{itemize}
		\item Let $\mathrm{n}:\Gamma\to\R^d$ be the outward unit normal. For Lipschitz $\Omega$ one has $\mathrm{n}\in L^\infty(\Gamma;\R^d)$.
		\item If $v\in H^1(\Omega;\R^d)$, then its trace satisfies
		      \begin{equation*}
			      \gamma v\in H^{1/2}(\Gamma;\R^d)\subset L^2(\Gamma;\R^d).
		      \end{equation*}
		      Hence the normal component
		      \begin{equation*}
			      v_{\mathrm n}:=(\gamma v)\cdot \mathrm n
		      \end{equation*}
		      lies in $L^2(\Gamma)$.
		\item For a stress field $\tau\in L^2(\Omega;\R^{d\times d})$, classical boundary values $\gamma\tau$ are \emph{not} defined in general, so the pointwise traction $\tau\,\mathrm n$ is not available.
		\item If we assumed $\tau\in H^1(\Omega;\R^{d\times d})$, then $\gamma\tau\in L^2(\Gamma;\R^{d\times d})$ and $(\gamma\tau)\mathrm n\in L^2(\Gamma;\R^d)$.
		\item This is often much stronger than needed. If we want an equilibrium equation
		      \begin{equation*}
			      -\diver\tau = f \quad\text{in }L^2(\Omega;\R^d),
		      \end{equation*}
		      it suffices to require $\diver\tau\in L^2(\Omega;\R^d)$.
		\item This leads naturally to $\tau\in H(\diver;\Omega)$.
		\item On $H(\diver;\Omega)$, the trace theorem does not hold. We need a different concept here.
	\end{itemize}
\end{frame}

\begin{frame}{Normal Traces and Boundary Tractions}
	\begin{itemize}
		\item On $H(\diver;\Omega)$, we define a normal trace $\gamma_{\mathrm n}(\tau)\in H^{-1/2}(\Gamma;\R^d)$ via integration by parts.
		\item For smooth $\tau$, Green's identity identifies the traction $\tau\,\mathrm n$ through its boundary work:
		      \begin{equation*}
			      \int_\Omega \tau:\nabla v\;\dx+\int_\Omega (\diver\tau)\cdot v\;\dx
			      =
			      \int_\Gamma (\tau\,\mathrm n)\cdot (\gamma v)\;\ds,
		      \end{equation*}
		      i.e.\ only via testing against $\gamma v$.
		\item The left-hand side remains well-defined for $\tau\in H(\diver;\Omega)$ and $v\in H^1(\Omega;\R^d)$.
		\item Define the functional
		      \begin{equation*}
			      F_\tau(v):=\int_\Omega \tau:\nabla v\;\dx+\int_\Omega (\diver\tau)\cdot v\;\dx,
			      \qquad v\in H^1(\Omega;\R^d).
		      \end{equation*}
		\item One can show: $F_\tau$ depends only on the trace $\gamma v$, since it vanishes on $H^1_0(\Omega;\R^d)$.
		\item Therefore, $F_\tau$ induces a continuous linear functional on $H^{1/2}(\Gamma;\R^d)$:
		      \begin{equation*}
			      \langle \gamma_{\mathrm n}(\tau),\,\gamma v\rangle_\Gamma := F_\tau(v),
			      \qquad \gamma_{\mathrm n}(\tau)\in H^{-1/2}(\Gamma;\R^d).
		      \end{equation*}
		\item In the smooth case, $\gamma_{\mathrm n}(\tau)$ coincides with the classical traction $\tau\,\mathrm n$. We thus write $\tau\,\mathrm{n} = \gamma_{\mathrm n}(\tau)$.
	\end{itemize}
\end{frame}

\begin{frame}{Normal Traces on Boundary Parts}
	\begin{itemize}
		\item Let $\Gamma_0\subset\Gamma$ be open in $\Gamma$.
		\item The normal trace on $\Gamma_0$ is defined as the unique element $\gamma_{\mathrm{n},0}(\tau)\in H^{-1/2}(\Gamma_0;\R^d)$ given by
		      \begin{equation*}
			      \langle \gamma_{\mathrm{n},0}(\tau),\,g\rangle_{\Gamma_0}
			      :=
			      \langle \gamma_{\mathrm{n}}(\tau),\,\mathrm{Ext}_0 (g)\rangle_{\Gamma},
			      \qquad g\in H^{1/2}_{00}(\Gamma_0;\R^d),
		      \end{equation*}
		      where $\mathrm{Ext}_0 (g)$ denotes the zero extension of $g$ from $\Gamma_0$ to $\Gamma$.
	\end{itemize}
\end{frame}

\section{Signorini's Problem}
\subsection{Geometric Setup}
\begin{frame}{Boundary Decomposition}
	\begin{itemize}
		\item Elastic Body $\Omega\subset\R^d,\,d\in\{2,3\}$ open and bounded domain with Lipschitz boundary $\Gamma:= \partial\Omega$.
		\item Decompose $\Gamma$ into three relatively open and pairwise disjoint parts
		      \begin{equation*}
			      \Gamma = \overline{\Gamma_D}\,\mathbin{\dot\cup}\,\overline{\Gamma_N}\,\mathbin{\dot\cup}\,\overline{\Gamma_C},\qquad |\Gamma_D|,|\Gamma_C|>0.
		      \end{equation*}
		\item Outward normal vector $\mathrm{n}\in L^\infty(\Gamma;\R^d)$.
		\item We aim to find displacement function $u\in H^1(\Omega;\R^d)$.
		\item The body in equilibrium $\Omega$ configuration is $\varphi(\Omega)$ with $\varphi(x) = x+u(x)$.
	\end{itemize}
\end{frame}

\begin{frame}{Boundary Decomposition}
	\begin{itemize}
		\item Each part has different boundary conditions:
		      \begin{itemize}
			      \item $\Gamma_D$: Homogeneous Dirichlet. (Clamped boundary part).
			      \item $\Gamma_N$: Neumann conditions. (We impose known surface force density).
			      \item $\Gamma_C$ is the {\em potential} contact boundary, i.e. $\Gamma_C$ is prescribed a priori but the active contact zone $\Gamma_{\mathrm{act}}(u)\subset\Gamma_C$ is unknown. On $\Gamma_C$: Contact conditions. (Nonpenetration, Complementarity, ...)
		      \end{itemize}
		\item Rigid foundation can be describe by function $g:\Gamma_C\to\R$ on the distance between $\Gamma_C$ and the foundation. We assume $g = 0$.
		\item No friction between $\Omega$ and foundation.
	\end{itemize}

		\centering
		\input{Figures/setup.tex}
	\end{figure}
\end{frame}

\subsection{Weak Formulations}
\begin{frame}{Admissible Displacements (Nonpenetration Constraint)}
	\begin{itemize}
		\item Homogeneous Dirichlet boundary conditions are incorporated in the underlying Hilbert space
		      \begin{equation*}
			      V:=\left\{ v\in H^1(\Omega;\R^d)\;\middle|\; (\gamma v)|_{\Gamma_D}=0\right\}\subset H^1(\Omega;\R^d).
		      \end{equation*}
		\item Nonpenetration constraint: $u_\mathrm{n}:= (\gamma u)\cdot n\le g = 0$ almost everywhere on $\Gamma_C$.
		\item Set of admissible displacements:
		      \begin{equation*}
			      K := \left\{ v\in V\;\middle|\; v_{\mathrm n}\le 0 \text{ a.e.\ on }\Gamma_C\right\}\subset V.
		      \end{equation*}
		\item $K$ is closed and convex in $V$.
		\item $K$ is a cone, i.e.\ if $v\in K$ then $\lambda v\in K$ for all $\lambda\in [0,\infty)$.
	\end{itemize}
\end{frame}

\begin{frame}{Small Strain Elasticity}
	\begin{itemize}
		\item {\em Small strain tensor}:
		      \begin{equation*}
			      \varepsilon:H^1(\Omega;\R^d)\to L^2(\Omega;\mathrm{Sym}^2(\R^d)),\qquad u\mapsto \varepsilon(u):= \frac{1}{2}\left(\nabla u + \nabla u^\top\right).
		      \end{equation*}
		\item {\em Elasticity tensor field} $\cA:\Omega\to\cL(\mathrm{Sym}^2(\R^d);\mathrm{Sym}^2(\R^d))$ is of {\em fourth order} and describes {\em elasticity} of the material from which $\Omega$ is made. (Heterogeneous, depends on $x\in\Omega$)
		\item $\cA$ is assumed symmetric in the sense that
		      \begin{equation*}
			      \cA_{ijkl}(x)=\cA_{jikl}(x)=\cA_{ijlk}(x)=\cA_{klij}(x),
			      \qquad \forall\,1\le i,j,k,l\le d,
		      \end{equation*}
		\item $\cA$ is assumed uniformly elliptic, i.e.\ $\exists\,\alpha_{\cA}>0$ such that for a.e. $x\in\Omega$
		      \begin{equation*}
			      s:\cA(x):s \ge \alpha_\cA\, (s:s)\qquad \forall\,s\in \mathrm{Sym}^2(\R^d).
		      \end{equation*}
		\item $\cA$ is assumed bounded, i.e.\ $\exists\,C_{\cA}>0$ such that for a.e. $x\in\Omega$
		      \begin{equation*}
			      |\cA_{ijkl}(x)|\le C_\cA \qquad \forall\,1\le i,j,k,l\le d.
		      \end{equation*}
		\item {\em (Cauchy) stress tensor}:
		      \begin{equation*}
			      \sigma: H^1(\Omega;\R^d)\to L^2(\Omega;\mathrm{Sym}^2(\R^d)),\qquad u\mapsto \cA : \varepsilon(u).
		      \end{equation*}
	\end{itemize}
\end{frame}

\begin{frame}{Virtual Work: External and Internal Forces}
	\begin{itemize}
		\item Let $f\in L^2(\Omega;\R^d)$ be given {\em volume force density} and $F\in L^2(\Gamma_N;\R^d)$ describe the {\em surface load} which we impose on $\Gamma_N$.
		\item Work = force $\cdot$ displacement. Thus
		      \begin{equation*}
			      \ell(v)
			      :=
			      \int_\Omega f\cdot v\;\dx
			      +
			      \int_{\Gamma_N} F\cdot (\gamma v)\;\ds,\qquad v\in V
		      \end{equation*}
		      describes the {\em virtual work of external forces}.
		\item {\em Internal forces} are described by the stress $\sigma(u) = \cA:\varphi(u)$.
		\item The {\em virtual work of internal forces} is thus
		      \begin{equation*}
			      a(u,v)
			      :=
			      \int_\Omega \sigma(u):\varepsilon(v)\;\dx,\qquad u,v\in V.
		      \end{equation*}
	\end{itemize}
\end{frame}

\begin{frame}{Standard Properties of $a$ and $\ell$}
	\begin{proposition}
		The symmetric bilinear form $a:V\times V\to \R$ defined above is
		\begin{itemize}
			\item[i)] {\em bounded}, i.e.\ for all $v,w\in V$ there exists $C_1<\infty$ such that
			      \begin{equation*}
				      |a(u,v)|\le C_1\,\|u\|_{1,\Omega}\,\|v\|_{1,\Omega}\qquad \text{with}\qquad C_1 = C_\cA,
			      \end{equation*}
			\item[ii)] $V${\em -elliptic}, i.e.\ for all $v\in V\setminus\{0\}$ there exist $C_2>0$ such that
			      \begin{equation*}
				      a(v,v)\ge C_2\,\|v\|_{1,\Omega}^2\qquad\text{with}\qquad C_2 = \frac{\alpha_\cA}{C_K^2}.
			      \end{equation*}
		\end{itemize}
		Moreover, the linear functional $\ell$ defined above is {\em bounded}, i.e.\ for all $v\in V$ there exists $C_3>0$ such that
		\begin{equation*}
			|\ell (v)|\le C_3\,\|v\|_{1,\Omega}\qquad\text{with}\qquad C_3 = \max\{1,C_\mathrm{tr}\}\left(\|f\|_{0,\Omega}+\|F\|_{0,\Gamma_N}\right)
		\end{equation*}
	\end{proposition}
\end{frame}

\begin{frame}{Energy Minimization}
	\begin{itemize}
		\item The {\em potential energy} of the elastic body is given by the quadratic functional
		      \begin{equation*}
			      \cJ(v):= \frac{1}{2}a(v,v)-\ell(v),\qquad v\in V.
		      \end{equation*}
		\item Restricted to $K\subset V$ due to nonpenetration.
	\end{itemize}
	\begin{problem}[Signorini's Problem -- Energy Minimization]
	Find $u\in K$ such that
	\begin{equation*}
		\cJ(u)=\min_{v\in K}\cJ(v) = \min_{v\in K}\,\frac{1}{2}\,a(v,v)-\ell(v).
	\end{equation*}
	\end{problem}
\end{frame}

\begin{frame}{Existence of Minimizer}
	\begin{lemma}[Weak lower semicontinuity of the norm]
		Let $(X,\|\cdot\|_X)$ be a normed space. Then $\|\cdot\|_X:X\to \R_{\ge 0}$ is a weakly lower semicontinuous functional, i.e.\ if $x_j\rightharpoonup x$ in $X$, then
		\begin{equation*}
			\|x\|_X \le \liminf_{j\to\infty}\|x_j\|_X.
		\end{equation*}
	\end{lemma}
	\textbf{\em Proof:} Omitted here.
\end{frame}

\begin{frame}{Existence of Minimizer}
	\begin{lemma}[$\cJ$ is coercive and w.l.s.c.]
		The functional $\cJ$ is
		\begin{itemize}
			\item[i)]{\em coercive} on $V$, i.e.
			      \begin{equation*}
				      \cJ(v)\to\infty\qquad\text{as}\qquad\|v\|_{1,\Omega}\to\infty,
			      \end{equation*}
			\item[ii)] {\em weakly lower semicontinuous} on $V$, i.e.\ if $v_j\rightharpoonup v$ in $V$, then
			      \begin{equation*}
				      \cJ(v)\le \liminf_{j\to\infty}\cJ(v_j).
			      \end{equation*}
		\end{itemize}
	\end{lemma}
	\textbf{\em Proof:}

	\begin{itemize}
		\item By boundedness of $\ell$ we have $|\ell(v)|\le \|\ell\|_{V^\prime}\|v\|_{1,\Omega}$. Hence
		      \begin{equation*}
			      \cJ(v)=\frac{1}{2}\,a(v,v)-\ell(v)
			      \ge \frac{\alpha}{2}\,\|v\|_{1,\Omega}^2-\|\ell\|_{V^\prime}\,\|v\|_{1,\Omega}
			      \xrightarrow{ \ \|v\|_{1,\Omega}\to\infty \ }\infty.
		      \end{equation*}
		\item This proves i).
	\end{itemize}
\end{frame}

\begin{frame}{Existence of Minimizer}
	\begin{itemize}
		\item For ii) we set
		      \begin{equation*}
			      \|s\|_{\cA}^2:= \int_\Omega s:\cA :s \;\dx,\qquad s\in L^2(\Omega;\mathrm{Sym}^2(\R^d)).
		      \end{equation*}
		\item This defines a norm on $L^2(\Omega;\mathrm{Sym}^2(\R^d))$ by uniform ellipticity of $\cA$.
		\item We have $a(v,v) = \|\varepsilon(v)\|_{\cA}^2$ and $\varepsilon$ is linear and bounded on $V$, i.e.\ if $v_j\hookrightarrow v$ in $V$, then $\varepsilon(v_j)\hookrightarrow\varepsilon(v)$ in $L^2(\Omega;\mathrm{Sym}^2(\R^d))$.
		\item Since norms are w.l.s.c., we have
		      \begin{equation*}
			      a(v,v)=\|\varepsilon(v)\|_\cA^2
			      \le \liminf_{j\to\infty}\|\varepsilon(v_j)\|_\cA^2
			      = \liminf_{j\to\infty} a(v_j,v_j).
		      \end{equation*}
		\item $\ell\in V^\prime$ and thus $\ell(v_j)\to\ell(v)$.
		\item Therefore,
		      \begin{equation*}
			      \cJ(v)=\frac{1}{2}\,a(v,v)-\ell(v)
			      \le \frac{1}{2}\,\liminf_{j\to\infty} a(v_j,v_j) - \lim_{j\to\infty}\ell(v_j)
			      = \liminf_{j\to\infty}\cJ(v_j).
		      \end{equation*}\qed
	\end{itemize}
\end{frame}

\begin{frame}{Existence of Minimizer}
	\begin{proposition}[Existence of Minimizer of $\cJ$]
		The functional $\cJ$ has at least one minimizer $u\in K$.
	\end{proposition}
	\textbf{\em Proof:}
	\begin{itemize}
		\item Set
		      \begin{equation*}
			      M := \inf_{v\in K}\cJ(v)\in[-\infty,\infty).
		      \end{equation*}
		\item Since $0\in K$ we have $M\le \cJ(0)=0$, hence $M$ is finite from above.
		\item Let $\{v_j\}_{j\in\N}\subset K$ such that $\cJ(v_j)\to M$ as $j\to\infty$.
		\item $\{v_j\}_{j\in\N}\subset K$ is bounded since $\cJ$ is coercive.
		\item Since $V$ is Hilbert space there exists subsequence $\{v_{j_k}\}_{k\in\N}$ and some $u\in V$ such that $v_{j_k} \rightharpoonup u$ in $V$.
		\item $u\in K$ since $K$ is (weakly) closed.
		\item By w.l.s.c. of $\cJ$, $u\in K$ is minimizer of $\cJ$ because
		      \begin{equation*}
			      \cJ(u)\le \liminf_{k\to\infty}\cJ(v_{j_k})=M=\inf_{v\in K}\cJ(v).
		      \end{equation*}\eop
	\end{itemize}
\end{frame}
\begin{frame}{Uniqueness of Minimizer}
	\begin{proposition}[Uniqueness of a Minimizer of $\cJ$]
		If there exists a minimizer of the functional $\cJ$, it is unique.
	\end{proposition}
	\textbf{\em Proof: } $\cJ$ is strictly convex on $V$ and thus on the convex set $K$. Thus, a minimizer $u\in K$ is unique.\eop
\end{frame}

\begin{frame}{A-Priori Stability Bound}
	\begin{proposition}[Stability estimate for the minimizer]
		The Unique minimizer $u\in K$ of $\cJ$ satisfies the a-priori bound
		\begin{equation*}
			\|u\|_{1,\Omega} \le \frac{2\,C_K^2\,\max\{1,C_\mathrm{tr}\}}{\alpha_\cA}\,\left(\|f\|_{0,\Omega} +\|F\|_{0,\Gamma_N}\right).
		\end{equation*}
	\end{proposition}
	\textbf{\em Proof:}
	\begin{itemize}
		\item Since $u\in K$ is the unique minimizer, we have $\cJ(u)\le\cJ(0) = 0$ and thus $\tfrac12a(u,u)\le \ell(u)$.
		\item For $u = 0$: trivial. Thus, assume $u\neq 0$.
		\item Recall: $a$ is $V$-elliptic with $a(u,u)\ge \alpha_\cA/C_K^2\,\|u\|_{1,\Omega}^2$ and $\ell$ is bounded with
		      \begin{equation*}
			      |\ell(u)|\le  \max\{1,C_{\mathrm{tr}}\}\left(\|f\|_{0,\Omega} +\|F\|_{0,\Gamma_N}\right).
		      \end{equation*}
		\item Finally,
		      \begin{equation*}
			      \frac{\alpha_\cA}{2\,C_K^2}\,\|u\|_{1,\Omega}^2\le \frac{1}{2}\,a(u,u)\le |\ell(u)|\le  \max\{1,C_{\mathrm{tr}}\}\left(\|f\|_{0,\Omega} +\|F\|_{0,\Gamma_N}\right)\|u\|_{1,\Omega}.
		      \end{equation*}\eop
	\end{itemize}
\end{frame}

\begin{frame}{Variational Inequality as an Optimality Condition}
	\begin{itemize}
		\item Fréchet Derivative of $\cJ$ at $u\in V$ in direction $h\in V$ is $\cJ^\prime(u)[h] = a(u,h)-\ell(h)$.
	\end{itemize}
	\begin{proposition}[First-order optimality condition on convex sets]
		A function $u\in K$ minimizes $\cJ$ over $K$ if and only if it satisfies the variational inequality
		\begin{equation*}
			\cJ^\prime(u)[v-u]\ge 0\qquad\forall\, v\in K.
		\end{equation*}
	\end{proposition}
	\textbf{\em Proof:}
	\begin{itemize}
		\item Let $u\in K$ minimize $\cJ$ over $K$ and let $v\in K$.
		\item $K$ is convex and thus $u+t(v-u)\in K$ for all $t\in[0,1]$.
		\item $\phi(t):= \cJ(u+t(v-u))$ attains minimum at $t = 0$, i.e.\ $\phi^\prime(0^+)\ge 0$.
		\item By chainrule, $\phi'(0^+)=\cJ'(u)[v-u]\ge 0$.
	\end{itemize}
\end{frame}
\begin{frame}{Variational Inequality as an Optimality Condition}
	\begin{itemize}
		\item Let $u\in K$ such that  $\cJ^\prime(u)[v-u]\ge 0$ for all $v\in K$ and set $h:= v-u$.
		\item Taylor-expansion:
		      \begin{equation*}
			      \cJ(u+h)-\cJ(u)=\underbrace{\cJ'(u)[h]}_{\ge 0}+\underbrace{\frac12\,a(h,h)}_{\ge 0} \ge 0.
		      \end{equation*}
		\item Thus, $\cJ(u+h)-\cJ(u) = \cJ(v)-\cJ(v)\ge 0$ for all $v\in K$.
		\item $u\in K$ minimizes $\cJ$ over $K$.\eop
	\end{itemize}
\end{frame}

\begin{frame}{Weak Formulations / Variational Inequality}
	\begin{problem}[Signorini's Problem -- Weak Formulation]
	Find $u\in K$ such that
	\begin{equation*}
		a(u,v-u)\ge \ell(v-u)\qquad \forall\, v\in K.
	\end{equation*}
	\end{problem}

	\begin{problem}[Signorini's Problem -- Weak Formulation]
	Find $u\in K$ such that
	\begin{equation*}
		\begin{cases}
			a(u,u) = \ell(u), &                   \\[2pt]
			a(u,v)\ge \ell(v) & \forall\, v\in K.
		\end{cases}
	\end{equation*}
	\end{problem}
\end{frame}

\begin{frame}{Another Stability Estimate}
	\begin{proposition}[Lipschitz Dependence on the Data]
		For $i\in\{1,2\}$, let $f_i\in L^2(\Omega;\mathbb R^d)$ and $F_i\in L^2(\Gamma_N;\mathbb R^d)$, and let $\ell_i\in V'$ be the corresponding load functional, i.e.
		\begin{equation*}
			\ell_i(v):=\int_\Omega f_i\cdot v\;\dx+\int_{\Gamma_N} F_i\cdot \gamma v\;\ds,
			\qquad v\in V.
		\end{equation*}
		Moreover, let $u_i\in K$ be the unique solution of one of the weak formulations above. Then the solution depends Lipschitz continuously on the load functional, i.e.
		\begin{equation*}
			\|u_1-u_2\|_{1,\Omega}\le \frac{C_K^2}{\alpha_\mathcal{A}}\,\|\ell_1-\ell_2\|_{V'}.
		\end{equation*}
		In particular, it depends Lipschitz continuously on the data $f_i,F_i$ in the sense that
		\begin{equation*}
			\|u_1-u_2\|_{1,\Omega}
			\le \frac{C_K^2\,\max\{1,C_\mathrm{tr}\}}{\alpha_\mathcal{A}}
			\Bigl(\|f_1-f_2\|_{0,\Omega}+\|F_1-F_2\|_{0,\Gamma_N}\Bigr).
		\end{equation*}
	\end{proposition}
\end{frame}

\subsection{Strong Formulations}

\begin{frame}{Stress Regularity}
	\begin{itemize}
		\item We need equilibrium equation $-\diver \sigma(u) = f$ in $L^2(\Omega;\R^d)$.
		\item In weak formulations $u\in K\subset H^1(\Omega;\R^d)$ and thus
		      \begin{equation*}
			      \varepsilon(u)\in L^2(\Omega;\mathrm{Sym}^2(\R^d))
			      \qquad\text{and}\qquad
			      \sigma(u)=\mathcal A:\varepsilon(u)\in L^2(\Omega;\mathrm{Sym}^2(\R^d)).
		      \end{equation*}
		\item This is enough for well-definedness of $a(u,v)$ but $\diver\sigma(u)\notin L^2(\Omega;\R^d)$ in general.
		\item Thus, we assume $\sigma(u)\in H(\diver;\Omega)$.
		\item Boundary traction $\sigma(u)\mathrm{n}:= \gamma_{\mathrm{n}}(\sigma(u))\in H^{-1/2}(\Gamma;\R^d)$.
		\item Since $\sigma(u)$ symmetric, we have
		      \begin{equation*}
			      \sigma(u):\nabla v = \sigma(u):\varepsilon(v)\qquad \forall\, v\in H^1(\Omega;\R^d).
		      \end{equation*}
		\item Therefore Green's formula is
		      \begin{equation*}
			      \langle \sigma(u)\mathrm n,\gamma v\rangle_\Gamma
			      =
			      \int_\Omega \sigma(u):\varepsilon(v)\;\dx
			      +
			      \int_\Omega \diver\sigma(u)\cdot v\;\dx
			      \qquad
			      \forall\, v\in H^1(\Omega;\R^d).
		      \end{equation*}
	\end{itemize}
\end{frame}

\begin{frame}{A Hybrid Formulation}
	\begin{problem}[Signorini's Problem -- Hybrid Formulation]
	For given $f\in L^2(\Omega;\R^d)$ and $F\in L^2(\Gamma_N;\R^d)$, find $u\in K$ such that $\sigma(u)\in H(\diver;\Omega)$ and
	\begin{subequations}\label{eq:hybrid}
		\begin{empheq}[left=\empheqlbrace]{alignat=2}
			-\diver\sigma(u) &= f
			&\qquad& \text{a.e.\ in }\Omega, \label{eq:hybrid_bulk}\\
			\langle \sigma(u)\mathrm n,\gamma u\rangle_\Gamma
			&= \mathrlap{\int_{\Gamma_N} F\cdot \gamma u\;\ds,}
			&& \label{eq:hybrid_eq}\\
			\langle \sigma(u)\mathrm n,\gamma v\rangle_\Gamma
			&\ge \int_{\Gamma_N} F\cdot \gamma v\;\ds
			&\qquad& \forall\, v\in K.
			\label{eq:hybrid_ineq}
		\end{empheq}
	\end{subequations}
	\end{problem}
\end{frame}

\begin{frame}{Equivalence of Hybrid and Weak Formulation}
	\begin{proposition}[Equivalence of Hybrid and Weak Formulation]
		Assume $\sigma(u)\in H(\diver;\Omega)$. Then $u\in K$ solves the hybrid problem above if and only if it solves the weak problem
		\begin{equation*}
			\text{Find }u\in K\text{ such that}\qquad a(u,v-u)\ge\ell(v-u)\qquad \forall v\in K.
		\end{equation*}
	\end{proposition}
\end{frame}


\begin{frame}{Key Identity for Proof}
	\begin{itemize}
		\item Assume $\sigma(u)\in H(\diver;\Omega)$ and $-\diver\sigma(u)=f$ in $L^2(\Omega;\R^d)$.
		\item Green's formula (previous slide) gives for all $w\in V$:
		      \begin{equation*}
			      \langle \sigma(u)\mathrm n,\gamma w\rangle_\Gamma
			      =
			      a(u,w)
			      +
			      \int_\Omega \diver\sigma(u)\cdot w\;\dx
			      =
			      a(u,w)-\int_\Omega f\cdot w\;\dx.
		      \end{equation*}
		\item Hence for all $w\in V$:
		      \begin{equation*}
			      a(u,w)-\ell(w)
			      =
			      \langle \sigma(u)\mathrm n,\gamma w\rangle_\Gamma
			      -
			      \int_{\Gamma_N}F\cdot \gamma w\;\ds.
		      \end{equation*}
	\end{itemize}
\end{frame}

\begin{frame}{Proof: Hybrid $\Rightarrow$ Weak}
	\begin{itemize}
		\item Subtract \eqref{eq:hybrid_eq} from \eqref{eq:hybrid_ineq}:
		      \begin{equation*}
			      \langle \sigma(u)\mathrm n,\gamma (v-u)\rangle_\Gamma
			      \ge
			      \int_{\Gamma_N}F\cdot \gamma (v-u)\;\ds.
		      \end{equation*}
		\item Insert the key identity from the previous slide:
		      \begin{equation*}
			      a(u,v-u)-\ell(v-u)\ge 0
			      \qquad\forall\,v\in K,
		      \end{equation*}
		      i.e.\ the weak formulation.
	\end{itemize}
\end{frame}

\begin{frame}{Proof: Weak $\Rightarrow$ Hybrid (Bulk Equation)}
	\begin{itemize}
		\item Let $u\in K$ solve the Hybrid Problem above and assume $\sigma(u)\in H(\diver;\Omega)$.
		\item Let $\varphi\in \cD(\Omega;\R^d)$. Then $u\pm t\varphi\in K$ for $|t|\ll 1$, hence the weak inequality implies
		      \begin{equation*}
			      a(u,\varphi)=\ell(\varphi)=\int_\Omega f\cdot \varphi\;\dx = \langle f,\,\varphi\rangle.
		      \end{equation*}
		\item Since $\sigma(u)$ is symmetric, the left hand side is
		      \begin{equation*}
			      a(u,\varphi) = \int_\Omega\sigma(u):\varepsilon(\varphi) = \int_\Omega\sigma(u):\nabla\varphi\;\dx = \langle -\diver\sigma(u),\,\varphi\rangle.
		      \end{equation*}
		\item Thus, $-\diver\sigma(u) = f$ in $\cD^\prime(\Omega;\R^d)$ and by $\sigma(u)\in H(\diver;\Omega)$ we have
		      \begin{equation*}
			      -\diver\sigma(u) = f\qquad \text{a.e. in }\Omega.
		      \end{equation*}
	\end{itemize}
\end{frame}

\begin{frame}{Proof: Weak $\Rightarrow$ Hybrid (Hybrid Inequality)}
	\begin{itemize}
		\item Since $K$ is a cone, we have $0\in K$ and $2u\in K$.
		\item Testing both in the weak inequality yields
		      \begin{equation*}
			      a(u,u)\le \ell(u)\quad\text{and}\quad a(u,u)\ge \ell(u)
			      \ \Rightarrow\ a(u,u)=\ell(u).
		      \end{equation*}
		\item With the key identity this is exactly \eqref{eq:hybrid_eq} since
		      \begin{equation*}
			      0 = a(u,u)-\ell(u) = \langle\sigma(u)\mathrm{n},\,\gamma u\rangle_\Gamma -\int_{\Gamma_N}F\cdot\gamma u\;\ds.
		      \end{equation*}
	\end{itemize}
\end{frame}

\begin{frame}{Proof: Weak $\Rightarrow$ Hybrid (Hybrid Equality)}
	\begin{itemize}
		\item Let $v\in K$. We have
		      \begin{equation*}
			      a(u,v-u)\ge\ell(v-u) = \int_\Omega f\cdot(v-u)\;\dx + \int_{\Gamma_N}F\cdot\gamma(v-u)\;\ds.
		      \end{equation*}
		\item On the other hand, by the key identity, we have
		      \begin{equation*}
			      a(u,v-u) = \langle \sigma(u)\mathrm{n},\,\gamma(v-u)\rangle_\Gamma + \int_\Omega f\cdot(v-u)\;\dx.
		      \end{equation*}
		\item Subtracting volume term yields
		      \begin{equation*}
			      \langle \sigma(u)\mathrm n,\gamma(v-u)\rangle_\Gamma
			      \ge
			      \int_{\Gamma_N}F\cdot\gamma(v-u)\;\ds.
		      \end{equation*}
		\item This means
		      \begin{equation*}
			      \langle\sigma(u)\mathrm{n}\,\gamma v\rangle_\Gamma\ge \int_{\Gamma_N}F\cdot \gamma v\;\ds + \underbrace{\left(\langle\sigma(u)\mathrm{n}\,\gamma u\rangle_\Gamma-\int_{\Gamma_N}F\cdot\gamma u\;\ds\right)}_{ = 0}.
		      \end{equation*}\eop
	\end{itemize}
\end{frame}

\begin{frame}{Normal/Tangential Splitting and Contact Trace Spaces}
	\begin{itemize}
		\item For {\em stronger} boundary conditions, we need to split functions into their {\em normal} and {\em tangential} component.
		\item Contact trace space:
		      \begin{equation*}
			      W_C := \left\{\gamma v|_{\Gamma_C}\; \middle|\; v\in V\right\}\subset H^{1/2}(\Gamma_C;\R^d).
		      \end{equation*}
		\item For $w\in W_C$, we define its {\em normal} and {\em tangential} component by
		      \begin{equation*}
			      w_\mathrm{n} := w\cdot \mathrm n
			      \qquad\text{and}\qquad
			      w_t := w - (w\cdot \mathrm n)\mathrm n.
		      \end{equation*}
		\item Normal and tangential trace spaces:
		      \begin{equation*}
			      W_{C,\mathrm{n}} := \left\{w\cdot \mathrm n\; \middle|\; w\in W_C\right\}
			      \qquad\text{and}\qquad
			      W_{C,t} := \left\{w\in W_C\; \middle|\; w\cdot \mathrm n = 0\right\}.
		      \end{equation*}
		\item Every $w\in W_C$ admits the decomposition
		      \begin{equation*}
			      w = w_\mathrm{n}\mathrm n + w_t\in W_{C,\mathrm{n}}\,\mathrm{n}\oplus W_{C,t}
		      \end{equation*}
	\end{itemize}
\end{frame}

\begin{frame}{Normal/Tangential Splitting and Contact Trace Spaces}
	\begin{itemize}
		\item Dual Spaces $W^\prime_C,\, W_{C,\mathrm{n}},\,W_{C,t}$ and duality pairing $\langle\cdot,\,\cdot\rangle_{\Gamma_C}$.
		\item Nonpenetration is incorporated in the space
		      \begin{equation*}
			      W_{C,\mathrm{n}}^-
			      :=
			      \left\{\eta\in W_{C,\mathrm{n}}\; \middle|\; \eta\le 0 \text{ a.e.\ on }\Gamma_C\right\}
		      \end{equation*}
		      of admissible normal traces.
		\item Dual cone of weakly nonpositive normal contact forces:
		      \begin{equation*}
			      \Lambda_C
			      :=
			      \left\{\lambda\in W_{C,\mathrm{n}}'\; \middle|\; \langle \lambda,\eta\rangle_{\Gamma_C} \ge 0 \quad \forall\,\eta\in W_{C,\mathrm{n}}^-\right\}.
		      \end{equation*}
	\end{itemize}
\end{frame}

\begin{frame}{Karush-Kuhn-Tucker Contact Conditions}
	\begin{itemize}
		\item We denote $\sigma(u)\mathrm{n} = \sigma_\mathrm{n}(u)\mathrm{n}+\sigma_t(u)\in W^\prime_{C,\mathrm{n}}\oplus W^\prime_{C,t}$.
	\end{itemize}
	\begin{problem}[Signorini's Problem -- Strong Formulation]
	For given $f\in L^2(\Omega;\R^d)$, find $u\in V$ such that
	\begin{subequations}\label{eq:strong}
		\begin{empheq}[left=\empheqlbrace]{alignat=2}
			-\diver \sigma(u) &= f
			&\qquad& \text{a.e.\ in }\Omega, \label{eq:strong_bulk}\\
			\sigma_t(u) &= 0
			&\qquad& \text{in }W_{C,t}^\prime, \label{eq:strong_tangential}\\
			u_\mathrm{n} &\le 0
			&\qquad& \text{a.e.\ on }\Gamma_C, \label{eq:strong_nonpen}\\
			\sigma_\mathrm{n}(u) &\in \Lambda_C,
			&& \label{eq:strong_lambda}\\
			\langle\sigma_\mathrm{n}(u),\, u_\mathrm{n}\rangle_{\Gamma_C} &= 0.
			&& \label{eq:strong_comp}
		\end{empheq}
	\end{subequations}
	\end{problem}
	\begin{itemize}
		\item Assuming $|\Gamma_N|= 0$, this is equivalent to previous formulations.
	\end{itemize}
\end{frame}
