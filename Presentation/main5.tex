\section{Introduction -- What's the Problem?}
\begin{frame}{Rough Problem Setup}
	\begin{itemize}
		\item {\em Body: } $\Omega\subset\R^d,\, d\in\{2,3\}$ open, bounded, connected with Lipschitz boundary $\Gamma:= \partial\Omega$.
		\item $\Omega$ is sitting on a {\em rigid foundation}.
		\item No frictions assumed between $\Omega$ and foundation.
		\item $\Omega$ may be subject to elasticity properties and internal/external forces.
		\item {\bf Not in equilibrium configuration}.
	\end{itemize}
	\begin{figure}[!h]
		\centering
		\input{Figures/setup_plain_initial.tex}
	\end{figure}
\end{frame}

\begin{frame}{What is to be found?}
	\begin{itemize}
		\item Let $\varphi(\Omega)$ denote $\Omega$ in equilibrium configuration.
		\item $\varphi:\overline{\Omega}\to\R^d$ maps $\overline{\Omega}$ onto $\varphi(\overline{\Omega})$ and can be written as
		      \begin{equation*}
			      \varphi(x) = x+ u(x),\qquad u:\overline{\Omega}\to\R^d.
		      \end{equation*}
		\item {\bf We aim to find the {\em displacement function}} $u$.
	\end{itemize}

\end{frame}

\section{Geometrical and Physical Setup}

\subsection{Boundary Decomposition}

\begin{frame}{Boundary Decomposition and Preview on the Boundary Conditions}
	\begin{itemize}
		\item Decompose $\Gamma = \partial\Omega$ into three relatively open and pairwise disjoint parts $\Gamma = \overline{\Gamma_D}\,\mathbin{\dot\cup}\,\overline{\Gamma_N}\,\mathbin{\dot\cup}\,\overline{\Gamma_C}$ with $|\Gamma_D|,|\Gamma_C|>0$ and $|\Gamma_N|\ge 0$.
		\item Each part has different boundary conditions:
		      \begin{itemize}
			      \item $\Gamma_D$: Homogeneous Dirichlet. (Clamped boundary part).
			      \item $\Gamma_N$: Neumann conditions. (We impose known surface force density).
			      \item $\Gamma_C$ is the {\em potential} contact boundary, i.e. $\Gamma_C$ is prescribed a priori but the active contact zone $\Gamma_{\mathrm{act}}(u)\subset\Gamma_C$ is unknown. On $\Gamma_C$: Contact conditions. (Nonpenetration, Complementarity, ...)
		      \end{itemize}
		\item Rigid foundation can be describe by gap-function $g:\Gamma_C\to\R$. We assume $g = 0$.
	\end{itemize}
	\begin{figure}[!h]
		\centering
		\input{Figures/setup.tex}
	\end{figure}
\end{frame}

\subsection{Elasticity and Forces}

\begin{frame}{Basic Notations on Sobolev Spaces}
	\begin{itemize}
		\item On $H^s(\Omega;\R^d)$ for $s =k+\theta$ with $k\in \N_0$ and $\theta\in[0,1)$, we use $(\cdot,\,\cdot)_{s,\Omega}$ and $\|\cdot\|_{s,\Omega}$
		\item $H^0(\Omega;\R^d) = L^2(\Omega;\R^d)$.
		\item {\em Trace Theorem}: Restriction of smooth functions $\overline{\Omega}\to\R^d$ onto the boundary $\Gamma = \partial\Omega\in\cC^{m,1}$ extends uniquely to a continuous surjective linear trace operator
		      \begin{equation*}
			      \gamma: H^s(\Omega;\R^d)\to H^{s-1/2}(\Gamma;\R^d),\qquad \frac{1}{2}<s\le m+1.
		      \end{equation*}
		\item For now, $\Omega$ is Lipschitz, i.e.\ $m=0$ and we need $s = 1$, so
		      \begin{equation*}
			      \gamma: H^1(\Omega;\R^d)\to H^{1/2}(\Gamma;\R^d)
		      \end{equation*}
	\end{itemize}
\end{frame}

\begin{frame}{Small Strain Elasticity}
	\begin{itemize}
		\item {\em Symmetric second order tensors}:
		      \begin{equation*}
			      \S^d:=\left\{s\in\R^d\otimes\R^d\;\middle|\;s^\top = s\right\}
		      \end{equation*}
		\item {\em Stresses} are described by fields $\tau:\Omega\to\S^d$
		\item We identify $\S^d$ with symmetric $d\times d$ matrices.
		\item On $\R^{d\times d}$, we use the {\em Frobenius product}
		      \begin{equation*}
			      A:B := \sum_{j=1}^d\sum_{k=1}^d A_{jk}B_{jk},
			      \qquad
			      |A| := \sqrt{A:A}.
		      \end{equation*}
		\item If $\cA:\Omega\to\cL(\S^d;\S^d)$ is a fourth order tensor and $s\in\S^d$, we use the {\em double contraction}
		      \begin{equation*}
			      s:\cA(x):s:=\sum_{i = 1}^d \sum_{j = 1}^d \sum_{k = 1}^d \sum_{l = 1}^d\cA_{ijkl}(x)s_{ij}s_{kl}.
		      \end{equation*}
	\end{itemize}
\end{frame}

\begin{frame}{Small Strain Elasticity}
	\begin{itemize}
		\item Let $s\ge 1$.
		\item {\em Small strain tensor}:
		      \begin{equation*}
			      \varepsilon:H^s(\Omega;\R^d)\to H^{s-1}(\Omega;\S^d),\qquad u\mapsto \varepsilon(u):= \frac{1}{2}\left(\nabla u + \nabla u^\top\right).
		      \end{equation*}

		\item {\em Elasticity} of $\Omega$ is describes by a {\em fourth order tensor} $\cA:\Omega\to\cL(\S^d;\S^d)$.
		\item $\cA$ is assumed {\em symmetric} and {\em uniformly elliptic} and {\em bounded}.
		\item {\em (Cauchy) stress tensor}:
		      \begin{equation*}
			      \sigma: H^s(\Omega;\R^d)\to H^{s-1}(\Omega;\S^d),\qquad u\mapsto \cA : \varepsilon(u).
		      \end{equation*}
		\item $\cA$ has to be {\em sufficiently smooth} for this.
		\item For now, we need $s = 1$, i.e.\ $\varepsilon(u)\in L^2(\Omega;\S^d)$ and $\sigma(u)\in L^2(\Omega;\S^d)$.
		\item $\cA_{ijkl}\in L^\infty(\Omega)$ is sufficient for $s = 1$. Later on, we need more.
	\end{itemize}
\end{frame}

\begin{frame}{Virtual Work of Internal and External Forces}
	\begin{itemize}
		\item {\em Volume force density}: $\ f\in L^2(\Omega;\R^d)$.
		\item {\em Surface load} on Neumann part: $\ F\in L^2(\Gamma_N;\R^d)$.
		\item {\em Work = force $\cdot$ displacement}: The linear functional
		      \begin{equation*}
			      \ell(v)
			      :=
			      \int_\Omega f\cdot v\;\dx
			      +
			      \int_{\Gamma_N} F\cdot (\gamma v)\;\ds,\qquad v\in H^1(\Omega;\R^d)
		      \end{equation*}
		      describes the {\em virtual work of external forces}.
		\item {\em Internal forces} are described by the {\em stress} $\sigma(u) = \cA:\varepsilon(u)$.
		\item {\em Work = force $\cdot$ displacement}: The symmetric bilinear form
		      \begin{equation*}
			      a(u,v)
			      :=
			      \int_\Omega \sigma(u):\varepsilon(v)\;\dx,\qquad u,v\in H^1(\Omega;\R^d).
		      \end{equation*}
		      describes the {\em virtual work of internal forces}.
		\item {\em Equilibrium} between both: $\ a(u,v) = \ell(v)\qquad\forall\,v\in H^1(\Omega;\R^d)$.
	\end{itemize}
\end{frame}

\begin{frame}{Standard Properties of $a$ and $\ell$}
	\begin{proposition}
		The symmetric bilinear form $a:H^1(\Omega;\R^d)\times H^1(\Omega;\R^d)\to \R$ defined above is
		\begin{itemize}
			\item[i)] {\em bounded}, i.e.\ for all $v,w\in H^1(\Omega;\R^d)$ there exists $C_1<\infty$ such that
			      \begin{equation*}
				      |a(u,v)|\le C_1\,\|u\|_{1,\Omega}\,\|v\|_{1,\Omega}.
			      \end{equation*}
			\item[ii)] $H^1(\Omega;\R^d)${\em -elliptic}, i.e.\ for all $v\in H^1(\Omega;\R^d)\setminus\{0\}$ there exist $C_2>0$ such that
			      \begin{equation*}
				      a(v,v)\ge C_2\,\|v\|_{1,\Omega}^2.
			      \end{equation*}
		\end{itemize}
		Moreover, the linear functional $\ell$ defined above is {\em bounded}, i.e.\ for all $v\in H^1(\Omega;\R^d)$ there exists $C_3>0$ such that
		\begin{equation*}
			|\ell (v)|\le C_3\,\|v\|_{1,\Omega}.
		\end{equation*}
	\end{proposition}
\end{frame}

\subsection{Admissible Displacements}

\begin{frame}{Admissible Displacements and the Nonpenetration Constraint}
	\begin{itemize}
		\item {\em Homogeneous Dirichlet} boundary conditions on $\Gamma_D$ are incorporated in the underlying Hilbert space
		      \begin{equation*}
			      V:=\left\{ v\in H^1(\Omega;\R^d)\;\middle|\; (\gamma v)|_{\Gamma_D}=0\right\}
		      \end{equation*}
		\item {\em Normal Component}: For $v\in V$, we define its {\em normal component} by $v_\mathrm{n}:= (\gamma v)\cdot \mathrm{n}\in H^{1/2}(\Gamma)$ with the {\em outward normal} $\mathrm{n}\in L^\infty(\Gamma;\R^d)$.
		\item {\em Nonpenetration constraint}: $u_\mathrm{n}\le g = 0$ almost everywhere on $\Gamma_C$.
		\item Set of {\em admissible displacements}: $K := \left\{ v\in V\;\middle|\; v_{\mathrm n}\le 0 \text{ a.e.\ on }\Gamma_C\right\}$.
		\item $K$ is a closed and convex {\em cone} in $V$, i.e.\ if $v\in K$ then $\lambda v\in K$ for all $\lambda\in [0,\infty)$.
	\end{itemize}
	\begin{figure}[!h]
		\centering
		\input{Figures/setup_initial_normal.tex}
	\end{figure}
\end{frame}

\section{Energy Minimization}

\begin{frame}{Energy Minimization}
	\begin{itemize}
		\item {\em Equilibrium}:
		      \begin{equation*}
			      a(u,v) = \int_\Omega \sigma(u):\varepsilon(v)\;\dx = \int_\Omega f\cdot v\;\dx +\int_{\Gamma_N} F\cdot(\gamma v)\;\ds = \ell(v)\qquad \forall\, v\in V
		      \end{equation*}
		\item The {\em potential energy} of the elastic body is given by the quadratic functional
		      \begin{equation*}
			      \cJ(v):= \frac{1}{2}a(v,v)-\ell(v),\qquad v\in V.
		      \end{equation*}
		\item {\em Nonpenetration}: Minimize $\cJ(v)$ over all {\em admissible} displacements $v\in K$.
	\end{itemize}

	\begin{problem}[Signorini's Problem -- Energy Minimization]
	\begin{equation*}
		\text{Find }u\in K\text{ such that:}\qquad \cJ(u) = \min_{v\in K}\cJ(v).
	\end{equation*}
	\end{problem}
\end{frame}

\begin{frame}{Well-Posedness}
	\begin{proposition}[Well-Posedness of Signorini's Problem]
		There exists exactly one $u\in K$ that minimizes $\cJ$ over $K$. Moreover, the unique minimizer satisfies the a-priori bound
		\begin{equation*}
			\|u\|_{1,\Omega} \le \frac{2}{C_2} \|\ell\|_{V^\prime}.
		\end{equation*}
	\end{proposition}
	\textbf{\em Proof sketch:}
	\begin{itemize}
		\item Prove that $\cJ$ is {\em coercive} and {\em weakly lower semicontinuous}.
		\item This implies existence.
		\item Prove that $\cJ$ is {\em strictly convex}.
		\item With convexity of $K$, this implies uniqueness.
		\item Stability bound follows from $V$-ellipticity of $a$ w.r.t. $C_2$.
	\end{itemize}
\end{frame}

\section{Weak Formulations}

\begin{frame}{Weak Formulations / Variational Inequality}
	\vfill
	\begin{proposition}[First-order optimality condition on convex sets]
		A function $u\in K$ minimizes $\cJ$ over $K$ if and only if it satisfies the variational inequality
		\begin{equation*}
			\cJ^\prime(u)[v-u] =  a(u,v-u)-\ell(v-u) \ge 0\qquad\forall\, v\in K.
		\end{equation*}
	\end{proposition}
	\vfill
	\begin{problem}[Signorini's Problem -- Weak Formulation 1]
	\begin{equation*}
		\text{Find }u\in K\text{ such that:}\qquad a(u,v-u)\ge \ell(v-u)\qquad \forall\, v\in K.
	\end{equation*}
	\end{problem}
	\vfill
	\begin{problem}[Signorini's Problem -- Weak Formulation 2]
	\begin{equation*}
		\text{Find }u\in K\text{ such that:}\qquad
		\begin{cases}
			a(u,u) = \ell(u), &                   \\[2pt]
			a(u,v)\ge \ell(v) & \forall\, v\in K.
		\end{cases}
	\end{equation*}
	\end{problem}
	\vfill
\end{frame}

\section{Strong Formulations}

\subsection{A Hybrid Formulation}

\begin{frame}{Additional Stress Regularity}
	\begin{itemize}
		\item \textbf{\em Goal 1}: For a strong equilibrium equation $-\diver\sigma(u) = f$ in $\Omega$ with $L^2$-source term $f$, we need $\diver\sigma(u)\in L^2(\Omega;\R^d)$.
		\item \textbf{\em Goal 2}: We need a concept of normal traces for the stress $\sigma(u)$ as for displacements: $v_\mathrm{n} = (\gamma v)\cdot\mathrm{n}$.
		\item For displacements, $v\in H^1(\Omega;\R^d)$, a normal trace $(\gamma v)\cdot\mathrm{n}\in H^{1/2}(\Gamma)$ is define by the trace theorem.
		\item For our $u\in V$, we only have $\sigma(u)\in L^2(\Omega;\S^d)$, i.e.\ no trace defined.
		\item Therefore, we need an additional assumption on $\sigma(u)$.
		\item If we assume $\sigma(u)\in H^1(\Omega;\S^d)$, then $(\gamma\sigma(u))\mathrm{n}\in L^2(\Gamma;\R^d)$ is defined.
		\item This requires $u\in H^2(\Omega;\R^d)$.
		\item This assumptions is much stronger than needed!
		\item Instead, we assume $\sigma(u)\in H(\diver;\Omega)$, where
		      \begin{equation*}
			      H(\diver;\Omega):= \left\{\tau\in L^2(\Omega;\S^d)\;\middle|\;\diver\tau\in L^2(\Omega;\R^d)\right\},
		      \end{equation*}
		      where $\diver \tau$ is understood row-wise in the weak sense.
		\item Then $\diver\sigma(u)\in L^2(\Omega;\R^d)$ and we can understand normal traces on $H(\diver;\Omega)$ in the following way.
	\end{itemize}
\end{frame}

\begin{frame}{Normal Traces on $H(\diver;\Omega)$}
	\begin{itemize}
		\item We denote by $H^{-1/2}(\Gamma;\R^d)$ the dual space of $H^{1/2}(\Gamma;\R^d)$ with duality pairing $\langle\cdot,\,\cdot\rangle_\Gamma$.
		\item For smooth $\tau,v$ {\em Green's identity} is
		      \begin{equation*}
			      \int_\Gamma (\gamma\tau)\,\mathrm n \cdot (\gamma v)\;\ds = \underbrace{\int_\Omega \tau:\nabla v\;\dx+\int_\Omega (\diver\tau)\cdot v\;\dx}_{:= F_\tau(v)}.
		      \end{equation*}
		\item This means $(\gamma\tau)\,\mathrm{n}$ is defined in a weak sense via Green's identity.
		\item $F_\tau(v)$ is still well-defined for $\tau\in H(\diver;\Omega)$ and $v\in H^1(\Omega;\R^d)$.
		\item $F_\tau$ depends only on the trace $\gamma v\in H^{1/2}(\Gamma;\R^d)$ and not on $v$ itself.
		\item Therefore, $F_\tau$ induces a continuous linear functional on $H^{1/2}(\Gamma;\R^d)$:
		      \begin{equation*}
			      \langle \gamma_{\mathrm n}(\tau),\,\gamma v\rangle_\Gamma := F_\tau(v),
			      \qquad \gamma_{\mathrm n}(\tau)\in H^{-1/2}(\Gamma;\R^d).
		      \end{equation*}
		\item In the smooth case, $\gamma_{\mathrm n}(\tau)$ coincides with the classical traction $(\gamma \tau)\,\mathrm n$.
	\end{itemize}
\end{frame}

\begin{frame}{Stress Regularity Assumption}
	\begin{itemize}
		\item Thus, we further assume $\sigma(u)\in H(\diver;\Omega)$ which yields $\sigma(u)\mathrm{n}:= \gamma_\mathrm{n}(\sigma(u))\in H^{-1/2}(\Gamma;\R^d)$.
		\item Since $\sigma(u)$ is symmetric, we have
		      \begin{equation*}
			      \sigma(u):\nabla v = \sigma(u):\varepsilon(v) \qquad\forall\, v\in V.
		      \end{equation*}
		\item Therefore
		      \begin{equation*}
			      \langle \sigma(u)\mathrm n,\gamma v\rangle_\Gamma
			      =
			      \int_\Omega \sigma(u):\varepsilon(v)\;\dx
			      +
			      \int_\Omega \diver\sigma(u)\cdot v\;\dx
			      \qquad
			      \forall\, v\in V.
		      \end{equation*}
		\item {\em Useful identity}: If $-\diver \sigma(u) = f$ almost everywhere in $\Omega$, then
		      \begin{equation*}
			      a(u,v)-\ell(v) = \langle \sigma(u)\mathrm n,\gamma v\rangle_\Gamma-\int_{\Gamma_N} F\cdot (\gamma v)\;\ds,\qquad \forall\, v\in V.
		      \end{equation*}
	\end{itemize}
\end{frame}

\begin{frame}{A Hybrid Formulation}
	\begin{problem}[Signorini's Problem -- Hybrid Formulation]
	For given $f\in L^2(\Omega;\R^d)$ and $F\in L^2(\Gamma_N;\R^d)$, find $u\in K$ such that
	\begin{subequations}\label{eq:hybrid}
		\begin{empheq}[left=\empheqlbrace]{alignat=2}
			-\diver\sigma(u) &= f
			&\qquad& \text{a.e.\ in }\Omega, \label{eq:hybrid_bulk}\\
			\langle \sigma(u)\mathrm n,\gamma u\rangle_\Gamma
			&= \mathrlap{\int_{\Gamma_N} F\cdot (\gamma u)\;\ds,}
			&& \label{eq:hybrid_eq}\\
			\langle \sigma(u)\mathrm n,\gamma v\rangle_\Gamma
			&\ge \int_{\Gamma_N} F\cdot (\gamma v)\;\ds
			&\qquad& \forall\, v\in K.
			\label{eq:hybrid_ineq}
		\end{empheq}
	\end{subequations}
	\end{problem}
\end{frame}

\begin{frame}{Equivalence of Hybrid and Weak Formulation}
	\begin{proposition}[Equivalence of Hybrid and Weak Formulation]
		Assume $\sigma(u)\in H(\diver;\Omega)$. Then $u\in K$ solves the hybrid problem above if and only if it solves the weak problem
		\begin{equation*}
			\text{Find }u\in K\text{ such that}\qquad a(u,v-u)\ge\ell(v-u)\qquad \forall v\in K.
		\end{equation*}
	\end{proposition}
	\textbf{\em Proof:} $\ \textit{Hybrid }\Rightarrow\textit{ Weak}$:
	\begin{itemize}
		\item Subtract \eqref{eq:hybrid_eq} from \eqref{eq:hybrid_ineq}:
		      \begin{equation*}
			      \langle \sigma(u)\mathrm n,\gamma (v-u)\rangle_\Gamma - \int_{\Gamma_N}F\cdot \gamma (v-u)\;\ds \ge 0\qquad\forall\, v\in K.
		      \end{equation*}
		\item Insert the identity from the above:
		      \begin{equation*}
			      a(u,v-u)-\ell(v-u)\ge 0
			      \qquad\forall\,v\in K,
		      \end{equation*}
		      i.e.\ the weak formulation.
	\end{itemize}
\end{frame}

\begin{frame}{Equivalence of Hybrid and Weak Formulation}
	\begin{proposition}[Equivalence of Hybrid and Weak Formulation]
		Assume $\sigma(u)\in H(\diver;\Omega)$. Then $u\in K$ solves the hybrid problem above if and only if it solves the weak problem
		\begin{equation*}
			\text{Find }u\in K\text{ such that}\qquad a(u,v-u)\ge\ell(v-u)\qquad \forall v\in K.
		\end{equation*}
	\end{proposition}
	\textbf{\em Proof:} $\ \textit{Weak }\Rightarrow\textit{ Hybrid (Bulk Equation)}$:
	\begin{itemize}
		\item Let $\varphi\in \cD(\Omega;\R^d)$. Then $u\pm t\varphi\in K$ for $t\in\R$ because $\gamma\varphi = 0$, hence the weak inequality implies
		      \begin{equation*}
			      a(u,\varphi)=\ell(\varphi)=\int_\Omega f\cdot \varphi\;\dx = \langle f,\,\varphi\rangle\qquad \forall\,\varphi\in \cD(\Omega;\R^d).
		      \end{equation*}
		\item The left hand side is
		      \begin{equation*}
			      a(u,\varphi) = \int_\Omega\sigma(u):\varepsilon(\varphi) = \int_\Omega\sigma(u):\nabla\varphi\;\dx = \langle -\diver\sigma(u),\,\varphi\rangle.
		      \end{equation*}
		\item Thus, $-\diver\sigma(u) = f$ in $\cD^\prime(\Omega;\R^d)$ and by $\sigma(u)\in H(\diver;\Omega)$, it holds in $L^2(\Omega;\R^d)$ as well.
	\end{itemize}
\end{frame}

\begin{frame}{Equivalence of Hybrid and Weak Formulation}
	\begin{proposition}[Equivalence of Hybrid and Weak Formulation]
		Assume $\sigma(u)\in H(\diver;\Omega)$. Then $u\in K$ solves the hybrid problem above if and only if it solves the weak problem
		\begin{equation*}
			\text{Find }u\in K\text{ such that}\qquad a(u,v-u)\ge\ell(v-u)\qquad \forall v\in K.
		\end{equation*}
	\end{proposition}
	\textbf{\em Proof:} $\ \textit{Weak }\Rightarrow\textit{ Hybrid (Hybrid Equality)}$:
	\begin{itemize}
		\item Since $K$ is a cone, we have $0,2u\in K$.
		\item Testing both in the variational inequality yields $a(u,u)\le \ell(u)$ and $a(u,u)\ge \ell(u)$ and thus
		      \begin{equation*}
			      a(u,u)= \ell(u).
		      \end{equation*}
		\item The identity earlier yields
		      \begin{equation*}
			      0 = a(u,u)-\ell(u) = \langle\sigma(u)\mathrm{n},\,\gamma u\rangle_\Gamma -\int_{\Gamma_N}F\cdot\gamma u\;\ds.
		      \end{equation*}
		\item This is exactly \eqref{eq:hybrid_eq}.
	\end{itemize}
\end{frame}

\begin{frame}{Equivalence of Hybrid and Weak Formulation}
	\begin{proposition}[Equivalence of Hybrid and Weak Formulation]
		Assume $\sigma(u)\in H(\diver;\Omega)$. Then $u\in K$ solves the hybrid problem above if and only if it solves the weak problem
		\begin{equation*}
			\text{Find }u\in K\text{ such that}\qquad a(u,v-u)\ge\ell(v-u)\qquad \forall v\in K.
		\end{equation*}
	\end{proposition}
	\textbf{\em Proof:} $\ \textit{Weak }\Rightarrow\textit{ Hybrid (Hybrid Inequality)}$:
	\begin{itemize}
		\item With the identity earlier, and the variational inequality, we have for all $v\in V$ that
		      \begin{equation*}
			      \langle \sigma(u)\mathrm n,\gamma(v-u)\rangle_\Gamma - \int_{\Gamma_N}F\cdot\gamma(v-u)\;\ds = a(u,v-u)-\ell(v-u)\ge 0.
		      \end{equation*}

		\item This means
		      \begin{equation*}
			      \langle\sigma(u)\mathrm{n},\,\gamma v\rangle_\Gamma\ge \int_{\Gamma_N}F\cdot \gamma v\;\ds + \underbrace{\left[\langle\sigma(u)\mathrm{n},\,\gamma u\rangle_\Gamma-\int_{\Gamma_N}F\cdot\gamma u\;\ds\right]}_{ = 0}\qquad \forall\,v\in V.
		      \end{equation*}
	\end{itemize}
\end{frame}

\subsection{Classical Strong Formulation}

\begin{frame}{Additional Regularity Assumptions and their Consequences}
	\begin{itemize}
		\item \textbf{\em Goal:} Formulate classical normal traces for the stress $\sigma(u)$ as for displacements: $v_\mathrm{n} = (\gamma v)\mathrm{n}$.
		\item Previously, $u\in H^1(\Omega;\R^d)$
		\item Now we assume $u\in H^s(\Omega;\R^d)$ with $s>3/2$.
		\item Then $\varepsilon(u)\in H^{s-1}(\Omega;\S^d)$.
		\item If $\cA$ sufficiently regular, then $\sigma(u) = \cA:\varepsilon(u)\in H^{s-1}(\Omega;\S^d)$.
		\item For example: $\cA$ constant or $\cA_{ijkl}\in W^{1,\infty}(\Omega)$ if $3/2<s\le 2$.
		\item Trace is defined: $\gamma(\sigma(u))\in H^{s-3/2}(\Gamma;\S^d)$.
		\item Here $\Gamma\in\cC^{m,1}$ with $m\ge\lceil s-1\rceil$ is required.
		\item Then $\mathrm{n}\in \cC^{m-1,1}(\Gamma;\R^d)$.
		\item Normal Trace: $t(u):=(\gamma \sigma(u))\,\mathrm{n}\in H^{s-3/2}(\Gamma;\R^d)$ is well-defined.
		\item If $\sigma(u)\in H(\diver;\Omega)$, the two concepts of normal traces coincide:
		      \begin{equation*}
			      \langle \sigma(u)\mathrm{n},\, \gamma v\rangle_\Gamma  = \int_\Gamma t(u)\cdot (\gamma v)\;\ds\qquad \forall\, v\in V.
		      \end{equation*}
	\end{itemize}
\end{frame}

\begin{frame}{Normal and Tangential Components}
	\begin{itemize}

		\item Normal and tangential splitting of $t(u)$ into
		      \begin{equation*}
			      \sigma_\mathrm{n}(u):=t(u)\cdot \mathrm{n}\in H^{s-3/2}(\Gamma),
			      \qquad
			      \sigma_t(u):=t(u)-\sigma_\mathrm{n}(u)\,\mathrm{n}\in H^{s-3/2}(\Gamma;\R^d).
		      \end{equation*}
		      such that
		      \begin{equation*}
			      t(u) = \sigma_\mathrm{n}(u)\mathrm{n} + \sigma_t(u).
		      \end{equation*}
		\item Furthermore, for $u\in H^{s}(\Omega;\R^d)$ we have $\gamma u\in H^{s-1/2}(\Gamma;\R^d)$ with normal/tangential splitting
		      \begin{equation*}
			      u_\mathrm{n}:= (\gamma u)\cdot \mathrm{n}\in H^{s-1/2}(\Gamma)\qquad\text{and}\qquad u_t:= \gamma u- u_\mathrm{n}\,\mathrm{n}\in H^{s-1/2}(\Gamma;\R^d)
		      \end{equation*}
		      such that
		      \begin{equation*}
			      \gamma u = u_\mathrm{n}\,\mathrm{n}+u_t.
		      \end{equation*}
	\end{itemize}
\end{frame}

\begin{frame}{Classical Strong Formulation under Additional Regularity}
	\begin{problem}[Signorini's Problem -- Classical Strong Formulation]
	For given $f\in L^2(\Omega;\R^d)$ and $F\in L^2(\Gamma_N;\R^d)$ find $u\in H^s(\Omega;\R^d)$ with $s>3/2$ such that the {\em small strain elasticity equations}
	\begin{subequations}\label{eq:strong_classical}
		\begin{empheq}[left=\empheqlbrace]{alignat=2}
			-\diver\sigma(u) &= f
			&\qquad& \text{in }\Omega, \label{eq:strong_classical_bulk}\\
			u &= 0
			&\qquad& \text{on }\Gamma_D, \label{eq:strong_classical_dirichlet}\\
			t(u) &= F
			&\qquad& \text{on }\Gamma_N, \label{eq:strong_classical_neumann}
		\end{empheq}
	\end{subequations}
	and the {\em contact conditions}
	\begin{subequations}\label{eq:strong_contact}
		\begin{empheq}[left=\empheqlbrace]{alignat=2}
			u_\mathrm{n} &\le 0
			&\qquad& \text{on }\Gamma_C, \label{eq:strong_contact_nonpen}\\
			\sigma_\mathrm{n}(u) &\le 0
			&\qquad& \text{on }\Gamma_C, \label{eq:strong_contact_sigma_n}\\
			\sigma_\mathrm{n}(u)\,u_\mathrm{n} &= 0
			&\qquad& \text{on }\Gamma_C, \label{eq:strong_contact_comp}\\
			\sigma_t(u) &= 0
			&\qquad& \text{on }\Gamma_C \label{eq:strong_contact_tangential}
		\end{empheq}
	\end{subequations}
	hold.
	\end{problem}
\end{frame}

\begin{frame}{Interpretation of the Classical Strong Formulation}
	\begin{itemize}
		\item {\em Equilibrium equation} $-\diver\sigma(u)=f$ in $\Omega$:
		      balance of internal stresses and body forces in the bulk.

		\item {\em Dirichlet condition} $u=0$ on $\Gamma_D$:
		      the body is clamped on $\Gamma_D$.

		\item {\em Neumann condition} $t(u)=F$ on $\Gamma_N$: boundary forces acting on the free part of the boundary.

		\item {\em Nonpenetration} $u_\mathrm{n}\le 0$ on $\Gamma_C$:
		      the body cannot penetrate the rigid foundation.

		\item {\em Compressive contact stress} $\sigma_\mathrm{n}(u)\le 0$ on $\Gamma_C$:
		      the foundation can exert only compressive normal forces.

		\item {\em Complementarity} $\sigma_\mathrm{n}(u)\,u_\mathrm{n}=0$ on $\Gamma_C$: either there is no contact pressure, or the body is in contact, but not both effects independently.

		\item {\em Frictionless contact} $\sigma_t(u)=0$ on $\Gamma_C$:
		      Foundation does not exert any tangential force on the body.
	\end{itemize}
\end{frame}

	\end{problem}

\section{Mixed Formulation}

\begin{frame}{Preparing for Mixed Formulation}
	\begin{itemize}
		\item Let $|\Gamma_N| = 0$.
		\item Define the contact trace space
		      \begin{equation*}
			      W_C
			      :=
			      \left\{
			      \gamma v|_{\Gamma_C}
			      \;\middle|\;
			      v\in V
			      \right\}
			      =
			      H^{1/2}_{00}(\Gamma_C;\R^d).
		      \end{equation*}
		\item Define contact normal and tangential trace spaces
		      \begin{equation*}
			      W_{C,\mathrm n}
			      :=
			      \left\{
			      w\cdot \mathrm{n}
			      \;\middle|\;
			      w\in W_C
			      \right\},\qquad W_{C,t}
			      :=
			      \left\{
			      w\in W_C
			      \;\middle|\;
			      w\cdot \mathrm{n} = 0
			      \right\}.
		      \end{equation*}

			      \langle \gamma_{\mathrm n,C}(\tau),\,g\rangle_{\Gamma_C}
			      :=
			      \langle \gamma_{\mathrm n}(\tau),\,\mathrm{Ext}_0(g)\rangle_\Gamma
			      \qquad
			      \forall\, g\in W_C.
		      \end{equation*}
		\item Set $\sigma_C(u)\mathrm{n}:= \gamma_{\mathrm{n},C}(\sigma(u))\in W_C^\prime$ with splitting
		      \begin{equation*}
			      \langle\sigma_C(u)\mathrm{n},\, w\rangle_{\Gamma_C} =\langle \sigma_{C,\mathrm{n}}(u),\, w_\mathrm{n}\rangle_{\Gamma_C}+\langle \sigma_{C,t}(u),\, w_t\rangle_{\Gamma_C},\qquad \forall\, w = w_\mathrm{n}\mathrm{n}+w_t\in W_C.
		      \end{equation*}
	\end{itemize}

\begin{frame}{Mixed Formulation}
	\begin{itemize}

		      \begin{equation*}
			      W_{C,\mathrm n}^-
			      :=
			      \left\{
			      \eta\in W_{C,\mathrm n}

		\item We treat $\sigma_{C,\mathrm{n}(u)}$ as an independent variable $\lambda\in \Lambda_C\subset W_{C,\mathrm{n}}^\prime$.

			a(u,v)-b(v,\lambda) &= \ell(v)
			&\qquad& \forall\,v\in V, \label{eq:mixed_eq}\\
			b(u,\mu-\lambda) &\ge 0
			&\qquad& \forall\,\mu\in \Lambda_C. \label{eq:mixed_ineq}

\begin{frame}{Equivalence of Weak and Mixed Formulation}
	\begin{proposition}[Equivalence of Weak and Mixed Formulation]
		Assume $|\Gamma_N|=0$. If $u$ solves the weak problem, then the pair $(u,\sigma_{C,\mathrm{n}}(u))$ solves the mixed problem. Conversely, if $(u,\lambda)$ solves the mixed problem. then $u$ solves the weak problem and $\lambda = \sigma_{C,\mathrm{n}}(u)$.
	\end{proposition}
\end{frame}
